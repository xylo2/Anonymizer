import argparse
import os
import sys


class RootContactsLauncher:
    """
    Encapsule l'analyse des arguments et l'exécution du pipeline
    de contacts depuis la racine du dépôt.
    """

    def __init__(self) -> None:
        """
        Initialise les chemins racine et du sous-projet contacts.
        """
        self.root_dir: str = os.path.dirname(os.path.abspath(__file__))
        self.project_dir: str = os.path.join(self.root_dir, "projet_contacts")

    @staticmethod
    def parse_args() -> argparse.Namespace:
        """
        Analyse les arguments de ligne de commande pour le lanceur racine.

        :return: Espace de noms contenant le chemin du fichier de configuration.
        """
        parser: argparse.ArgumentParser = argparse.ArgumentParser(
            description=(
                "Lance le pipeline d'extraction et d'anonymisation des contacts "
                "(projet_contacts) depuis la racine."
            )
        )
        parser.add_argument(
            "-c", "--config",
            default="config.ini",
            help="Chemin vers le fichier config.ini (défaut : config.ini).",
        )
        return parser.parse_args()

    def run(self) -> None:
        """
        Vérifie l'arborescence, configure sys.path et exécute le pipeline.
        """
        if not os.path.isdir(self.project_dir):
            raise FileNotFoundError(
                f"Sous-dossier introuvable : '{self.project_dir}'. "
                "Vérifiez l'arborescence de votre espace de travail."
            )

        if self.project_dir not in sys.path:
            sys.path.insert(0, self.project_dir)

        from google_contacts_parser.config_manager import ContactsConfig
        from google_contacts_parser.pipeline_runner import ContactsPipelineRunner

        args: argparse.Namespace = self.parse_args()
        config_path: str = (
            args.config
            if os.path.isabs(args.config)
            else os.path.join(self.root_dir, args.config)
        )

        if not os.path.exists(config_path):
            raise FileNotFoundError(
                f"Fichier de configuration introuvable : '{config_path}'"
            )

        config: ContactsConfig = ContactsConfig(config_path=config_path)
        runner: ContactsPipelineRunner = ContactsPipelineRunner(config=config)
        runner.run_all()


def main() -> None:
    """
    Point d'entrée racine pour exécuter google_contacts_parser.
    """
    launcher: RootContactsLauncher = RootContactsLauncher()
    launcher.run()


if __name__ == "__main__":
    try:
        main()
    except KeyboardInterrupt:
        print(
            "\n[ARRÊT] Traitement des contacts interrompu par l'utilisateur "
            "(Ctrl+C). Aucun fichier final modifié.",
            flush=True,
        )
        sys.exit(130)

import argparse
import os
import sys


class RootDocumentsLauncher:
    """
    Encapsule l'analyse des arguments et l'exécution du pipeline
    de documents depuis la racine du dépôt.
    """

    def __init__(self) -> None:
        """
        Initialise les chemins racine et du sous-projet documents.
        """
        self.root_dir: str = os.path.dirname(os.path.abspath(__file__))
        self.project_dir: str = os.path.join(self.root_dir, "projet_documents")

    @staticmethod
    def parse_args() -> argparse.Namespace:
        """
        Analyse les arguments de ligne de commande pour le lanceur racine.

        :return: Espace de noms contenant les options d'exécution.
        """
        parser: argparse.ArgumentParser = argparse.ArgumentParser(
            description=(
                "Lance le pipeline d'extraction et d'anonymisation de documents "
                "(projet_documents) depuis la racine."
            )
        )
        parser.add_argument(
            "-c", "--config",
            default="config.ini",
            help="Chemin vers le fichier config.ini (défaut : config.ini).",
        )
        parser.add_argument(
            "--sync-contacts",
            default=None,
            help="Chemin d'un contacts.txt externe à fusionner avant traitement.",
        )
        parser.add_argument(
            "--overwrite-contacts",
            action="store_true",
            help="Écrase totalement contacts.txt lors de la synchronisation.",
        )
        parser.add_argument(
            "--allow-sleep",
            action="store_true",
            help="Autorise le système à se mettre en veille pendant l'exécution.",
        )
        parser.add_argument(
            "--complete-manual",
            action="store_true",
            help=(
                "Complète automatiquement les lignes vides 'Prénom Nom = ' "
                "dans contacts_manuels.txt."
            ),
        )
        parser.add_argument(
            "--dry-run",
            action="store_true",
            help=(
                "Prévisualise les complétions de contacts_manuels.txt sans "
                "écrire sur le disque (à combiner avec --complete-manual)."
            ),
        )
        parser.add_argument(
            "--manual-only",
            action="store_true",
            help=(
                "Applique uniquement contacts_manuels.txt directement sur "
                "les fichiers déjà présents dans resultats_anonymisation."
            ),
        )
        return parser.parse_args()
    def run(self) -> None:
        """
        Vérifie l'arborescence, exécute les synchronisations éventuelles
        puis lance l'étape demandée (auto-complétion, passe manuelle ou pipeline complet).
        """
        if not os.path.isdir(self.project_dir):
            raise FileNotFoundError(
                f"Sous-dossier introuvable : '{self.project_dir}'. "
                "Vérifiez l'arborescence de votre espace de travail."
            )

        if self.project_dir not in sys.path:
            sys.path.insert(0, self.project_dir)

        from completer_contacts_manuels import ManualContactsAutoCompleter
        from doc_anonymizer.anonymization_runner import DocumentAnonymizationRunner
        from doc_anonymizer.config_manager import PipelineConfig
        from doc_anonymizer.pipeline_runner import DocAnonymizerPipelineRunner
        from sync_contacts import merge_and_sync_contacts

        args: argparse.Namespace = self.parse_args()
        config_path: str = (
            args.config
            if os.path.isabs(args.config)
            else os.path.join(self.root_dir, args.config)
        )

        if not os.path.exists(config_path):
            raise FileNotFoundError(
                f"Fichier de configuration introuvable : '{config_path}'"
            )

        config: PipelineConfig = PipelineConfig(config_path=config_path)

        if args.sync_contacts:
            print("=== SYNCHRONISATION DE CONTACTS.TXT ===", flush=True)
            merge_and_sync_contacts(
                source_path=args.sync_contacts,
                dest_path=config.contacts_file,
                preserve_manual_rules=not args.overwrite_contacts,
            )
            print("", flush=True)

        if args.complete_manual:
            completer: ManualContactsAutoCompleter = ManualContactsAutoCompleter(
                config=config, dry_run=args.dry_run
            )
            completer.process()
            if not args.manual_only:
                return
            if args.dry_run:
                print(
                    "\n[INFO] Mode --dry-run actif : passe --manual-only "
                    "ignorée tant que les règles ne sont pas enregistrées.",
                    flush=True,
                )
                return
            print("", flush=True)

        if args.manual_only:
            manual_runner: DocumentAnonymizationRunner = (
                DocumentAnonymizationRunner(config=config, manual_only=True)
            )
            manual_runner.run_on_existing_results()
            return

        runner: DocAnonymizerPipelineRunner = DocAnonymizerPipelineRunner(
            config=config,
            prevent_sleep=not args.allow_sleep,
        )
        runner.run_all()


def main() -> None:
    """
    Point d'entrée racine pour exécuter doc_anonymizer.
    """
    launcher: RootDocumentsLauncher = RootDocumentsLauncher()
    launcher.run()


if __name__ == "__main__":
    try:
        main()
    except KeyboardInterrupt:
        print(
            "\n[ARRÊT] Pipeline interrompu proprement par l'utilisateur "
            "(Ctrl+C).",
            flush=True,
        )
        sys.exit(130)

import configparser
import os
import sys
import unicodedata
from typing import Dict, List, Set

_PROJ_CONTACTS_DIR: str = os.path.join(
    os.path.dirname(os.path.dirname(os.path.abspath(__file__))),
    "projet_contacts",
)
if os.path.isdir(_PROJ_CONTACTS_DIR) and _PROJ_CONTACTS_DIR not in sys.path:
    sys.path.insert(0, _PROJ_CONTACTS_DIR)

from google_contacts_parser.geo_data import FrenchGeoCatalog


class ContactsConfig:
    """
    Charge et expose de manière typée la configuration du pipeline de contacts
    depuis le fichier config.ini unifié.
    """

    def __init__(self, config_path: str = "config.ini") -> None:
        """
        Initialise les paramètres par défaut puis charge le fichier INI.

        :param config_path: Chemin vers le fichier config.ini.
        """
        self.config_path: str = os.path.abspath(config_path)
        self._base_dir: str = os.path.dirname(self.config_path)
        self._parser: configparser.ConfigParser = configparser.ConfigParser()

        self.csv_directory: str = os.path.join(self._base_dir, "export_contacts")
        self.csv_prefixes: List[str] = ["contacts", "autres contacts"]
        self.first_names_file: str = os.path.join(self._base_dir, "prenoms.txt")
        self.output_file: str = os.path.join(self._base_dir, "contacts_indexes.txt")
        self.anonymized_output_file: str = os.path.join(
            self._base_dir, "contacts_anonymises.txt"
        )
        self.deduplicated_output_file: str = os.path.join(
            self._base_dir, "contacts_dedoublonnes.txt"
        )
        self.inverted_output_file: str = os.path.join(self._base_dir, "contacts.txt")
        self.encoding: str = "utf-8-sig"

        self.clean_phone_spaces: bool = False
        self.lowercase_emails: bool = True
        self.separator: str = "="
        self.merge_separator: str = "et"

        self.ignored_domains: Set[str] = set()
        self.entity_prefixes: Set[str] = set()
        self.company_keywords: Set[str] = set()
        self.extra_locations: Set[str] = set()
        self.domain_aliases: Dict[str, str] = {}

        self._load()

        base_locs: Set[str] = self._collect_geo_catalog_locations()
        self.all_excluded_locations_norm: Set[str] = {
            self._normalize_token(loc)
            for loc in (base_locs | self.extra_locations)
            if loc.strip()
        }
        self.all_ignored_domains_norm: Set[str] = {
            self._normalize_token(dom)
            for dom in self.ignored_domains
            if dom.strip()
        }
        self.all_company_keywords_norm: Set[str] = {
            self._normalize_token(kw)
            for kw in self.company_keywords
            if kw.strip()
        }
    @staticmethod
    def _collect_geo_catalog_locations() -> Set[str]:
        """
        Récupère dynamiquement tous les lieux définis dans FrenchGeoCatalog.

        :return: Ensemble des lieux français de référence.
        """
        collected: Set[str] = set()
        if hasattr(FrenchGeoCatalog, "get_all_locations"):
            collected.update(FrenchGeoCatalog.get_all_locations())
        for attr_name in dir(FrenchGeoCatalog):
            if attr_name.startswith("_"):
                continue
            val = getattr(FrenchGeoCatalog, attr_name)
            if isinstance(val, (set, list, tuple)):
                collected.update(str(item) for item in val if isinstance(item, str))
        return collected

    @staticmethod
    def _normalize_token(text: str) -> str:
        """
        Normalise un terme en minuscules sans accents.

        :param text: Texte source.
        :return: Texte normalisé.
        """
        nfkd: str = unicodedata.normalize("NFKD", text.strip().lower())
        return "".join(ch for ch in nfkd if not unicodedata.combining(ch))

    def _resolve_path(self, raw_path: str) -> str:
        """
        Résout un chemin relativement au dossier contenant config.ini.

        :param raw_path: Chemin lu dans le fichier INI.
        :return: Chemin absolu normalisé.
        """
        cleaned: str = raw_path.strip()
        if os.path.isabs(cleaned):
            return os.path.normpath(cleaned)
        return os.path.normpath(os.path.join(self._base_dir, cleaned))

    def _parse_csv_list(self, raw_val: str) -> List[str]:
        """
        Convertit une chaîne séparée par des virgules en liste nettoyée.

        :param raw_val: Chaîne brute issue du INI.
        :return: Liste des éléments non vides.
        """
        return [item.strip() for item in raw_val.split(",") if item.strip()]

    def _load(self) -> None:
        """
        Lit les sections [files], [formatting], [anonymizer] et [domain_aliases].
        """
        if not os.path.exists(self.config_path):
            return

        self._parser.read(self.config_path, encoding="utf-8")
        if "files" in self._parser:
            f = self._parser["files"]
            if "csv_directory" in f:
                self.csv_directory = self._resolve_path(f["csv_directory"])
            if "csv_prefixes" in f:
                self.csv_prefixes = [
                    p.lower() for p in self._parse_csv_list(f["csv_prefixes"])
                ]
            if "first_names_file" in f:
                self.first_names_file = self._resolve_path(f["first_names_file"])
            if "output_file" in f:
                self.output_file = self._resolve_path(f["output_file"])
            if "anonymized_output_file" in f:
                self.anonymized_output_file = self._resolve_path(
                    f["anonymized_output_file"]
                )
            if "deduplicated_output_file" in f:
                self.deduplicated_output_file = self._resolve_path(
                    f["deduplicated_output_file"]
                )
            if "inverted_output_file" in f:
                self.inverted_output_file = self._resolve_path(
                    f["inverted_output_file"]
                )
            self.encoding = f.get("encoding", fallback=self.encoding)

        if "formatting" in self._parser:
            fmt = self._parser["formatting"]
            self.clean_phone_spaces = fmt.getboolean(
                "clean_phone_spaces", fallback=self.clean_phone_spaces
            )
            self.lowercase_emails = fmt.getboolean(
                "lowercase_emails", fallback=self.lowercase_emails
            )
            self.separator = fmt.get("separator", fallback=self.separator)
            self.merge_separator = fmt.get(
                "merge_separator", fallback=self.merge_separator
            )

        if "anonymizer" in self._parser:
            az = self._parser["anonymizer"]
            if "ignored_domains" in az:
                self.ignored_domains = {
                    d.lower() for d in self._parse_csv_list(az["ignored_domains"])
                }
            if "entity_prefixes" in az:
                self.entity_prefixes = {
                    p.lower() for p in self._parse_csv_list(az["entity_prefixes"])
                }
            if "company_keywords" in az:
                self.company_keywords = {
                    k.lower() for k in self._parse_csv_list(az["company_keywords"])
                }
            if "extra_locations" in az:
                self.extra_locations = {
                    loc.lower() for loc in self._parse_csv_list(az["extra_locations"])
                }

        if "domain_aliases" in self._parser:
            self.domain_aliases = {
                k.strip().lower(): v.strip()
                for k, v in self._parser["domain_aliases"].items()
                if k.strip() and v.strip()
            }

    def get_matching_csv_files(self) -> List[str]:
        """
        Recherche dans csv_directory tous les fichiers .csv dont le nom
        commence par l'un des préfixes configurés.

        :return: Liste triée des chemins complets des fichiers CSV trouvés.
        """
        if not os.path.isdir(self.csv_directory):
            return []
        matched: List[str] = []
        for fname in sorted(os.listdir(self.csv_directory)):
            lower_name: str = fname.lower()
            if lower_name.endswith(".csv") and any(
                lower_name.startswith(pref) for pref in self.csv_prefixes
            ):
                matched.append(os.path.join(self.csv_directory, fname))
        return matched

    def get_csv_files(self) -> List[str]:
        """
        Alias rétrocompatible de get_matching_csv_files().

        :return: Liste triée des chemins complets des fichiers CSV trouvés.
        """
        return self.get_matching_csv_files()

import argparse
import os
import sys

CURRENT_DIR: str = os.path.dirname(os.path.abspath(__file__))
if CURRENT_DIR not in sys.path:
    sys.path.insert(0, CURRENT_DIR)

from google_contacts_parser.config_manager import ContactsConfig
from google_contacts_parser.pipeline_runner import ContactsPipelineRunner


def resolve_config_path(raw_config: str) -> str:
    """
    Résout le chemin vers config.ini en vérifiant d'abord le chemin fourni,
    puis le répertoire parent racine si le script est exécuté depuis le sous-dossier.

    :param raw_config: Chemin brut passé en argument.
    :return: Chemin absolu existant vers config.ini.
    """
    if os.path.isabs(raw_config):
        if os.path.exists(raw_config):
            return raw_config
        raise FileNotFoundError(f"Fichier de configuration introuvable : '{raw_config}'")

    local_candidate: str = os.path.abspath(raw_config)
    if os.path.exists(local_candidate):
        return local_candidate

    parent_candidate: str = os.path.normpath(
        os.path.join(CURRENT_DIR, "..", raw_config)
    )
    if os.path.exists(parent_candidate):
        return parent_candidate

    raise FileNotFoundError(
        f"Fichier de configuration '{raw_config}' introuvable (cherché dans "
        f"'{local_candidate}' et '{parent_candidate}')."
    )


def parse_args() -> argparse.Namespace:
    """
    Analyse les arguments de ligne de commande pour le pipeline de contacts.

    :return: Espace de noms contenant le chemin vers le fichier INI.
    """
    parser: argparse.ArgumentParser = argparse.ArgumentParser(
        description=(
            "Exécute l'extraction, l'anonymisation et le dédoublonnage "
            "des exports Google Contacts et Microsoft 365."
        )
    )
    parser.add_argument(
        "-c", "--config",
        default="config.ini",
        help="Chemin vers le fichier config.ini (par défaut : config.ini).",
    )
    return parser.parse_args()


def main() -> None:
    """
    Point d'entrée principal du programme de contacts.
    """
    args: argparse.Namespace = parse_args()
    config_path: str = resolve_config_path(args.config)
    config: ContactsConfig = ContactsConfig(config_path=config_path)
    runner: ContactsPipelineRunner = ContactsPipelineRunner(config=config)
    runner.run_all()


if __name__ == "__main__":
    try:
        main()
    except KeyboardInterrupt:
        print(
            "\n[ARRÊT] Traitement des contacts interrompu par l'utilisateur "
            "(Ctrl+C). Aucun fichier final modifié.",
            flush=True,
        )
        sys.exit(130)

        self.name_to_last_name: Dict[str, str] = {}
        self.name_to_first_email: Dict[str, str] = {}

    def _load_first_names(self) -> Set[str]:
        """
        Charge la liste des prénoms fréquents pour éviter de confondre un prénom avec une enseigne.

        :return: Ensemble des prénoms normalisés.
        """
        names: Set[str] = {
            ContactTextUtils.normalize_text(extra) for extra in self.DEFAULT_EXTRA_FIRST_NAMES if extra.strip()
        }
        if not os.path.exists(self.config.first_names_file):
            return names
        with open(self.config.first_names_file, mode="r", encoding="utf-8") as f:
            for line in f:
                cleaned: str = ContactTextUtils.normalize_text(line)
                if cleaned and not cleaned.startswith("#"):
                    names.add(cleaned)
        return names

    def _extract_row_identity(self, row: Dict[str, str]) -> Tuple[str, str, str]:
        """
        Extrait le prénom (simple ou composé avec Middle Name), le nom de famille et le nom complet.

        :param row: Dictionnaire représentant une ligne du CSV Google Contacts.
        :return: Tuple (prénom, nom_de_famille, nom_complet).
        """
        raw_first: str = (row.get("First Name") or row.get("Given Name") or row.get("Prénom") or "").strip()
        raw_middle: str = (row.get("Middle Name") or row.get("Additional Name") or row.get("Deuxième prénom") or "").strip()
        first_name: str = f"{raw_first}-{raw_middle}" if (raw_first and raw_middle) else (raw_first or raw_middle)
        last_name: str = (row.get("Last Name") or row.get("Family Name") or row.get("Nom de famille") or row.get("Nom") or "").strip()

        full_name: str = f"{first_name} {last_name}".strip()
        if not full_name:
            full_name = (row.get("Name") or row.get("Nom complet") or "").strip()
            if full_name and " " in full_name and not last_name:
                p_first, p_last = full_name.split(" ", 1)
                first_name, last_name = p_first.strip(), p_last.strip()

        return first_name, last_name, full_name

    def _extract_first_email(self, row: Dict[str, str], fieldnames: List[str]) -> str:
        """
        Récupère la première adresse e-mail renseignée pour un contact dans la ligne CSV.

        :param row: Dictionnaire représentant une ligne CSV.
        :param fieldnames: Liste des colonnes du fichier CSV.
        :return: Première adresse e-mail trouvée, ou chaîne vide.
        """
        for col in fieldnames:
            lower_col: str = col.lower()
            if "mail" in lower_col and "type" not in lower_col and "label" not in lower_col:
                raw_val: str = (row.get(col) or "").strip()
                if raw_val:
                    first_val: str = next(iter(raw_val.split(":::")), "").strip()
                    if first_val:
                        return first_val.lower() if self.config.lowercase_emails else first_val
        return ""
    def _clean_label_text(self, raw_label: str) -> str:
        """
        Retire les mentions de fournisseurs de messagerie et abrège les emplacements
        géographiques (départements, régions, grandes villes et exceptions configurées)
        par leur première lettre suivie d'un point.

        :param raw_label: Libellé brut du contact ou de l'entité.
        :return: Libellé nettoyé prêt pour l'anonymisation.
        """
        without_provider: str = ContactTextUtils.strip_provider_suffix(raw_label, self.config.ignored_domains)
        return ContactTextUtils.abbreviate_locations_in_text(
            without_provider, self.config.all_excluded_locations_norm
        )

    @staticmethod
    def _detect_csv_delimiter(sample: str) -> str:
        """
        Détecte automatiquement si le CSV utilise ',' ou ';'.

        :param sample: Début du fichier CSV.
        :return: Délimiteur détecté.
        """
        lines: List[str] = sample.splitlines()
        first_line: str = next(iter(lines), "")
        return ";" if first_line.count(";") > first_line.count(",") else ","

    def _load_csv_metadata(self) -> None:
        """
        Parcourt tous les fichiers CSV détectés (virgule ou point-virgule) pour mémoriser
        les noms de famille et la première adresse e-mail de chaque contact.
        """
        for csv_path in self.config.get_csv_files():
            if not os.path.exists(csv_path):
                continue
            with open(csv_path, mode="r", encoding=self.config.encoding, newline="") as f:
                sample: str = f.read(4096)
                f.seek(0)
                delimiter: str = self._detect_csv_delimiter(sample)
                reader: csv.DictReader = csv.DictReader(f, delimiter=delimiter)
                if not reader.fieldnames:
                    continue
                fieldnames: List[str] = list(reader.fieldnames)
                for row in reader:
                    _, last_name, full_name = self._extract_row_identity(row)
                    if not full_name:
                        continue
                    cleaned_full: str = self._clean_label_text(full_name)
                    base_name, _, is_rfc = ContactTextUtils.clean_display_name(cleaned_full)
                    if not is_rfc and last_name:
                        cleaned_last_raw: str = self._clean_label_text(last_name)
                        clean_last, _, _ = ContactTextUtils.clean_display_name(cleaned_last_raw)
                        if (
                            clean_last
                            and ContactTextUtils.normalize_text(clean_last)
                            not in self.config.all_excluded_locations_norm
                        ):
                            self.known_last_names_norm.add(ContactTextUtils.normalize_text(clean_last))
                            for part in re.split(r"[\s\-]+", clean_last):
                                if len(part) >= 2:
                                    self.known_last_names_norm.add(ContactTextUtils.normalize_text(part))
                            if base_name not in self.name_to_last_name:
                                self.name_to_last_name[base_name] = clean_last

                    first_email: str = self._extract_first_email(row, fieldnames)
                    if first_email:
                        if cleaned_full not in self.name_to_first_email:
                            self.name_to_first_email[cleaned_full] = first_email
                        if base_name not in self.name_to_first_email:
                            self.name_to_first_email[base_name] = first_email
    def _format_domain_display(self, domain_root: str) -> str:
        """
        Formate l'affichage d'un nom de domaine en consultant les alias configurés,
        en remplaçant les tirets par des espaces et en abrégeant les lieux géographiques.

        :param domain_root: Racine du domaine extraite sans extension.
        :return: Libellé lisible pour l'affichage entre parenthèses.
        """
        lower_root: str = domain_root.strip().lower()
        if lower_root in self.config.domain_aliases:
            display_val: str = self.config.domain_aliases[lower_root]
        elif "-" in lower_root:
            words: List[str] = [w.capitalize() for w in lower_root.split("-") if w]
            display_val = " ".join(words)
        else:
            display_val = domain_root

        return ContactTextUtils.abbreviate_locations_in_text(
            display_val, self.config.all_excluded_locations_norm
        )

    def _get_domain_suffix(self, email: str, contact_last_name: str, base_name: str = "") -> str:
        """
        Retourne ' (domaine)' sans extension si le domaine n'est ni un nom de famille,
        ni un fournisseur grand public, ni un lieu seul, ni déjà inclus dans le libellé.

        :param email: Adresse e-mail à évaluer.
        :param contact_last_name: Nom de famille du contact associé.
        :param base_name: Nom ou enseigne complète pour éviter la redondance avec le domaine.
        :return: Chaîne ' (domaine)' ou chaîne vide.
        """
        domain_root: str = ContactTextUtils.extract_domain_root(email)
        if not domain_root or domain_root in self.config.ignored_domains:
            return ""

        norm_domain: str = ContactTextUtils.normalize_text(domain_root)
        if norm_domain in self.known_last_names_norm or norm_domain in self.config.all_excluded_locations_norm:
            return ""
        if contact_last_name and norm_domain == ContactTextUtils.normalize_text(contact_last_name):
            return ""
        if base_name:
            norm_base: str = ContactTextUtils.normalize_text(base_name)
            if norm_domain == norm_base or (len(norm_domain) >= 4 and norm_domain in norm_base):
                return ""

        formatted_domain: str = self._format_domain_display(domain_root)
        return f" ({formatted_domain})"

    def _anonymize_full_name(self, base_name: str, extra_chars: int = 0) -> Tuple[str, str]:
        """
        Formate le prénom et remplace le nom de famille par ses initiales suivies d'un point.
        Gère également les couples séparés par 'et' ou '&' (ex: 'Jean Dupont et Marie Martin').

        :param base_name: Chaîne 'Prénom Nom' nettoyée de toute parenthèse.
        :param extra_chars: Nombre de lettres supplémentaires sur le nom en cas d'homonymie.
        :return: Tuple (nom_anonymisé, nom_de_famille_identifié).
        """
        couple_match: Optional[re.Match[str]] = re.match(
            r"^(.+?)\s+(et|&)\s+(.+)$", base_name, flags=re.IGNORECASE
        )
        if couple_match:
            left_part: str = couple_match.group(1).strip()
            conj: str = couple_match.group(2)
            right_part: str = couple_match.group(3).strip()
            anon_left, last_left = self._anonymize_full_name(left_part, extra_chars=extra_chars)
            anon_right, last_right = self._anonymize_full_name(right_part, extra_chars=extra_chars)
            return f"{anon_left} {conj} {anon_right}", f"{last_left} {last_right}".strip()

        last_name: Optional[str] = self.name_to_last_name.get(base_name)
        if last_name and base_name.endswith(last_name):
            first_part: str = ContactTextUtils.format_first_name(base_name[: -len(last_name)].strip())
            initials: str = ContactTextUtils.format_initials(last_name, extra_chars=extra_chars)
            anonymized: str = f"{first_part} {initials}".strip() if first_part else initials
            return anonymized, last_name

        if " " in base_name:
            first_part_raw, guessed_last = base_name.rsplit(" ", 1)
            first_part = ContactTextUtils.format_first_name(first_part_raw)
            initials = ContactTextUtils.format_initials(guessed_last, extra_chars=extra_chars)
            return f"{first_part} {initials}".strip(), guessed_last

        return ContactTextUtils.format_first_name(base_name), ""
    def _resolve_homonym_collision(
        self,
        base_name: str,
        initial_name: str,
        last_name: str,
        suffix: str,
        assigned_owners: Dict[str, str],
        identity_to_display: Dict[str, str],
    ) -> str:
        """
        Vérifie si un libellé anonymisé est déjà pris par une autre personne : tente d'abord
        d'étendre le nom à 2 caractères maximum, puis remplace le point final par un tiret suivi
        d'un chiffre (ex: Jean Du-2, Jean Du-3).

        :param base_name: Identité complète d'origine ('Prénom Nom').
        :param initial_name: Nom anonymisé sans le suffixe parenthèse ('Jean D.').
        :param last_name: Nom de famille d'origine.
        :param suffix: Suffixe éventuel entre parenthèses (' (domaine)' ou '').
        :param assigned_owners: Dictionnaire associant chaque libellé sorti à son identité normalisée.
        :param identity_to_display: Cache associant chaque identité normalisée à son libellé retenu.
        :return: Libellé anonymisé unique pour cette personne.
        """
        norm_identity: str = ContactTextUtils.normalize_text(base_name)
        if norm_identity in identity_to_display:
            return identity_to_display[norm_identity]

        initial_display: str = ContactTextUtils.sanitize_final_line(f"{initial_name}{suffix}")
        owner: Optional[str] = assigned_owners.get(initial_display)
        if owner is None or owner == norm_identity:
            assigned_owners[initial_display] = norm_identity
            identity_to_display[norm_identity] = initial_display
            return initial_display

        base_name_for_counter: str = initial_name
        if last_name:
            clean_alpha: str = "".join(c for c in last_name if c.isalpha())
            if len(clean_alpha) >= 2:
                alt_name, _ = self._anonymize_full_name(base_name, extra_chars=1)
                candidate: str = ContactTextUtils.sanitize_final_line(f"{alt_name}{suffix}")
                cand_owner: Optional[str] = assigned_owners.get(candidate)
                if cand_owner is None or cand_owner == norm_identity:
                    assigned_owners[candidate] = norm_identity
                    identity_to_display[norm_identity] = candidate
                    print(f"[HOMONYME] '{base_name}' -> '{candidate}' (conflit avec '{initial_display}')")
                    return candidate
                base_name_for_counter = alt_name

        stem_without_trailing_dot: str = base_name_for_counter.rstrip(".")
        counter: int = 2
        while True:
            fallback: str = ContactTextUtils.sanitize_final_line(f"{stem_without_trailing_dot}-{counter}{suffix}")
            fb_owner: Optional[str] = assigned_owners.get(fallback)
            if fb_owner is None or fb_owner == norm_identity:
                assigned_owners[fallback] = norm_identity
                identity_to_display[norm_identity] = fallback
                print(f"[HOMONYME] '{base_name}' -> '{fallback}' (conflit avec '{base_name_for_counter}')")
                return fallback
            counter += 1

    def process(self) -> int:
        """
        Exécute l'anonymisation complète et écrit le nouveau fichier de sortie.

        :return: Nombre de lignes écrites dans le fichier anonymisé.
        """
        self._load_csv_metadata()
        if not os.path.exists(self.config.output_file):
            raise FileNotFoundError(f"Le fichier d'indexation '{self.config.output_file}' est introuvable.")

        assigned_owners: Dict[str, str] = {}
        identity_to_display: Dict[str, str] = {}
        output_lines: List[str] = []

        with open(self.config.output_file, mode="r", encoding="utf-8") as in_file:
            for raw_line in in_file:
                line: str = raw_line.rstrip("\r\n")
                if not line or self.config.separator not in line:
                    continue
                contact_val, raw_full_name = line.split(self.config.separator, 1)
                contact_val = contact_val.strip()
                cleaned_full_name: str = self._clean_label_text(raw_full_name)
                base_name, existing_paren, is_rfc = ContactTextUtils.clean_display_name(cleaned_full_name)

                if is_rfc:
                    clean_rfc: str = ContactTextUtils.sanitize_final_line(base_name)
                    output_lines.append(f"{contact_val}{self.config.separator}{clean_rfc}\n")
                    continue

                if ContactTextUtils.is_non_person_entity(
                    base_name,
                    contact_val,
                    self.config.entity_prefixes,
                    self.config.company_keywords,
                    self.frequent_first_names,
                ):
                    entity_label: str = ContactTextUtils.format_entity_label(base_name)
                    if existing_paren:
                        suffix: str = existing_paren
                    elif "@" in contact_val:
                        suffix = self._get_domain_suffix(contact_val, "", base_name)
                    else:
                        first_email: str = self.name_to_first_email.get(base_name, "")
                        suffix = self._get_domain_suffix(first_email, "", base_name) if first_email else ""
                    final_entity: str = ContactTextUtils.sanitize_final_line(f"{entity_label}{suffix}")
                    output_lines.append(f"{contact_val}{self.config.separator}{final_entity}\n")
                    continue

                anonymized_name, last_name = self._anonymize_full_name(base_name)
                if existing_paren:
                    suffix = existing_paren
                elif "@" in contact_val:
                    suffix = self._get_domain_suffix(contact_val, last_name, base_name)
                else:
                    first_email = self.name_to_first_email.get(base_name, "")
                    suffix = self._get_domain_suffix(first_email, last_name, base_name) if first_email else ""

                final_right: str = self._resolve_homonym_collision(
                    base_name=base_name,
                    initial_name=anonymized_name,
                    last_name=last_name,
                    suffix=suffix,
                    assigned_owners=assigned_owners,
                    identity_to_display=identity_to_display,
                )
                output_lines.append(f"{contact_val}{self.config.separator}{final_right}\n")

        with open(self.config.anonymized_output_file, mode="w", encoding="utf-8", newline="\n") as out_file:
            out_file.writelines(output_lines)

        return len(output_lines)

import configparser
import os
import unicodedata
from typing import Dict, List, Set
from google_contacts_parser.geo_data import FrenchGeoCatalog


class ContactsConfig:
    """
    Charge et expose de manière typée la configuration du pipeline de contacts
    depuis le fichier config.ini unifié.
    """

    def __init__(self, config_path: str = "config.ini") -> None:
        """
        Initialise les paramètres par défaut puis charge le fichier INI.

        :param config_path: Chemin vers le fichier config.ini.
        """
        self.config_path: str = os.path.abspath(config_path)
        self._base_dir: str = os.path.dirname(self.config_path)
        self._parser: configparser.ConfigParser = configparser.ConfigParser()

        self.csv_directory: str = os.path.join(self._base_dir, "export_contacts")
        self.csv_prefixes: List[str] = ["contacts", "autres contacts"]
        self.first_names_file: str = os.path.join(self._base_dir, "prenoms.txt")
        self.output_file: str = os.path.join(self._base_dir, "contacts_indexes.txt")
        self.anonymized_output_file: str = os.path.join(
            self._base_dir, "contacts_anonymises.txt"
        )
        self.deduplicated_output_file: str = os.path.join(
            self._base_dir, "contacts_dedoublonnes.txt"
        )
        self.inverted_output_file: str = os.path.join(self._base_dir, "contacts.txt")
        self.encoding: str = "utf-8-sig"

        self.clean_phone_spaces: bool = False
        self.lowercase_emails: bool = True
        self.separator: str = "="
        self.merge_separator: str = "et"

        self.ignored_domains: Set[str] = set()
        self.entity_prefixes: Set[str] = set()
        self.company_keywords: Set[str] = set()
        self.extra_locations: Set[str] = set()
        self.domain_aliases: Dict[str, str] = {}

        self._load()

        base_locs: Set[str] = self._collect_geo_catalog_locations()
        self.all_excluded_locations_norm: Set[str] = {
            self._normalize_token(loc)
            for loc in (base_locs | self.extra_locations)
            if loc.strip()
        }
        self.all_ignored_domains_norm: Set[str] = {
            self._normalize_token(dom)
            for dom in self.ignored_domains
            if dom.strip()
        }
        self.all_company_keywords_norm: Set[str] = {
            self._normalize_token(kw)
            for kw in self.company_keywords
            if kw.strip()
        }
    @staticmethod
    def _collect_geo_catalog_locations() -> Set[str]:
        """
        Récupère dynamiquement tous les lieux définis dans FrenchGeoCatalog.

        :return: Ensemble des lieux français de référence.
        """
        collected: Set[str] = set()
        if hasattr(FrenchGeoCatalog, "get_all_locations"):
            collected.update(FrenchGeoCatalog.get_all_locations())
        for attr_name in dir(FrenchGeoCatalog):
            if attr_name.startswith("_"):
                continue
            val = getattr(FrenchGeoCatalog, attr_name)
            if isinstance(val, (set, list, tuple)):
                collected.update(str(item) for item in val if isinstance(item, str))
        return collected

    @staticmethod
    def _normalize_token(text: str) -> str:
        """
        Normalise un terme en minuscules sans accents.

        :param text: Texte source.
        :return: Texte normalisé.
        """
        nfkd: str = unicodedata.normalize("NFKD", text.strip().lower())
        return "".join(ch for ch in nfkd if not unicodedata.combining(ch))

    def _resolve_path(self, raw_path: str) -> str:
        """
        Résout un chemin relativement au dossier contenant config.ini.

        :param raw_path: Chemin lu dans le fichier INI.
        :return: Chemin absolu normalisé.
        """
        cleaned: str = raw_path.strip()
        if os.path.isabs(cleaned):
            return os.path.normpath(cleaned)
        return os.path.normpath(os.path.join(self._base_dir, cleaned))

    def _parse_csv_list(self, raw_val: str) -> List[str]:
        """
        Convertit une chaîne séparée par des virgules en liste nettoyée.

        :param raw_val: Chaîne brute issue du INI.
        :return: Liste des éléments non vides.
        """
        return [item.strip() for item in raw_val.split(",") if item.strip()]

    def _load(self) -> None:
        """
        Lit les sections [files], [formatting], [anonymizer] et [domain_aliases].
        """
        if not os.path.exists(self.config_path):
            return

        self._parser.read(self.config_path, encoding="utf-8")
        if "files" in self._parser:
            files_sec = self._parser["files"]
            if "csv_directory" in files_sec:
                self.csv_directory = self._resolve_path(files_sec["csv_directory"])
            if "csv_prefixes" in files_sec:
                self.csv_prefixes = [
                    p.lower() for p in self._parse_csv_list(files_sec["csv_prefixes"])
                ]
            if "first_names_file" in files_sec:
                self.first_names_file = self._resolve_path(files_sec["first_names_file"])
            if "output_file" in files_sec:
                self.output_file = self._resolve_path(files_sec["output_file"])
            if "anonymized_output_file" in files_sec:
                self.anonymized_output_file = self._resolve_path(
                    files_sec["anonymized_output_file"]
                )
            if "deduplicated_output_file" in files_sec:
                self.deduplicated_output_file = self._resolve_path(
                    files_sec["deduplicated_output_file"]
                )
            if "inverted_output_file" in files_sec:
                self.inverted_output_file = self._resolve_path(
                    files_sec["inverted_output_file"]
                )
            self.encoding = files_sec.get("encoding", fallback=self.encoding)

        if "formatting" in self._parser:
            fmt_sec = self._parser["formatting"]
            self.clean_phone_spaces = fmt_sec.getboolean(
                "clean_phone_spaces", fallback=self.clean_phone_spaces
            )
            self.lowercase_emails = fmt_sec.getboolean(
                "lowercase_emails", fallback=self.lowercase_emails
            )
            self.separator = fmt_sec.get("separator", fallback=self.separator)
            self.merge_separator = fmt_sec.get(
                "merge_separator", fallback=self.merge_separator
            )

        if "anonymizer" in self._parser:
            anon_sec = self._parser["anonymizer"]
            if "ignored_domains" in anon_sec:
                self.ignored_domains = {
                    d.lower() for d in self._parse_csv_list(anon_sec["ignored_domains"])
                }
            if "entity_prefixes" in anon_sec:
                self.entity_prefixes = {
                    p.lower() for p in self._parse_csv_list(anon_sec["entity_prefixes"])
                }
            if "company_keywords" in anon_sec:
                self.company_keywords = {
                    k.lower() for k in self._parse_csv_list(anon_sec["company_keywords"])
                }
            if "extra_locations" in anon_sec:
                self.extra_locations = {
                    loc.lower() for loc in self._parse_csv_list(anon_sec["extra_locations"])
                }

        if "domain_aliases" in self._parser:
            self.domain_aliases = {
                k.strip().lower(): v.strip()
                for k, v in self._parser["domain_aliases"].items()
                if k.strip() and v.strip()
            }

    def get_matching_csv_files(self) -> List[str]:
        """
        Recherche dans csv_directory tous les fichiers .csv dont le nom
        commence par l'un des préfixes configurés.

        :return: Liste triée des chemins complets des fichiers CSV trouvés.
        """
        if not os.path.isdir(self.csv_directory):
            return []
        matched: List[str] = []
        for file_name in sorted(os.listdir(self.csv_directory)):
            lower_name: str = file_name.lower()
            if lower_name.endswith(".csv") and any(
                lower_name.startswith(pref) for pref in self.csv_prefixes
            ):
                matched.append(os.path.join(self.csv_directory, file_name))
        return matched

    def get_csv_files(self) -> List[str]:
        """
        Alias rétrocompatible de get_matching_csv_files().

        :return: Liste triée des chemins complets des fichiers CSV trouvés.
        """
        return self.get_matching_csv_files()

import csv
import os

import unicodedata
from typing import Dict, List, Optional, Set, Tuple
from google_contacts_parser.config_manager import ContactsConfig
from google_contacts_parser.text_utils import ContactTextUtils


class GoogleContactsReader:
    """
    Lit les exports CSV Google Contacts et Microsoft 365 / Outlook (FR/EN),
    fusionne les prénoms composés, corrige les inversions, détecte les
    enseignes ou contacts couples et déduit l'identité depuis l'e-mail.
    """

    COMMON_NICKNAMES: Dict[str, str] = {
        "nico": "Nicolas",
        "steph": "Stéphanie",
        "seb": "Sébastien",
        "alex": "Alexandre",
        "fred": "Frédéric",
        "greg": "Grégory",
        "chris": "Christophe",
        "flo": "Florence",
        "caro": "Caroline",
        "ben": "Benjamin",
        "guigui": "Guillaume",
        "math": "Mathieu",
        "tom": "Thomas",
        "sam": "Samuel",
        "max": "Maxime",
        "val": "Valérie",
        "nat": "Nathalie",
        "nath": "Nathalie",
        "pat": "Patrick",
        "phil": "Philippe",
        "dom": "Dominique",
        "jp": "Jean-Pierre",
        "jc": "Jean-Claude",
        "jl": "Jean-Luc",
        "jm": "Jean-Marc",
    }

    def __init__(self, config: ContactsConfig) -> None:
        """
        Initialise le lecteur avec les règles de formatage et charge la liste
        des prénoms fréquents.

        :param config: Instance de ContactsConfig contenant les paramètres.
        """
        self.config: ContactsConfig = config
        self.frequent_first_names: Set[str] = set()
        self.norm_to_display_first_name: Dict[str, str] = {}
        self._load_first_names()

    def _normalize_name_token(self, text: str) -> str:
        """
        Normalise un prénom ou nom (minuscules, sans accents ni espaces).

        :param text: Chaîne brute à normaliser.
        :return: Chaîne normalisée sans accents.
        """
        nfkd: str = unicodedata.normalize("NFKD", text.strip().lower())
        return "".join(c for c in nfkd if not unicodedata.combining(c))

    def _sanitize_name_part(self, raw_part: str) -> str:
        """
        Nettoie un prénom ou un nom : retire les guillemets, ignore les
        e-mails bruts, supprime les crochets techniques et la ponctuation
        finale superflue.

        :param raw_part: Prénom ou nom brut extrait du CSV.
        :return: Chaîne nettoyée.
        """
        cleaned: str = raw_part.strip().strip('"').strip()
        if not cleaned:
            return ""
        if "@" in cleaned or ("\x3c" in cleaned and "\x3e" in cleaned):
            return ""
        cleaned = re.sub(r"\[[^\]]*\]", "", cleaned).strip().strip('"').strip()
        if not cleaned or not any(ch.isalnum() for ch in cleaned):
            return ""
        cleaned = re.sub(r"([^\w\s\)\.]+)$", "", cleaned).strip()
        cleaned = re.sub(r"\.{2,}$", ".", cleaned)
        cleaned = re.sub(r"\){2,}$", ")", cleaned)
        return cleaned
    def _load_first_names(self) -> None:
        """
        Charge et normalise la liste des prénoms fréquents depuis le fichier
        configuré, en mémorisant leur orthographe accentuée éventuelle.
        """
        if not os.path.exists(self.config.first_names_file):
            print(
                f"[AVERTISSEMENT] Fichier de prénoms introuvable : "
                f"'{self.config.first_names_file}'"
            )
            return

        with open(self.config.first_names_file, mode="r", encoding="utf-8") as f:
            for line in f:
                raw: str = line.strip()
                if not raw or raw.startswith("#"):
                    continue
                cleaned: str = self._normalize_name_token(raw)
                if cleaned:
                    self.frequent_first_names.add(cleaned)
                    if (
                        cleaned not in self.norm_to_display_first_name
                        or raw.lower() != cleaned
                    ):
                        self.norm_to_display_first_name[cleaned] = (
                            self._capitalize_word_segment(raw)
                        )

    def _capitalize_word_segment(self, raw_text: str) -> str:
        """
        Met en majuscule la première lettre de chaque segment séparé par
        espace, tiret, point ou apostrophe, tout en préservant les casses mixtes
        volontaires (ex: 'McDonald', 'O'Connor').

        :param raw_text: Segment brut.
        :return: Texte capitalisé proprement.
        """
        parts: List[str] = re.split(r"([\s\-\.']+)", raw_text.strip())
        formatted: List[str] = []
        for part in parts:
            if not part or re.match(r"^[\s\-\.']+$", part):
                formatted.append(part)
            elif part.isupper() or part.islower():
                formatted.append(part.capitalize())
            else:
                formatted.append(f"{part[:1].upper()}{part[1:]}")
        return "".join(formatted)

    def _resolve_first_name_display(self, token: str) -> Optional[str]:
        """
        Vérifie si un jeton est un prénom connu ou un diminutif courant et
        retourne sa forme affichable.

        :param token: Jeton brut extrait d'une adresse e-mail.
        :return: Prénom capitalisé si reconnu, sinon None.
        """
        norm: str = self._normalize_name_token(token)
        if norm in self.COMMON_NICKNAMES:
            return self.COMMON_NICKNAMES[norm]
        if norm in self.frequent_first_names:
            return self.norm_to_display_first_name.get(
                norm, self._capitalize_word_segment(token)
            )
        return None

    def _format_org_label(self, raw_org: str) -> str:
        """
        Capitalise la première lettre de l'enseigne ainsi que les prénoms
        qu'elle contient.

        :param raw_org: Libellé brut de l'enseigne.
        :return: Libellé formaté proprement.
        """
        words: List[str] = raw_org.strip().split()
        if not words:
            return ""
        formatted: List[str] = []
        for idx, word in enumerate(words):
            norm_w: str = self._normalize_name_token(word)
            if norm_w in self.norm_to_display_first_name:
                formatted.append(self.norm_to_display_first_name[norm_w])
            elif idx == 0:
                formatted.append(self._capitalize_word_segment(word))
            else:
                formatted.append(word)
        return " ".join(formatted)
    def _extract_identity_from_email(
        self, emails: List[str]
    ) -> Optional[Tuple[str, str]]:
        """
        Cherche une adresse nominative avec séparateur ou collée contenant
        un prénom fréquent ou diminutif connu.

        :param emails: Liste des adresses e-mail du contact.
        :return: Tuple (Prénom, Nom) capitalisé si trouvé, sinon None.
        """
        if not self.frequent_first_names and not self.COMMON_NICKNAMES:
            return None

        all_known_tokens: Set[str] = set(self.frequent_first_names) | set(
            self.COMMON_NICKNAMES.keys()
        )
        sorted_first_names: List[str] = sorted(
            all_known_tokens, key=len, reverse=True
        )

        for email in emails:
            canonical_email: str = ContactTextUtils.normalize_email_alias(email)
            if "@" not in canonical_email:
                continue
            local_part, _ = canonical_email.rsplit("@", 1)
            local_part = local_part.strip()
            tokens: List[str] = [
                t for t in re.split(r"[\.\-_]+", local_part) if t
            ]

            if len(tokens) == 2:
                part1, part2 = tokens
                if not any(ch.isdigit() for ch in part1 + part2):
                    disp1: Optional[str] = self._resolve_first_name_display(
                        part1
                    )
                    disp2: Optional[str] = self._resolve_first_name_display(
                        part2
                    )
                    if disp1 and not disp2 and len(part2) >= 2:
                        return disp1, self._capitalize_word_segment(part2)
                    if disp2 and not disp1 and len(part1) >= 2:
                        return disp2, self._capitalize_word_segment(part1)

            elif len(tokens) == 1 and not any(
                ch.isdigit() for ch in local_part
            ):
                norm_local: str = self._normalize_name_token(local_part)
                if norm_local in all_known_tokens:
                    continue
                for fname in sorted_first_names:
                    if len(fname) < 4:
                        continue
                    if norm_local.endswith(fname) and (
                        len(norm_local) >= len(fname) + 3
                    ):
                        raw_last: str = local_part[
                            : len(local_part) - len(fname)
                        ]
                        disp: str = (
                            self._resolve_first_name_display(fname)
                            or self._capitalize_word_segment(fname)
                        )
                        return disp, self._capitalize_word_segment(raw_last)
                    if norm_local.startswith(fname) and (
                        len(norm_local) >= len(fname) + 3
                    ):
                        raw_last = local_part[len(fname) :]
                        disp = (
                            self._resolve_first_name_display(fname)
                            or self._capitalize_word_segment(fname)
                        )
                        return disp, self._capitalize_word_segment(raw_last)

        return None

    def _infer_last_name_from_emails(
        self, first_name: str, emails: List[str]
    ) -> str:
        """
        Déduit un nom de famille si un e-mail est au format
        'prenom.nom@domaine', 'surnom.nom@domaine' ou 'prenomnom@domaine'.

        :param first_name: Prénom ou surnom connu du contact.
        :param emails: Liste des adresses e-mail associées au contact.
        :return: Nom de famille déduit et capitalisé, ou chaîne vide.
        """
        norm_first: str = self._normalize_name_token(first_name)
        if not norm_first:
            return ""

        for email in emails:
            canonical_email: str = ContactTextUtils.normalize_email_alias(email)
            if "@" not in canonical_email:
                continue
            local_part, _ = canonical_email.rsplit("@", 1)
            local_part = local_part.strip()

            if "." in local_part:
                prefix, candidate_last = local_part.split(".", 1)
                norm_prefix: str = self._normalize_name_token(prefix)
                is_exact_match: bool = norm_prefix == norm_first
                is_nickname_match: bool = (
                    len(norm_first) >= 3
                    and len(norm_prefix) >= 3
                    and norm_first[:3] == norm_prefix[:3]
                    and (
                        norm_prefix in self.frequent_first_names
                        or norm_first in self.frequent_first_names
                    )
                )
                if (is_exact_match or is_nickname_match) and (
                    len(candidate_last) >= 2
                ):
                    if not any(ch.isdigit() for ch in candidate_last):
                        return self._capitalize_word_segment(
                            candidate_last.replace(".", " ")
                        )
            else:
                norm_local: str = self._normalize_name_token(local_part)
                if (
                    norm_local.startswith(norm_first)
                    and len(norm_local) >= len(norm_first) + 2
                    and norm_local not in self.frequent_first_names
                ):
                    candidate_last = local_part[len(norm_first) :].lstrip("_-")
                    if len(candidate_last) >= 2 and not any(
                        ch.isdigit() for ch in candidate_last
                    ):
                        return self._capitalize_word_segment(candidate_last)
        return ""
    def _format_couple_from_paired_fields(
        self, clean_first: str, clean_last: str
    ) -> Optional[str]:
        """
        Détecte un couple saisi sous la forme 'Prénom1 - Prénom2' et
        'Nom1 - Nom2' et associe chaque prénom à son nom respectif.

        :param clean_first: Champ prénom contenant potentiellement deux prénoms.
        :param clean_last: Champ nom contenant potentiellement deux noms.
        :return: Libellé 'Prénom1 Nom1 et Prénom2 Nom2' si détecté, sinon None.
        """
        if not (clean_first and clean_last):
            return None

        first_match: Optional[re.Match[str]] = re.match(
            r"^(.+?)\s*(?:-|/|–|—|\bet\b|&)\s*(.+)$",
            clean_first,
            flags=re.IGNORECASE,
        )
        last_match: Optional[re.Match[str]] = re.match(
            r"^(.+?)\s*(?:-|/|–|—|\bet\b|&)\s*(.+)$",
            clean_last,
            flags=re.IGNORECASE,
        )
        if not (first_match and last_match):
            return None

        f1_raw: str = first_match.group(1).strip()
        f2_raw: str = first_match.group(2).strip()
        l1_raw: str = last_match.group(1).strip()
        l2_raw: str = last_match.group(2).strip()

        if (
            self._normalize_name_token(f1_raw) in self.frequent_first_names
            and self._normalize_name_token(f2_raw) in self.frequent_first_names
        ):
            f1: str = (
                self._capitalize_word_segment(f1_raw)
                if f1_raw.isupper()
                else f1_raw
            )
            f2: str = (
                self._capitalize_word_segment(f2_raw)
                if f2_raw.isupper()
                else f2_raw
            )
            l1: str = (
                self._capitalize_word_segment(l1_raw)
                if l1_raw.isupper()
                else l1_raw
            )
            l2: str = (
                self._capitalize_word_segment(l2_raw)
                if l2_raw.isupper()
                else l2_raw
            )
            return f"{f1} {l1} et {f2} {l2}"

        return None

    @staticmethod
    def _get_first_non_empty_field(
        row: Dict[str, str], candidates: Tuple[str, ...]
    ) -> str:
        """
        Retourne la première valeur non vide parmi une liste de colonnes
        candidates (insensible à la casse).

        :param row: Ligne CSV lue par DictReader.
        :param candidates: Noms de colonnes possibles (Google / Outlook).
        :return: Valeur nettoyée ou chaîne vide.
        """
        lower_map: Dict[str, str] = {
            k.strip().lower(): (v or "").strip()
            for k, v in row.items()
            if k
        }
        for cand in candidates:
            val: str = lower_map.get(cand.lower(), "")
            if val:
                return val
        return ""

    def _extract_full_name(
        self, row: Dict[str, str], row_emails: List[str]
    ) -> str:
        """
        Construit la chaîne 'Prénom Nom' depuis les colonnes Google ou
        Microsoft 365 / Outlook (FR/EN), gère les couples et déduit l'identité.

        :param row: Dictionnaire représentant une ligne du fichier CSV.
        :param row_emails: Adresses e-mail extraites pour cette ligne CSV.
        :return: Nom complet nettoyé et ordonné, ou chaîne vide.
        """
        raw_first: str = self._get_first_non_empty_field(
            row, ("First Name", "Given Name", "Prénom", "Prenom")
        )
        raw_middle: str = self._get_first_non_empty_field(
            row,
            ("Middle Name", "Additional Name", "Deuxième prénom", "Deuxieme prenom"),
        )
        raw_last: str = self._get_first_non_empty_field(
            row, ("Last Name", "Family Name", "Nom de famille", "Nom")
        )

        clean_first: str = self._sanitize_name_part(raw_first)
        clean_middle: str = self._sanitize_name_part(raw_middle)
        clean_last: str = self._sanitize_name_part(raw_last)

        if not clean_first and not clean_middle and not clean_last:
            raw_full: str = self._get_first_non_empty_field(
                row, ("Name", "Nom complet", "Display Name", "Nom affiché")
            )
            if not raw_full:
                email_only_id: Optional[Tuple[str, str]] = (
                    self._extract_identity_from_email(row_emails)
                )
                if email_only_id:
                    e_first, e_last = email_only_id
                    print(
                        f"[IDENTITÉ DÉDUITE MAIL] {row_emails} -> "
                        f"'{e_first} {e_last}'"
                    )
                    return f"{e_first} {e_last}"
                return ""
            if "\x3c" in raw_full or "@" in raw_full:
                return raw_full
            if " " in raw_full:
                part_first, part_last = raw_full.split(" ", 1)
                clean_first = self._sanitize_name_part(part_first)
                clean_last = self._sanitize_name_part(part_last)
            else:
                clean_first = self._sanitize_name_part(raw_full)

        if clean_middle and re.search(r"\b(?:et|&)\b", clean_middle, flags=re.IGNORECASE):
            couple_parts: List[str] = [
                p for p in [clean_first, clean_middle, clean_last] if p
            ]
            return " ".join(couple_parts)

        if not clean_middle:
            paired_couple: Optional[str] = (
                self._format_couple_from_paired_fields(clean_first, clean_last)
            )
            if paired_couple:
                return paired_couple
        if clean_first and not clean_middle and not clean_last and row_emails:
            inferred_single: str = self._infer_last_name_from_emails(
                clean_first, row_emails
            )
            if inferred_single:
                print(
                    f"[NOM DÉDUIT] '{clean_first}' + {row_emails} -> "
                    f"'{clean_first} {inferred_single}'"
                )
                return f"{clean_first} {inferred_single}"

        org_candidate: str = " ".join(
            p for p in [clean_first, clean_middle, clean_last] if p
        )
        words_in_label: List[str] = org_candidate.split()
        if len(words_in_label) >= 1 and row_emails and self.frequent_first_names:
            first_label_word: str = next(iter(words_in_label), "")
            first_tok: str = self._normalize_name_token(
                next(iter(re.split(r"[\s\-]+", first_label_word)), "")
            )
            last_label_word: str = words_in_label[-1]
            last_tok: str = self._normalize_name_token(
                next(iter(re.split(r"[\s\-]+", last_label_word)), "")
            )

            if (
                first_tok not in self.frequent_first_names
                and last_tok not in self.frequent_first_names
            ):
                email_identity: Optional[Tuple[str, str]] = (
                    self._extract_identity_from_email(row_emails)
                )
                if email_identity:
                    e_first, e_last = email_identity
                    clean_org: str = self._format_org_label(org_candidate)
                    print(
                        f"[ENSEIGNE + MAIL] '{org_candidate}' + "
                        f"{row_emails} -> '{e_first} {e_last} ({clean_org})'"
                    )
                    return f"{e_first} {e_last} ({clean_org})"

        if clean_first and clean_middle and not clean_last:
            norm_f: str = self._normalize_name_token(clean_first)
            norm_m: str = self._normalize_name_token(clean_middle)
            if (
                norm_m in self.frequent_first_names
                and norm_f not in self.frequent_first_names
            ):
                first_name: str = clean_middle
                last_name: str = clean_first
            else:
                sep: str = "-" if norm_f in self.frequent_first_names else " "
                first_name = f"{clean_first}{sep}{clean_middle}"
                last_name = ""
        elif clean_first and clean_middle:
            sep = (
                "-"
                if self._normalize_name_token(clean_first)
                in self.frequent_first_names
                else " "
            )
            first_name = f"{clean_first}{sep}{clean_middle}"
            last_name = clean_last
        else:
            first_name = clean_first or clean_middle
            last_name = clean_last

        if first_name and not last_name and row_emails:
            inferred: str = self._infer_last_name_from_emails(
                first_name, row_emails
            )
            if inferred:
                last_name = inferred
                print(
                    f"[NOM DÉDUIT] '{first_name}' + {row_emails} -> "
                    f"'{first_name} {last_name}'"
                )

        if first_name and last_name and self.frequent_first_names:
            first_word: str = next(iter(re.split(r"[\s\-]+", first_name)), "")
            last_word: str = next(iter(re.split(r"[\s\-]+", last_name)), "")
            if (
                self._normalize_name_token(last_word)
                in self.frequent_first_names
                and self._normalize_name_token(first_word)
                not in self.frequent_first_names
            ):
                print(
                    f"[INVERSION] '{first_name} {last_name}' -> "
                    f"'{last_name} {first_name}'"
                )
                first_name, last_name = last_name, first_name

        if first_name.isupper():
            first_name = self._capitalize_word_segment(first_name)
        if last_name.isupper():
            last_name = self._capitalize_word_segment(last_name)

        return f"{first_name} {last_name}".strip()

    def _is_contact_column(self, col_name: str) -> Tuple[bool, bool]:
        """
        Détermine si une colonne CSV Google ou Outlook (FR/EN) contient un
        téléphone ou une adresse e-mail.

        :param col_name: Nom de l'en-tête de colonne CSV.
        :return: Tuple (est_téléphone, est_email).
        """
        lower_col: str = col_name.strip().lower()
        ignored_markers: Tuple[str, ...] = (
            "type",
            "label",
            "étiquette",
            "etiquette",
            "display name",
            "nom affiché",
            "nom affiche",
        )
        if any(marker in lower_col for marker in ignored_markers):
            return False, False

        is_phone: bool = any(
            k in lower_col
            for k in ("phone", "téléphone", "telephone", "tél.", "tel.")
        )
        is_email: bool = any(
            k in lower_col
            for k in (
                "e-mail",
                "email",
                "mail",
                "adresse de messagerie",
                "courriel",
            )
        )
        return is_phone, is_email

    def _normalize_value(
        self, raw_val: str, is_phone: bool, is_email: bool
    ) -> str:
        """
        Applique les règles de nettoyage définies dans config.ini sur une
        coordonnée (et supprime les suffixes '+alias' sur les e-mails).

        :param raw_val: Valeur brute extraite de la cellule CSV.
        :param is_phone: Indique s'il s'agit d'un numéro de téléphone.
        :param is_email: Indique s'il s'agit d'une adresse e-mail.
        :return: Coordonnée normalisée.
        """
        cleaned: str = raw_val.strip()
        if is_phone and self.config.clean_phone_spaces:
            cleaned = re.sub(r"[\s\-\.]", "", cleaned)
        if is_email:
            cleaned = ContactTextUtils.normalize_email_alias(cleaned)
            if self.config.lowercase_emails:
                cleaned = cleaned.lower()
        return cleaned

    @staticmethod
    def _detect_csv_delimiter(sample: str) -> str:
        """
        Détecte automatiquement si le CSV utilise ',' (Google / Outlook EN)
        ou ';' (Outlook FR).

        :param sample: Début du contenu du fichier CSV.
        :return: Caractère délimiteur détecté.
        """
        lines: List[str] = sample.splitlines()
        first_line: str = next(iter(lines), "")
        if first_line.count(";") > first_line.count(","):
            return ";"
        return ","

    def parse_file(self, file_path: str) -> List[Tuple[str, str]]:
        """
        Parcourt un fichier CSV Google Contacts ou Microsoft 365 / Outlook
        et retourne la liste des paires (coordonnée, nom).

        :param file_path: Chemin vers le fichier CSV à analyser.
        :return: Liste de tuples (coordonnée, prénom nom).
        """
        if not os.path.exists(file_path):
            print(f"[AVERTISSEMENT] Fichier introuvable ignoré : '{file_path}'")
            return []

        results: List[Tuple[str, str]] = []
        with open(
            file_path, mode="r", encoding=self.config.encoding, newline=""
        ) as csv_file:
            sample: str = csv_file.read(4096)
            csv_file.seek(0)
            delimiter: str = self._detect_csv_delimiter(sample)
            reader: csv.DictReader = csv.DictReader(
                csv_file, delimiter=delimiter
            )
            if not reader.fieldnames:
                return []

            target_cols: List[Tuple[str, bool, bool]] = [
                (col, *self._is_contact_column(col))
                for col in reader.fieldnames
                if any(self._is_contact_column(col))
            ]

            for row in reader:
                row_values: List[Tuple[str, bool]] = []
                row_emails: List[str] = []
                for col_name, is_phone, is_email in target_cols:
                    cell_value: str = (row.get(col_name) or "").strip()
                    if not cell_value:
                        continue
                    for sub_val in cell_value.split(":::"):
                        normalized: str = self._normalize_value(
                            sub_val, is_phone, is_email
                        )
                        if normalized:
                            row_values.append((normalized, is_email))
                            if is_email:
                                row_emails.append(normalized)

                full_name: str = self._extract_full_name(row, row_emails)
                if full_name:
                    for val, _ in row_values:
                        results.append((val, full_name))

        print(
            f"[INFO] {len(results)} entrée(s) extraite(s) depuis "
            f"'{file_path}' (délimiteur '{delimiter}')."
        )
        return results

import os
import re
from typing import Dict, List, Tuple
from google_contacts_parser.config_manager import ContactsConfig
from google_contacts_parser.text_utils import ContactTextUtils


class ContactsDeduplicator:
    """
    Lit systématiquement le fichier anonymisé, normalise les alias '+...' des adresses e-mail,
    supprime les doublons exacts ou équivalents et fusionne les valeurs distinctes partageant
    une même clé tout en préservant l'ordre d'apparition d'origine.
    """

    def __init__(self, config: ContactsConfig) -> None:
        """
        Initialise le dédoublonneur avec la configuration du projet.

        :param config: Instance de ContactsConfig contenant les chemins et séparateurs.
        """
        self.config: ContactsConfig = config

    def _split_merged_values(self, raw_right: str) -> List[str]:
        """
        Découpe la partie droite uniquement sur le séparateur de fusion entouré
        d'espaces afin de ne pas couper les mots contenant 'et' (ex: 'Cabinet').

        :param raw_right: Partie droite de la ligne après le séparateur principal.
        :return: Liste des libellés individuels nettoyés.
        """
        cleaned: str = raw_right.strip()
        if not cleaned:
            return []
        sep_token: str = self.config.merge_separator.strip()
        if sep_token:
            pattern: str = rf"\s+{re.escape(sep_token)}\s+"
            return [part.strip() for part in re.split(pattern, cleaned) if part.strip()]
        return [cleaned]

    def _normalize_key(self, raw_key: str) -> str:
        """
        Normalise la clé de gauche (supprime l'alias '+...' s'il s'agit d'une adresse e-mail).

        :param raw_key: Coordonnée brute située à gauche du séparateur.
        :return: Clé canonique normalisée.
        """
        cleaned: str = raw_key.strip()
        if "@" in cleaned:
            cleaned = ContactTextUtils.normalize_email_alias(cleaned)
            if self.config.lowercase_emails:
                cleaned = cleaned.lower()
        return cleaned

    def _compute_value_signature(self, label: str) -> Tuple[str, ...]:
        """
        Calcule une signature canonique triée des prénoms et initiales d'un libellé
        afin de détecter deux variantes d'une même personne ou d'un même couple.

        :param label: Libellé anonymisé complet (avec éventuel suffixe entre parenthèses).
        :return: Tuple trié des jetons normalisés représentant l'identité.
        """
        base_part, _, _ = ContactTextUtils.clean_display_name(label)
        raw_tokens: List[str] = [t for t in re.split(r"[\s\.\-&]+", base_part) if t]
        meaningful: List[str] = []
        for tok in raw_tokens:
            norm_tok: str = ContactTextUtils.normalize_text(tok)
            if norm_tok and norm_tok not in {"et"}:
                meaningful.append(norm_tok)
        if not meaningful:
            return (ContactTextUtils.normalize_text(label),)
        return tuple(sorted(meaningful))
    def _is_better_variant(self, candidate: str, existing: str) -> bool:
        """
        Détermine si un libellé candidat est plus lisible qu'un libellé équivalent déjà retenu
        (privilégie la forme explicite avec 'et'/'&' plutôt qu'une fusion par tiret).

        :param candidate: Nouveau libellé équivalent rencontré.
        :param existing: Libellé déjà mémorisé pour la même signature.
        :return: True si le candidat doit remplacer l'existant.
        """
        cand_has_conj: bool = bool(re.search(r"\b(?:et|&)\b", candidate, flags=re.IGNORECASE))
        exist_has_conj: bool = bool(re.search(r"\b(?:et|&)\b", existing, flags=re.IGNORECASE))
        if cand_has_conj and not exist_has_conj:
            return True
        return False

    def process(self) -> Tuple[int, int, int]:
        """
        Lit le fichier anonymisé, dédoublonne les lignes exactes ou équivalentes,
        fusionne les clés communes distinctes et écrit le résultat sans modifier l'ordre.

        :return: Tuple (lignes_lues, lignes_écrites, clés_fusionnées).
        """
        input_path: str = self.config.anonymized_output_file
        output_path: str = self.config.deduplicated_output_file

        if not os.path.exists(input_path):
            raise FileNotFoundError(f"Le fichier anonymisé '{input_path}' est introuvable.")

        grouped_entries: Dict[str, List[str]] = {}
        grouped_signatures: Dict[str, Dict[Tuple[str, ...], int]] = {}
        raw_line_count: int = 0
        sep_joiner: str = f" {self.config.merge_separator.strip()} "

        with open(input_path, mode="r", encoding="utf-8") as in_file:
            for raw_line in in_file:
                line: str = raw_line.rstrip("\r\n")
                if not line or self.config.separator not in line:
                    continue
                raw_line_count += 1
                left_key, raw_right = line.split(self.config.separator, 1)
                clean_key: str = self._normalize_key(left_key)
                values: List[str] = self._split_merged_values(raw_right)

                if clean_key not in grouped_entries:
                    grouped_entries[clean_key] = []
                    grouped_signatures[clean_key] = {}

                for val in values:
                    sig: Tuple[str, ...] = self._compute_value_signature(val)
                    sig_map: Dict[Tuple[str, ...], int] = grouped_signatures[clean_key]
                    if sig not in sig_map:
                        sig_map[sig] = len(grouped_entries[clean_key])
                        grouped_entries[clean_key].append(val)
                    else:
                        existing_idx: int = sig_map[sig]
                        if self._is_better_variant(val, grouped_entries[clean_key][existing_idx]):
                            grouped_entries[clean_key][existing_idx] = val

        merged_keys_count: int = sum(1 for vals in grouped_entries.values() if len(vals) > 1)
        output_dir: str = os.path.dirname(output_path)
        if output_dir:
            os.makedirs(output_dir, exist_ok=True)

        with open(output_path, mode="w", encoding="utf-8", newline="\n") as out_file:
            for key, values in grouped_entries.items():
                merged_right: str = sep_joiner.join(values)
                out_file.write(f"{key}{self.config.separator}{merged_right}\n")

        return raw_line_count, len(grouped_entries), merged_keys_count

        """
        Normalise un nom géographique (minuscules, sans accents, espaces, tirets

        :param text: Libellé géographique brut.

        without_accents: str = "".join(c for c in nfkd if not unicodedata.combining(c))
        return re.sub(r"[\s\-'\u2019]", "", without_accents)

    @classmethod
    def build_normalized_set(cls, extra_locations: Set[str]) -> Set[str]:
        """

            | set(cls.REGIONS)
            | set(cls.MAJOR_CITIES)
            | set(extra_locations)
        )
        return {cls.normalize_geo_token(item) for item in all_raw if item.strip()}

import os
import tempfile
from typing import List, Tuple
from google_contacts_parser.config_manager import ContactsConfig
from google_contacts_parser.text_utils import ContactTextUtils


class ContactsInverter:
    """
    Lit le fichier dédoublonné, inverse la partie gauche et la partie droite du séparateur
    et trie les lignes obtenues par ordre alphabétique sur la nouvelle partie gauche.
    """

    def __init__(self, config: ContactsConfig) -> None:
        """
        Initialise l'inverseur avec la configuration du projet.

        :param config: Instance de ContactsConfig contenant les chemins et séparateurs.
        """
        self.config: ContactsConfig = config

    def _sort_key(self, pair: Tuple[str, str]) -> Tuple[str, str, str]:
        """
        Calcule la clé de tri alphabétique basée sur la nouvelle partie gauche (libellé)
        puis sur la partie droite (coordonnée).

        :param pair: Tuple (nouvelle_gauche, nouvelle_droite).
        :return: Tuple normalisé pour un tri alphabétique insensible aux accents et à la casse.
        """
        new_left, new_right = pair
        norm_left: str = ContactTextUtils.normalize_text(new_left)
        return (norm_left, new_left.lower(), new_right.lower())

    @staticmethod
    def _atomic_write_lines(target_path: str, lines: List[str]) -> None:
        """
        Écrit les lignes sur le disque de façon atomique via un fichier temporaire.

        :param target_path: Chemin final du fichier de sortie.
        :param lines: Liste complète des lignes formatées à enregistrer.
        """
        output_dir: str = os.path.dirname(os.path.abspath(target_path))
        if output_dir:
            os.makedirs(output_dir, exist_ok=True)
        fd, tmp_path = tempfile.mkstemp(
            dir=output_dir, prefix=".tmp_inv_", suffix=".tmp"
        )
        try:
            with os.fdopen(fd, mode="w", encoding="utf-8", newline="\n") as out_file:
                out_file.writelines(lines)
            os.replace(tmp_path, target_path)
        except Exception:
            if os.path.exists(tmp_path):
                try:
                    os.remove(tmp_path)
                except OSError:
                    pass
            raise

    def process(self) -> int:
        """
        Inverse chaque ligne du fichier dédoublonné (droite = gauche), trie par la nouvelle
        partie gauche et écrit le résultat de manière atomique dans le fichier contacts.txt.

        :return: Nombre de lignes écrites dans le fichier inversé.
        """
        input_path: str = self.config.deduplicated_output_file
        output_path: str = self.config.inverted_output_file

        if not os.path.exists(input_path):
            raise FileNotFoundError(f"Le fichier dédoublonné '{input_path}' est introuvable.")

        inverted_pairs: List[Tuple[str, str]] = []
        with open(input_path, mode="r", encoding="utf-8") as in_file:
            for raw_line in in_file:
                line: str = raw_line.rstrip("\r\n")
                if not line or self.config.separator not in line:
                    continue
                old_left, old_right = line.split(self.config.separator, 1)
                clean_left: str = old_right.strip()
                clean_right: str = old_left.strip()
                if clean_left and clean_right:
                    inverted_pairs.append((clean_left, clean_right))

        inverted_pairs.sort(key=self._sort_key)

        formatted_lines: List[str] = [
            f"{new_left}{self.config.separator}{new_right}\n"
            for new_left, new_right in inverted_pairs
        ]
        self._atomic_write_lines(output_path, formatted_lines)

        return len(inverted_pairs)

import time
from typing import List, Tuple
from google_contacts_parser.anonymizer import ContactsAnonymizer
from google_contacts_parser.config_manager import ContactsConfig
from google_contacts_parser.csv_reader import GoogleContactsReader
from google_contacts_parser.deduplicator import ContactsDeduplicator
from google_contacts_parser.inverter import ContactsInverter

class ContactsPipelineRunner:
    """
    Ordonnance l'exécution séquentielle des quatre étapes de traitement des contacts :
    1. Extraction et indexation depuis les fichiers CSV Google Contacts.
    2. Anonymisation des noms de famille et résolution des homonymes.
    3. Dédoublonnage exact ou équivalent et fusion des clés communes.
    4. Inversion gauche/droite et tri alphabétique par la nouvelle partie gauche.
    """

    def __init__(self, config: ContactsConfig) -> None:
        """
        Initialise l'ordonnanceur avec la configuration partagée par les quatre modules.

        :param config: Instance de ContactsConfig contenant tous les chemins et règles.
        """
        self.config: ContactsConfig = config

    def _run_extraction(self) -> int:
        """
        Exécute la première étape : lecture des CSV détectés et écriture du fichier indexé.

        :return: Nombre de correspondances extraites.
        """
        print("=== [1/4] EXTRACTION DES CONTACTS CSV ===")
        csv_files: List[str] = self.config.get_csv_files()
        if not csv_files:
            raise FileNotFoundError(
                f"Aucun fichier CSV correspondant trouvé dans '{self.config.csv_directory}'."
            )

        reader: GoogleContactsReader = GoogleContactsReader(config=self.config)
        writer: ContactsOutputWriter = ContactsOutputWriter(config=self.config)

        all_entries: List[Tuple[str, str]] = []
        for csv_path in csv_files:
            entries: List[Tuple[str, str]] = reader.parse_file(file_path=csv_path)
            all_entries.extend(entries)

        writer.write_entries(entries=all_entries)
        print(
            f"[OK] {len(all_entries)} correspondance(s) extraite(s) depuis {len(csv_files)} fichier(s) "
            f"-> '{self.config.output_file}'.\n"
        )
        return len(all_entries)

    def _run_anonymization(self) -> int:
        """
        Exécute la deuxième étape : anonymisation du fichier indexé.

        :return: Nombre de lignes écrites dans le fichier anonymisé.
        """
        print("=== [2/4] ANONYMISATION DES CONTACTS ===")
        anonymizer: ContactsAnonymizer = ContactsAnonymizer(config=self.config)
        count: int = anonymizer.process()
        print(f"[OK] {count} ligne(s) anonymisée(s) -> '{self.config.anonymized_output_file}'.\n")
        return count

    def _run_deduplication(self) -> Tuple[int, int, int]:
        """
        Exécute la troisième étape : suppression des doublons et fusion des clés communes.

        :return: Tuple (lignes_lues, lignes_écrites, clés_fusionnées).
        """
        print("=== [3/4] DÉDOUBLONNAGE ET FUSION ===")
        deduplicator: ContactsDeduplicator = ContactsDeduplicator(config=self.config)
        read_count, written_count, merged_count = deduplicator.process()
        removed_count: int = read_count - written_count
        print(
            f"[OK] {read_count} ligne(s) -> {written_count} ligne(s) dédoublonnée(s) "
            f"({removed_count} doublon(s) réduit(s), {merged_count} clé(s) fusionnée(s)) "
            f"-> '{self.config.deduplicated_output_file}'.\n"
        )
        return read_count, written_count, merged_count

    def _run_inversion(self) -> int:
        """
        Exécute la quatrième étape : inversion gauche/droite et tri alphabétique.

        :return: Nombre de lignes écrites dans le fichier inversé final.
        """
        print("=== [4/4] INVERSION ET TRI ALPHABÉTIQUE ===")
        inverter: ContactsInverter = ContactsInverter(config=self.config)
        count: int = inverter.process()
        print(f"[OK] {count} ligne(s) inversée(s) et triée(s) -> '{self.config.inverted_output_file}'.\n")
        return count

    def run_all(self) -> None:
        """
        Enchaîne les quatre étapes du pipeline de bout en bout et affiche la durée totale.
        """
        start_time: float = time.perf_counter()
        self._run_extraction()
        self._run_anonymization()
        self._run_deduplication()
        self._run_inversion()

        print(f"Pipeline complet terminé avec succès en {elapsed:.2f}s.")

import re
import unicodedata
from typing import List, Optional, Set, Tuple


class ContactTextUtils:
    """
    Fournit des utilitaires de normalisation, formatage et détection d'enseignes/sociétés
    pour le traitement des noms et coordonnées de contacts.
    """

    @staticmethod
    def normalize_email_alias(email: str) -> str:
        """
        Supprime l'extension d'alias '+...' dans la partie locale d'une adresse e-mail
        (ex: 'jean.dupont+factures@gmail.com' -> 'jean.dupont@gmail.com').

        :param email: Adresse e-mail brute à normaliser.
        :return: Adresse e-mail canonique sans suffixe '+'.
        """
        cleaned: str = email.strip()
        if "@" not in cleaned:
            return cleaned
        local_part, domain_part = cleaned.split("@", 1)
        if "+" in local_part:
            local_part, _ = local_part.split("+", 1)
        return f"{local_part}@{domain_part}"

    @staticmethod
    def normalize_text(text: str) -> str:
        """
        Supprime les accents, espaces superflus et met en minuscules pour comparer deux chaînes.

        :param text: Texte brut à normaliser.
        :return: Chaîne normalisée en minuscules sans accents.
        """
        nfkd: str = unicodedata.normalize("NFKD", text.strip().lower())
        without_accents: str = "".join(c for c in nfkd if not unicodedata.combining(c))
        return re.sub(r"[\s\-']", "", without_accents)

    @classmethod
    def strip_provider_suffix(cls, name: str, ignored_domains: Set[str]) -> str:
        """
        Supprime un suffixe final de fournisseur de messagerie (ex: ' - Gmail', ' (Yahoo)')
        lorsqu'il correspond à l'un des domaines ignorés configurés.

        :param name: Libellé brut du contact.
        :param ignored_domains: Ensemble des fournisseurs de mail à ignorer (en minuscules).
        :return: Libellé nettoyé de toute mention finale de fournisseur.
        """
        cleaned: str = name.strip()
        if not cleaned or not ignored_domains:
            return cleaned

        sep_match: Optional[re.Match[str]] = re.match(r"^(.*?)\s*(?:-|/|–|—)\s*([^\s\-/]+)$", cleaned)
        if sep_match:
            prefix_part: str = sep_match.group(1).strip()
            suffix_token: str = sep_match.group(2).strip().lower()
            suffix_root: str = next(iter(suffix_token.split(".", 1)), "")
            if prefix_part and (suffix_token in ignored_domains or suffix_root in ignored_domains):
                return prefix_part

        paren_match: Optional[re.Match[str]] = re.match(r"^(.*?)\s*\(([^)]+)\)$", cleaned)
        if paren_match:
            prefix_part = paren_match.group(1).strip()
            paren_token: str = paren_match.group(2).strip().lower()
            paren_root: str = next(iter(paren_token.split(".", 1)), "")
            if prefix_part and (paren_token in ignored_domains or paren_root in ignored_domains):
                return prefix_part

        return cleaned

    @staticmethod
    def _abbreviate_geo_label(location_text: str) -> str:
        """
        Remplace un nom géographique par sa première lettre majuscule suivie d'un point
        (ex: 'Normandie' -> 'N.', 'Bretteville-sur-Laize' -> 'B.').

        :param location_text: Nom du lieu géographique détecté.
        :return: Initiale majuscule suivie d'un point, ou chaîne vide.
        """
        for ch in location_text.strip():
            if ch.isalpha():
                return f"{ch.upper()}."
        return ""

    @classmethod
    def _abbreviate_hyphenated_word(cls, word: str, excluded_locations_norm: Set[str]) -> str:
        """
        Abrège tout sous-segment géographique à l'intérieur d'un mot composé avec tiret
        (ex: 'Caen-Sud' -> 'C.-Sud') si le mot complet n'est pas un lieu unique.

        :param word: Mot contenant éventuellement des tirets.
        :param excluded_locations_norm: Ensemble des lieux normalisés à abréger.
        :return: Mot transformé avec les segments géographiques abrégés.
        """
        if "-" not in word:
            return word
        parts: List[str] = word.split("-")
        replaced: bool = False
        new_parts: List[str] = []
        for part in parts:
            if part and cls.normalize_text(part) in excluded_locations_norm:
                abbr: str = cls._abbreviate_geo_label(part)
                new_parts.append(abbr if abbr else part)
                replaced = True
            else:
                new_parts.append(part)
        return "-".join(new_parts) if replaced else word
    @classmethod
    def abbreviate_locations_in_text(cls, text: str, excluded_locations_norm: Set[str]) -> str:
        """
        Remplace toute mention d'un département, d'une région, d'une grande ville ou d'une
        commune configurée (y compris dans un mot composé comme 'Caen-Sud' ou un mot unique)
        par son initiale.

        :param text: Libellé ou nom de domaine formaté à inspecter.
        :param excluded_locations_norm: Ensemble des lieux normalisés à abréger.
        :return: Texte avec les lieux remplacés par leur initiale suivie d'un point.
        """
        cleaned: str = text.strip()
        if not cleaned or not excluded_locations_norm:
            return cleaned

        paren_match: Optional[re.Match[str]] = re.match(r"^(.*?)\s*\(([^)]+)\)$", cleaned)
        if paren_match:
            prefix_part: str = paren_match.group(1).strip()
            inside_paren: str = paren_match.group(2).strip()
            if prefix_part and cls.normalize_text(inside_paren) in excluded_locations_norm:
                abbr: str = cls._abbreviate_geo_label(inside_paren)
                cleaned = f"{prefix_part} {abbr}".strip()

        sep_match: Optional[re.Match[str]] = re.match(r"^(.*?)\s*(?:-|/|–|—|,)\s*(.+)$", cleaned)
        if sep_match:
            prefix_part = sep_match.group(1).strip()
            tail_part: str = sep_match.group(2).strip()
            if prefix_part and cls.normalize_text(tail_part) in excluded_locations_norm:
                abbr = cls._abbreviate_geo_label(tail_part)
                cleaned = f"{prefix_part} {abbr}".strip()

        words: List[str] = cleaned.split()
        idx: int = 0
        output_tokens: List[str] = []
        while idx < len(words):
            matched: bool = False
            max_span: int = min(len(words) - idx, 4)
            for span in range(max_span, 0, -1):
                candidate: str = " ".join(words[idx : idx + span])
                if cls.normalize_text(candidate) in excluded_locations_norm:
                    abbr = cls._abbreviate_geo_label(candidate)
                    if abbr:
                        output_tokens.append(abbr)
                    idx += span
                    matched = True
                    break
            if not matched:
                output_tokens.append(cls._abbreviate_hyphenated_word(words[idx], excluded_locations_norm))
                idx += 1

        return " ".join(output_tokens)

    @staticmethod
    def format_first_name(first_name: str) -> str:
        """
        Convertit un prénom tout en majuscules en capitale initiale suivie de minuscules.

        :param first_name: Prénom brut à formater.
        :return: Prénom proprement capitalisé.

        cleaned: str = first_name.strip().strip('"').strip()
        if not cleaned or not cleaned.isupper():

        parts: List[str] = re.split(r"([\s\-]+)", cleaned)
        return "".join(p.capitalize() if p and not re.match(r"^[\s\-]+$", p) else p for p in parts)

    @staticmethod
    def extract_domain_root(email: str) -> str:
        """

            return ""

        if not domain_part:
            return ""
        labels: List[str] = [lbl for lbl in domain_part.split(".") if lbl]
        if not labels:
            return ""
        if len(labels) == 1:
            return next(iter(labels), "").strip()
        if len(labels) >= 3 and len(labels[-1]) == 2 and labels[-2] in {"co", "com", "org", "net", "gouv", "asso"}:
            return labels[-3].strip()
        return labels[-2].strip()

    @staticmethod
    def format_initials(last_name: str, extra_chars: int = 0) -> str:
        """
        Convertit un nom de famille en initiales suivies d'un point, avec possibilité

        :param last_name: Nom de famille complet (ex: 'Durand' ou 'Dupont-Martin').
        :param extra_chars: Nombre de lettres supplémentaires à afficher (limité à 1).
        :return: Initiales formatées (ex: 'D.' ou 'Du.').
        """
        tokens: List[str] = [t for t in re.split(r"[\s\-]+", last_name.strip()) if any(c.isalpha() for c in t)]

        letters: List[str] = []

            if idx == total_tokens - 1 and clamped_extra > 0:
                letters.append(alpha_only[: 1 + clamped_extra].capitalize())
            else:
                letters.append(alpha_only[:1].upper())
        return ".".join(letters) + "." if letters else ""

import argparse

from doc_anonymizer.config_manager import PipelineConfig
from doc_anonymizer.text_utils import DocumentTextUtils


class ManualContactsAutoCompleter:
    """
    Lit contacts_manuels.txt, repère les entrées sans valeur après '='
    (ex. 'Prénom Nom = ') et génère automatiquement leur alias selon
    les règles standard (Prénom I.) avant de sauvegarder atomiquement.
    """

    def __init__(self, config: PipelineConfig, dry_run: bool = False) -> None:
        """
        Initialise le compléteur automatique avec la configuration active.

        :param config: Instance de PipelineConfig.
        :param dry_run: Si True, prévisualise sans modifier le fichier.
        """
        self.config: PipelineConfig = config
        self.dry_run: bool = dry_run
        self._auto_id_counter: int = 1

    def _is_first_name(self, token: str) -> bool:
        """
        Vérifie si un mot appartient au référentiel des prénoms connus.

        :param token: Jeton brut à tester.
        :return: True si le mot est un prénom connu.
        """
        norm: str = DocumentTextUtils.normalize_word(token)
        if norm in self.config.known_first_names:
            return True
        if "-" in norm:
            parts: List[str] = [p for p in norm.split("-") if p]
            return bool(parts) and all(
                p in self.config.known_first_names for p in parts
            )
        return False

    @staticmethod
    def _format_first_name(token: str) -> str:
        """
        Capitalise proprement un prénom simple ou composé (ex. 'JEAN-LUC' -> 'Jean-Luc').

        :param token: Prénom brut.
        :return: Prénom correctement capitalisé.
        """
        parts: List[str] = re.split(r"(-)", token.strip())
        return "".join(p.capitalize() if p != "-" else p for p in parts)

    @staticmethod
    def _extract_initial(last_name: str) -> str:
        """
        Extrait l'initiale en majuscule d'un nom de famille (gère les noms composés).

        :param last_name: Nom de famille brut.
        :return: Initiale suivie d'un point (ex. 'D.' ou 'L.-M.').
        """
        clean: str = last_name.strip()
        if not clean:
            return ""
        if "-" in clean:
            sub_initials: List[str] = [
                f"{p.upper()}." for p in clean.split("-") if p.strip()
            ]
            if sub_initials:
                return "-".join(sub_initials)
        return f"{clean.upper()}."
    def generate_alias(self, raw_target: str, used_aliases: Set[str]) -> str:
        """
        Génère un alias anonymisé à partir de la cible brute.

        :param raw_target: Texte situé à gauche du symbole '='.
        :param used_aliases: Ensemble des alias déjà présents dans le fichier.
        :return: Alias généré (ex. 'Aurélien D.').
        """
        target: str = raw_target.strip()
        if DocumentTextUtils.is_contact_identifier(target):
            norm_id: str = DocumentTextUtils.normalize_identifier(target)
            prefix: str = "Contact_Mail" if "@" in norm_id else "Contact_Tel"
            while True:
                candidate: str = f"{prefix}_{self._auto_id_counter:03d}"
                self._auto_id_counter += 1
                if candidate not in used_aliases:
                    return candidate

        tokens: List[str] = [t for t in target.split() if t]
        if len(tokens) == 2:
            first_cand, second_cand = tokens
            is_first_1: bool = self._is_first_name(first_cand)
            is_first_2: bool = self._is_first_name(second_cand)
            if is_first_2 and not is_first_1:
                first_tok, last_tok = second_cand, first_cand
            else:
                first_tok, last_tok = first_cand, second_cand
            formatted_first: str = self._format_first_name(first_tok)
            initial: str = self._extract_initial(last_tok)
            return f"{formatted_first} {initial}".strip()

        if len(tokens) > 2:
            first_tok = next(iter(tokens))
            last_tok = tokens[-1]
            if self._is_first_name(first_tok):
                return (
                    f"{self._format_first_name(first_tok)} "
                    f"{self._extract_initial(last_tok)}"
                )
            initials: List[str] = [
                f"{tok.upper()}." for tok in tokens if tok
            ]
            return " ".join(initials)

        single: str = next(iter(tokens), target)
        if self._is_first_name(single):
            return f"{self._format_first_name(single)} X."
        return self._extract_initial(single) or "Unknown_Contact"

    def process(self) -> Tuple[int, int]:
        """
        Parcourt contacts_manuels.txt, complète les lignes vides après '='
        et sauvegarde le résultat (sauf en mode --dry-run).

        :return: Tuple (nombre_lignes_completees, total_lignes_regles).
        """
        manual_path: str = self.config.manual_contacts_file
        if not os.path.exists(manual_path):
            print(
                f"[ERREUR] Fichier introuvable : '{manual_path}'.",
                flush=True,
            )
            return 0, 0

        with open(manual_path, "r", encoding="utf-8") as f_in:
            raw_lines: List[str] = f_in.readlines()

        used_aliases: Set[str] = set()
        for line in raw_lines:
            stripped: str = line.strip()
            if stripped and not stripped.startswith("#") and "=" in stripped:
                left_ex, right_ex = stripped.split("=", 1)
                if left_ex.strip() and right_ex.strip():
                    _, resolved_alias, _ = DocumentTextUtils.resolve_contact_pair(
                        left_ex, right_ex
                    )
                    if resolved_alias:
                        used_aliases.add(resolved_alias)
        updated_lines: List[str] = []
        completed_count: int = 0
        total_rules: int = 0

        for line in raw_lines:
            stripped = line.strip()
            if not stripped or stripped.startswith("#") or "=" not in stripped:
                updated_lines.append(line)
                continue

            total_rules += 1
            left_raw, right_raw = line.split("=", 1)
            left_clean: str = left_raw.strip()
            right_clean: str = right_raw.strip()

            if left_clean and not right_clean:
                new_alias: str = self.generate_alias(
                    left_clean, used_aliases
                )
                used_aliases.add(new_alias)
                new_line: str = f"{left_clean} = {new_alias}\n"
                updated_lines.append(new_line)
                completed_count += 1
                tag: str = "[PRÉVISUALISATION]" if self.dry_run else "[COMPLÉTÉ]"
                print(
                    f"{tag} '{left_clean} ='  -->  '{left_clean} = {new_alias}'",
                    flush=True,
                )
            elif right_clean and not left_clean:
                new_alias = self.generate_alias(
                    right_clean, used_aliases
                )
                used_aliases.add(new_alias)
                new_line = f"{right_clean} = {new_alias}\n"
                updated_lines.append(new_line)
                completed_count += 1
                tag = "[PRÉVISUALISATION]" if self.dry_run else "[COMPLÉTÉ]"
                print(
                    f"{tag} '= {right_clean}'  -->  '{right_clean} = {new_alias}'",
                    flush=True,
                )
            else:
                updated_lines.append(line)

        if completed_count > 0 and not self.dry_run:
            DocumentTextUtils.atomic_write_text(
                manual_path, "".join(updated_lines)
            )
            print(
                f"\n[OK] {completed_count} règle(s) enregistrée(s) dans "
                f"'{manual_path}' (sur {total_rules} règle(s) au total).",
                flush=True,
            )
        elif completed_count > 0 and self.dry_run:
            print(
                f"\n[SIMULATION] {completed_count} règle(s) seraient "
                f"complétées (aucune modification écrite sur le disque).",
                flush=True,
            )
        else:
            print(
                "\nAucune ligne vide après '=' détectée dans "
                f"'{manual_path}'.",
                flush=True,
            )

        return completed_count, total_rules

    """Point d'entrée en ligne de commande."""

class MarkdownPdfConverter:
    """
    Convertit un fichier Markdown en document PDF paginé et mis en forme via PyMuPDF.
    """

    @staticmethod
    def format_inline_markdown(text: str) -> str:
        """
        Convertit la mise en forme Markdown en ligne (gras, italique, code, liens).

        :param text: Ligne de texte brute.
        :return: Chaîne HTML échappée et formatée.
        """
        escaped: str = html.escape(text)
        escaped = re.sub(
            r"`([^`]+)`",
            r"<code style='background:#f1f5f9;padding:1px 4px;border-radius:3px;"
            r"font-family:monospace;font-size:9pt;'>\1</code>",
            escaped,
        )
        escaped = re.sub(r"\*\*([^*]+)\*\*", r"<b>\1</b>", escaped)
        escaped = re.sub(r"\*([^*]+)\*", r"<i>\1</i>", escaped)
        escaped = re.sub(
            r"\[([^\]]+)\]\(([^)]+)\)",
            r"<a href='\2' style='color:#1d4ed8;'>\1</a>",
            escaped,
        )
        return escaped

    @classmethod
    def markdown_to_styled_html(cls, md_text: str) -> str:
        """
        Transforme le contenu Markdown en document HTML stylisé prêt pour PyMuPDF,
        en préservant strictement les commentaires '#' et séparateurs '---' dans
        les blocs de code.

        :param md_text: Contenu complet du fichier Markdown.
        :return: Code HTML complet avec feuilles de style intégrées.
        """
        html_parts: List[str] = [
            "<!DOCTYPE html><html><head><meta charset='utf-8'><style>",
            "body { font-family: sans-serif; font-size: 10pt; color: #1e293b; line-height: 1.45; }",
            "h1 { color: #0f172a; font-size: 17pt; border-bottom: 2px solid #2563eb; padding-bottom: 4px; }",
            "h2 { color: #1e3a8a; font-size: 13pt; margin-top: 16px; border-bottom: 1px solid #cbd5e1; }",
            "h3 { color: #1d4ed8; font-size: 11pt; margin-top: 12px; }",
            "h4 { color: #334155; font-size: 10pt; margin-top: 10px; }",
            "pre { background: #f8fafc; border: 1px solid #cbd5e1; border-left: 3px solid #2563eb; "
            "padding: 8px; font-family: monospace; font-size: 8.5pt; white-space: pre-wrap; }",
            "blockquote { background: #eff6ff; border-left: 3px solid #3b82f6; padding: 6px 10px; margin: 8px 0; }",
            "ul, ol { margin-top: 4px; margin-bottom: 8px; padding-left: 20px; }",
            "li { margin-bottom: 3px; }",
            "hr { border: 0; border-top: 1px solid #e2e8f0; margin: 12px 0; }",
            "</style></head><body>",
        ]

        in_code_block: bool = False
        code_lines: List[str] = []
        list_mode: str = ""

        def _close_list() -> None:
            nonlocal list_mode
            if list_mode:
                html_parts.append(f"</{list_mode}>")
                list_mode = ""

        def _flush_code_block() -> None:
            nonlocal in_code_block, code_lines
            if in_code_block:
                code_body: str = html.escape("\n".join(code_lines))
                html_parts.append(f"<pre>{code_body}</pre>")
                code_lines = []
                in_code_block = False
        for raw_line in md_text.splitlines():
            stripped: str = raw_line.strip()

            if stripped.startswith("```"):
                _close_list()
                if in_code_block:
                    _flush_code_block()
                else:
                    in_code_block = True
                continue

            if in_code_block:
                code_lines.append(raw_line)
                continue

            if not stripped:
                _close_list()
                continue

            if stripped == "---":
                _close_list()
                html_parts.append("<hr>")
                continue

            if stripped.startswith("# "):
                _close_list()
                html_parts.append(f"<h1>{cls.format_inline_markdown(stripped[2:])}</h1>")
            elif stripped.startswith("## "):
                _close_list()
                html_parts.append(f"<h2>{cls.format_inline_markdown(stripped[3:])}</h2>")
            elif stripped.startswith("### "):
                _close_list()
                html_parts.append(f"<h3>{cls.format_inline_markdown(stripped[4:])}</h3>")
            elif stripped.startswith("#### "):
                _close_list()
                html_parts.append(f"<h4>{cls.format_inline_markdown(stripped[5:])}</h4>")
            elif stripped.startswith("> "):
                _close_list()
                html_parts.append(f"<blockquote>{cls.format_inline_markdown(stripped[2:])}</blockquote>")
            elif re.match(r"^[\*\-]\s+", stripped):
                if list_mode != "ul":
                    _close_list()
                    list_mode = "ul"
                    html_parts.append("<ul>")
                item_txt: str = re.sub(r"^[\*\-]\s+", "", stripped)
                html_parts.append(f"<li>{cls.format_inline_markdown(item_txt)}</li>")
            elif re.match(r"^\d+\.\s+", stripped):
                if list_mode != "ol":
                    _close_list()
                    list_mode = "ol"
                    html_parts.append("<ol>")
                item_txt = re.sub(r"^\d+\.\s+", "", stripped)
                html_parts.append(f"<li>{cls.format_inline_markdown(item_txt)}</li>")
            else:
                _close_list()
                html_parts.append(f"<p>{cls.format_inline_markdown(stripped)}</p>")

        _flush_code_block()
        _close_list()
        html_parts.append("</body></html>")
        return "\n".join(html_parts)
    @classmethod
    def convert(
        cls,
        md_path: str = "GUIDE_UTILISATION.md",
        pdf_path: Optional[str] = None,
        footer_label: Optional[str] = None,
    ) -> None:
        """
        Génère un fichier PDF mis en page et paginé à partir d'un fichier Markdown.

        :param md_path: Chemin vers le fichier Markdown source.
        :param pdf_path: Chemin de sortie du fichier PDF (déduit du .md si omis).
        :param footer_label: Libellé affiché avant le numéro de page.
        """
        src_path: Path = Path(md_path)
        if not src_path.exists():
            raise FileNotFoundError(f"Le fichier '{md_path}' est introuvable.")

        resolved_pdf: str = pdf_path if pdf_path else str(src_path.with_suffix(".pdf"))
        resolved_label: str = footer_label if footer_label else src_path.stem

        md_text: str = src_path.read_text(encoding="utf-8")

        styled_html: str = cls.markdown_to_styled_html(md_text)
        page_rect = pymupdf.paper_rect("a4")
        margin: float = 42.0
        footer_height: float = 24.0
        content_rect = pymupdf.Rect(
            margin,
            margin,
            page_rect.width - margin,
            page_rect.height - margin - footer_height,
        )

        pdf_buffer = io.BytesIO()
        story = pymupdf.Story(html=styled_html)
        writer = pymupdf.DocumentWriter(pdf_buffer)
        more: bool = True
        while more:
            device = writer.begin_page(page_rect)
            more, _ = story.place(content_rect)
            story.draw(device, None)
            writer.end_page()
        writer.close()

        doc = pymupdf.open(stream=pdf_buffer.getvalue(), filetype="pdf")
        total_pages: int = len(doc)
        for page_idx in range(total_pages):
            page = doc[page_idx]
            footer_rect = pymupdf.Rect(
                margin,
                page_rect.height - margin - 14.0,
                page_rect.width - margin,
                page_rect.height - margin + 6.0,
            )
            page.insert_textbox(
                footer_rect,
                f"{resolved_label} - Page {page_idx + 1} / {total_pages}",
                fontsize=8.5,
                fontname="helv",
                color=(0.45, 0.5, 0.58),
                align=pymupdf.TEXT_ALIGN_CENTER,
            )

        doc.save(resolved_pdf)
        doc.close()

        print(
            f"PDF paginé généré avec succès : {os.path.abspath(resolved_pdf)}",
            flush=True,
        )

import argparse
import os
import re
import sys
from typing import Dict, Optional, Tuple

CURRENT_DIR: str = os.path.dirname(os.path.abspath(__file__))
if CURRENT_DIR not in sys.path:
    sys.path.insert(0, CURRENT_DIR)

from doc_anonymizer.config_manager import PipelineConfig
from doc_anonymizer.extraction_runner import DocumentExtractionRunner
from doc_anonymizer.text_utils import DocumentTextUtils


class TranscriptionNameRestorer:
    """
    Restaure les noms d'origine des transcriptions ayant subi un masquage
    prématuré et les replace dans le bon sous-dossier périodique AAAA-MM.
    """

    HEADER_SOURCE_RE = re.compile(
        r"^(?:AUDIO|DOCUMENT PDF|DOCUMENT WORD|DOCUMENT ODT|DOCUMENT RTF|"
        r"CLASSEUR EXCEL|PRÉSENTATION PPTX|IMAGE \(OCR\)|PAGE WEB|TEXTE|"
        r"AGENDA ICS|EMAIL)\s*(?:\(([^)]+)\)|:\s*(.+?))\s*(?:\|.*)?$",
        re.MULTILINE,
    )

    def __init__(self, config: PipelineConfig, dry_run: bool = True) -> None:
        """
        Initialise le restaurateur avec la configuration du pipeline.

        :param config: Instance de PipelineConfig.
        :param dry_run: Si True, simule sans modifier les fichiers.
        """
        self.config: PipelineConfig = config
        self.dry_run: bool = dry_run
        self.extractor: DocumentExtractionRunner = DocumentExtractionRunner(
            config=config
        )

    def _build_masked_to_raw_map(self) -> Dict[Tuple[str, str], str]:
        """
        Construit une table (catégorie, ancien_nom_masqué) -> nom_brut
        à partir des fichiers présents dans le répertoire sources.

        :return: Dictionnaire de correspondance des noms.
        """
        target_phones, _ = self.extractor.load_all_filter_contacts()
        phone_map: Dict[str, str] = target_phones or {}
        mapping: Dict[Tuple[str, str], str] = {}

        for src_path in self.extractor.collect_source_files():
            file_name: str = os.path.basename(src_path)
            stem, ext_raw = os.path.splitext(file_name)
            cat: str = self.extractor.categorize_extension(ext_raw.lower())
            raw_stem: str = DocumentExtractionRunner._sanitize_filesystem_stem(
                stem
            )
            old_masked: str = DocumentTextUtils.mask_identifiers_in_filename(
                stem, phone_map
            )
            old_masked_clean: str = (
                DocumentExtractionRunner._sanitize_filesystem_stem(old_masked)
            )
            if old_masked_clean != raw_stem:
                mapping[(cat, f"{old_masked_clean}.txt")] = f"{raw_stem}.txt"
        return mapping
    def _extract_raw_name_from_header(self, txt_path: str) -> Optional[str]:
        """
        Lit les premières lignes d'une transcription pour récupérer le nom
        du fichier source enregistré dans l'en-tête.

        :param txt_path: Chemin complet vers le fichier .txt.
        :return: Nom de fichier .txt restauré ou None.
        """
        try:
            with open(txt_path, "r", encoding="utf-8", errors="ignore") as f:
                head: str = f.read(1500)
        except OSError:
            return None

        match = self.HEADER_SOURCE_RE.search(head)
        if not match:
            return None
        source_file_name: str = (match.group(1) or match.group(2) or "").strip()
        if not source_file_name:
            return None
        stem, _ = os.path.splitext(os.path.basename(source_file_name))
        raw_stem: str = DocumentExtractionRunner._sanitize_filesystem_stem(stem)
        return f"{raw_stem}.txt"

    def _resolve_target_path(
        self, current_root: str, category: str, target_name: str
    ) -> str:
        """
        Détermine le chemin cible transcriptions/AAAA-MM/Catégorie/fichier.txt
        (hors SMS/Emails déjà ventilés par message) si une date est détectée.

        :param current_root: Dossier parent actuel du fichier.
        :param category: Catégorie thématique (ex: Audio, PDF, SMS).
        :param target_name: Nom de fichier restauré.
        :return: Chemin complet cible sur disque.
        """
        if category in {"SMS", "Emails"}:
            return os.path.join(current_root, target_name)

        detected_period: Optional[str] = (
            DocumentTextUtils.detect_period_in_text(target_name)
        )
        if detected_period:
            return os.path.join(
                self.config.transcriptions_dir,
                detected_period,
                category,
                target_name,
            )
        return os.path.join(current_root, target_name)

    def _cleanup_empty_dirs(self, base_dir: str) -> None:
        """
        Supprime les sous-dossiers devenus vides après déplacement.

        :param base_dir: Répertoire racine transcriptions.
        """
        for root_path, dirs, files in os.walk(base_dir, topdown=False):
            if root_path == base_dir:
                continue
            if not dirs and not files:
                try:
                    os.rmdir(root_path)
                except OSError:
                    pass
    def run(self) -> int:
        """
        Parcourt transcriptions, restaure les noms et replace les fichiers
        dans leur dossier périodique AAAA-MM exact.

        :return: Nombre de fichiers déplacés ou renommés.
        """
        trans_dir: str = self.config.transcriptions_dir
        if not os.path.isdir(trans_dir):
            print(f"Dossier introuvable : '{trans_dir}'", flush=True)
            return 0

        fallback_map: Dict[Tuple[str, str], str] = (
            self._build_masked_to_raw_map()
        )
        moved_count: int = 0

        for root_path, _, files in os.walk(trans_dir):
            category: str = os.path.basename(root_path)
            for file_name in sorted(files):
                if (
                    file_name.startswith(".")
                    or file_name == "rapport_extraction.txt"
                    or not file_name.lower().endswith(".txt")
                ):
                    continue

                full_path: str = os.path.join(root_path, file_name)
                target_name: str = fallback_map.get(
                    (category, file_name), file_name
                )
                if target_name == file_name and (
                    "NUMERO_MASQUE" in file_name or "EMAIL_MASQUE" in file_name
                ):
                    extracted = self._extract_raw_name_from_header(full_path)
                    if extracted:
                        target_name = extracted

                new_path: str = self._resolve_target_path(
                    root_path, category, target_name
                )
                if os.path.normpath(new_path) == os.path.normpath(full_path):
                    continue

                if os.path.exists(new_path):
                    print(
                        f"[IGNORÉ - EXISTE DÉJÀ] '{full_path}' -> '{new_path}'",
                        flush=True,
                    )
                    continue

                rel_src: str = os.path.relpath(full_path, trans_dir)
                rel_dst: str = os.path.relpath(new_path, trans_dir)
                mode_tag: str = "[SIMULATION]" if self.dry_run else "[DÉPLACÉ]"
                print(f"{mode_tag} '{rel_src}' -> '{rel_dst}'", flush=True)

                if not self.dry_run:
                    os.makedirs(os.path.dirname(new_path), exist_ok=True)
                    os.rename(full_path, new_path)
                moved_count += 1

        if not self.dry_run and moved_count > 0:
            self._cleanup_empty_dirs(trans_dir)

        action: str = "à déplacer/renommer" if self.dry_run else "traité(s)"
        print(f"\nBilan : {moved_count} fichier(s) {action}.", flush=True)
        return moved_count


def main() -> None:
    """Point d'entrée du script provisoire de restauration des noms."""
    parser = argparse.ArgumentParser(
        description="Restaure les noms bruts et classe par période AAAA-MM."
    )
    parser.add_argument(
        "-c", "--config",
        default="config.ini",
        help="Chemin vers le fichier config.ini (défaut : config.ini).",
    )
    parser.add_argument(
        "--apply",
        action="store_true",
        help="Applique réellement les renommages et déplacements.",
    )
    args = parser.parse_args()
    config = PipelineConfig(config_path=args.config)
    restorer = TranscriptionNameRestorer(
        config=config, dry_run=not args.apply
    )
    restorer.run()


if __name__ == "__main__":
    main()

    de l'étape 1 d'extraction multi-formats.
    """

    @staticmethod
    def resolve_config_path(raw_config: str) -> str:
        """
        Résout le chemin vers config.ini en vérifiant le répertoire courant
        puis le répertoire parent racine.

        :param raw_config: Chemin brut passé en argument.
        :return: Chemin absolu existant vers config.ini.
        """
        if os.path.isabs(raw_config):
            if os.path.exists(raw_config):
                return raw_config
            raise FileNotFoundError(
                f"Fichier de configuration introuvable : '{raw_config}'"
            )

        local_candidate: str = os.path.abspath(raw_config)
        if os.path.exists(local_candidate):
            return local_candidate

        parent_candidate: str = os.path.normpath(
            os.path.join(CURRENT_DIR, "..", raw_config)
        )
        if os.path.exists(parent_candidate):
            return parent_candidate

        raise FileNotFoundError(
            f"Fichier de configuration '{raw_config}' introuvable."
        )

    @staticmethod
    def parse_args() -> argparse.Namespace:
        """
        Analyse les arguments de ligne de commande pour l'extraction de documents.

        :return: Espace de noms contenant les options fournies.
        """
        parser: argparse.ArgumentParser = argparse.ArgumentParser(
            description=(
                "Extrait et transcrit les documents et communications "
                "vers le dossier transcriptions."
            )
        )
        parser.add_argument(
            "-c", "--config",
            default="config.ini",
            help="Chemin vers le fichier config.ini (par défaut : config.ini).",
        )
        return parser.parse_args()

    @classmethod
    def run(cls) -> None:
        """
        Point d'entrée exécutant l'extraction multi-formats.
        """
        args: argparse.Namespace = cls.parse_args()
        config_path: str = cls.resolve_config_path(args.config)
        config: PipelineConfig = PipelineConfig(config_path=config_path)
        runner: DocumentExtractionRunner = DocumentExtractionRunner(
            config=config
        )
        runner.run()


def main() -> None:
    """
    Point d'entrée principal coordonnant l'étape 1 d'extraction multi-formats.
    """
    ExtractionCliLauncher.run()

    main()

import argparse
import os
import shutil
import tempfile
from typing import Optional


class ContactsSynchronizer:
    """
    Gère l'initialisation atomique de contacts_manuels.txt et la copie
    sécurisée du référentiel contacts.txt vers le projet documents.
    """

    @staticmethod
    def atomic_write_text(target_path: str, content: str) -> None:
        """
        Écrit un fichier texte de façon atomique via un fichier temporaire.

        :param target_path: Chemin final du fichier à créer.
        :param content: Contenu UTF-8 complet à enregistrer.
        """
        parent_dir: str = os.path.dirname(os.path.abspath(target_path))
        if parent_dir:
            os.makedirs(parent_dir, exist_ok=True)
        fd, tmp_path = tempfile.mkstemp(
            dir=parent_dir, prefix=".tmp_sync_", suffix=".tmp"
        )
        try:
            with os.fdopen(fd, "w", encoding="utf-8", newline="\n") as f_tmp:
                f_tmp.write(content)
            os.replace(tmp_path, target_path)
        except Exception:
            if os.path.exists(tmp_path):
                try:
                    os.remove(tmp_path)
                except OSError:
                    pass
            raise

    @classmethod
    def ensure_manual_contacts_file(cls, manual_path: str) -> None:
        """
        Crée le fichier contacts_manuels.txt avec son en-tête de façon atomique
        s'il n'existe pas encore.

        :param manual_path: Chemin vers le fichier des règles manuelles.
        """
        resolved_manual: str = os.path.abspath(manual_path)
        if os.path.exists(resolved_manual):
            return
        header: str = (
            "# RÈGLES D'ANONYMISATION MANUELLES\n"
            "# Ce fichier n'est jamais écrasé par lancer_contacts.py.\n"
            "# Ajoutez ici les noms issus de audit_noms_residuels.txt :\n"
            "# Format : Prénom Nom = Alias\n\n"
        )
        cls.atomic_write_text(resolved_manual, header)
        print(
            f"[OK] Fichier manuel initialisé : '{resolved_manual}'.",
            flush=True,
        )
    @classmethod
    def merge_and_sync(
        cls,
        source_path: str,
        dest_path: str,
        preserve_manual_rules: bool = True,
        manual_path: Optional[str] = None,
    ) -> None:
        """
        Copie le fichier contacts.txt généré automatiquement vers la destination
        en vérifiant qu'il n'est pas vide et en gardant les règles manuelles isolées.

        :param source_path: Chemin vers le fichier contacts.txt source.
        :param dest_path: Chemin de destination de contacts.txt.
        :param preserve_manual_rules: Préserve le fichier contacts_manuels.txt dédié.
        :param manual_path: Chemin optionnel vers contacts_manuels.txt.
        """
        resolved_src: str = os.path.abspath(source_path)
        resolved_dst: str = os.path.abspath(dest_path)

        if not os.path.exists(resolved_src):
            raise FileNotFoundError(
                f"Fichier source introuvable : '{resolved_src}'"
            )

        if os.path.getsize(resolved_src) == 0:
            raise ValueError(
                f"Le fichier source '{resolved_src}' est vide : copie annulée "
                "pour protéger la destination."
            )

        dst_dir: str = os.path.dirname(resolved_dst)
        if dst_dir:
            os.makedirs(dst_dir, exist_ok=True)

        if resolved_src != resolved_dst:
            shutil.copy2(resolved_src, resolved_dst)
            print(
                f"[OK] '{resolved_src}' copié vers '{resolved_dst}'.",
                flush=True,
            )
        else:
            print(
                f"[INFO] Source et destination identiques ('{resolved_dst}').",
                flush=True,
            )

        target_manual: str = (
            manual_path
            if manual_path
            else os.path.join(os.path.dirname(resolved_dst), "contacts_manuels.txt")
        )
        if preserve_manual_rules:
            cls.ensure_manual_contacts_file(target_manual)


def ensure_manual_contacts_file(manual_path: str) -> None:
    """Wrapper rétrocompatible vers ContactsSynchronizer.ensure_manual_contacts_file."""

    """

import os
import time
from typing import List
from doc_anonymizer.audit_state_manager import AuditAndStateManager
from doc_anonymizer.comm_extractors import CommunicationExtractor
from doc_anonymizer.config_manager import PipelineConfig
from doc_anonymizer.contact_rules_loader import ContactRulesLoader
from doc_anonymizer.text_utils import DocumentTextUtils


class DocumentAnonymizationRunner:
    """
    Orchestre l'anonymisation par lots au fil de l'eau et l'audit
    en temps réel des noms propres résiduels.
    """

    def __init__(
        self, config: PipelineConfig, manual_only: bool = False
    ) -> None:
        """
        Initialise le moteur d'anonymisation et le gestionnaire d'état.

        :param config: Instance de PipelineConfig.
        :param manual_only: Si True, utilise uniquement contacts_manuels.txt.
        """
        self.config: PipelineConfig = config
        self.manual_only: bool = manual_only
        self.rules: ContactRulesLoader = ContactRulesLoader(
            config, manual_only=manual_only
        )
        self.audit: AuditAndStateManager = AuditAndStateManager(
            config, self.rules
        )
        self._initialized: bool = False
        self.current_chunk: List[str] = []
        self.current_size: int = 0
        self.current_batch_files: List[str] = []
        self.processed_in_session: int = 0

    def initialize(self) -> bool:
        """
        Charge les règles de contacts et l'état de reprise.

        :return: True si l'initialisation a réussi, False sinon.
        """
        if self._initialized:
            return True
        if not self.rules.load_all():
            print(
                "[ERREUR ANONYMIZER] Chargement des règles impossible.",
                flush=True,
            )
            return False
        self.audit.load()
        print(
            f"[INIT ANONYMIZER] État chargé : "
            f"{len(self.audit.already_processed)} fichier(s) déjà traité(s), "
            f"prochain index de lot = {self.audit.file_idx}.",
            flush=True,
        )
        self._initialized = True
        return True
    def _anonymize_text(self, text: str) -> str:
        """
        Applique toutes les règles de masquage sur un texte en une seule passe
        via l'arbre préfixe compilé, puis traite les variantes collées et URLs.

        :param text: Texte brut à anonymiser.
        :return: Texte anonymisé.
        """
        token: str = CommunicationExtractor.MAIN_CHARACTER_TOKEN
        text = text.replace(token, self.config.main_character_name)

        if self.rules.combined_keyword_pattern is not None:
            text = self.rules.combined_keyword_pattern.sub(
                lambda m: self.rules.lookup_keyword_alias(m.group(0)), text
            )
        else:
            for pattern, alias in self.rules.compiled_rules:
                text = pattern.sub(alias, text)

        for glued_pattern, alias in self.rules.glued_rules:
            text = glued_pattern.sub(alias, text)

        if self.config.mask_ssn:
            text = DocumentTextUtils.REGEX_SSN.sub("[NUM_SECU_MASQUE]", text)
        if self.config.mask_financial_and_admin_ids:
            text = DocumentTextUtils.REGEX_IBAN.sub("[IBAN_MASQUE]", text)
            text = DocumentTextUtils.REGEX_BIC.sub("[BIC_MASQUE]", text)
            text = DocumentTextUtils.REGEX_FISCAL_ID.sub(
                "[NUM_FISCAL_MASQUE]", text
            )
            text = DocumentTextUtils.REGEX_FRANCE_TRAVAIL.sub(
                "[ID_FRANCE_TRAVAIL_MASQUE]", text
            )
            text = DocumentTextUtils.REGEX_CAF_ALLOCATAIRE.sub(
                "[NUM_ALLOCATAIRE_MASQUE]", text
            )
        text = DocumentTextUtils.REGEX_EMAIL.sub(
            lambda m: self.audit.pseudonymize_id(m.group(0), "EMAIL"), text
        )
        text = DocumentTextUtils.REGEX_PHONE.sub(
            lambda m: self.audit.pseudonymize_id(m.group(0), "TEL"), text
        )
        if self.config.mask_urls:
            text = DocumentTextUtils.REGEX_URL.sub("[URL_MASQUEE]", text)
        if self.config.mask_addresses:
            text = DocumentTextUtils.REGEX_STREET_ADDRESS.sub(
                "[ADRESSE_MASQUEE]", text
            )
            text = DocumentTextUtils.REGEX_POSTAL_CITY.sub(
                "[LIEU_MASQUE]", text
            )

        if self.rules.combined_location_pattern is not None:
            text = self.rules.combined_location_pattern.sub(
                "[LIEU_MASQUE]", text
            )
        else:
            for loc_pattern in self.rules.location_rules:
                text = loc_pattern.sub("[LIEU_MASQUE]", text)
        return text

    def _anonymize_display_path(self, trans_path: str) -> str:
        """
        Masque les données sensibles dans le chemin affiché.

        :param trans_path: Chemin vers le fichier transcription.
        :return: Chemin nettoyé de toute donnée personnelle.
        """
        folder, file_name = os.path.split(trans_path)
        masked: str = file_name
        if self.config.mask_ssn:
            masked = DocumentTextUtils.REGEX_SSN.sub("NUM_SECU_MASQUE", masked)
        if self.config.mask_financial_and_admin_ids:
            masked = DocumentTextUtils.REGEX_IBAN.sub("IBAN_MASQUE", masked)
        masked = DocumentTextUtils.REGEX_EMAIL.sub(
            lambda m: self.audit.pseudonymize_id(m.group(0), "EMAIL"), masked
        )
        masked = DocumentTextUtils.REGEX_PHONE.sub(
            lambda m: self.audit.pseudonymize_id(m.group(0), "TEL"), masked
        )
        if self.config.mask_urls:
            masked = DocumentTextUtils.REGEX_URL.sub("URL_MASQUEE", masked)
        if self.rules.combined_keyword_pattern is not None:
            masked = self.rules.combined_keyword_pattern.sub(
                lambda m: self.rules.lookup_keyword_alias(m.group(0)), masked
            )
        for glued_pattern, alias in self.rules.glued_rules:
            masked = glued_pattern.sub(alias, masked)
        if self.rules.combined_location_pattern is not None:
            masked = self.rules.combined_location_pattern.sub(
                "LIEU_MASQUE", masked
            )
        return os.path.join(folder, masked)
    def flush_batch(self) -> None:
        """
        Écrit le lot courant de façon atomique, sauvegarde l'état
        et actualise les rapports d'audit.
        """
        if not self.current_chunk:
            self.audit.save()
            self.audit.write_report()
            return
        batch_num: int = (
            (self.audit.file_idx - 1) // self.config.files_per_folder
        ) + 1
        sub_folder: str = f"lot_{batch_num:02d}"
        batch_dir: str = os.path.join(self.config.results_dir, sub_folder)
        os.makedirs(batch_dir, exist_ok=True)
        out_name: str = (
            f"{self.config.file_prefix}partie_{self.audit.file_idx:03d}.txt"
        )
        full_out_path: str = os.path.join(batch_dir, out_name)
        DocumentTextUtils.atomic_write_text(
            full_out_path, "".join(self.current_chunk)
        )

        self.audit.already_processed.update(self.current_batch_files)
        self.audit.file_idx += 1
        self.audit.written_batches += 1
        self.current_chunk = []
        self.current_size = 0
        self.current_batch_files = []

        self.audit.save()
        count: int = self.audit.write_report()
        print(
            f"    [+] LOT ÉCRIT : '{full_out_path}' (Audit : {count} nom(s))",
            flush=True,
        )

    def anonymize_single_file(
        self, trans_path: str, force_reprocess: bool = False
    ) -> bool:
        """
        Anonymise un fichier transcription et l'ajoute au lot courant,
        en déclenchant l'écriture disque dès que le seuil de taille ou
        de nombre de fichiers est atteint.

        :param trans_path: Chemin vers le fichier transcription.
        :param force_reprocess: Force le retraitement même si déjà indexé.
        :return: True si le fichier a été anonymisé, False sinon.
        """
        if not self._initialized and not self.initialize():
            return False
        norm_path: str = os.path.normpath(trans_path)
        if (
            not force_reprocess
            and norm_path in self.audit.already_processed
        ):
            return False
        if not os.path.exists(trans_path):
            print(
                f"    [AVERTISSEMENT ANONYMIZER] Fichier introuvable : "
                f"'{trans_path}'",
                flush=True,
            )
            return False

        with open(trans_path, "r", encoding="utf-8", errors="ignore") as f:
            raw_content: str = f.read()

        if not raw_content.strip():
            print(
                f"    [AVERTISSEMENT ANONYMIZER] Transcription vide : "
                f"'{trans_path}'",
                flush=True,
            )
            return False

        display_path: str = self._anonymize_display_path(trans_path)
        sep: str = "#" * 72
        header: str = (
            f"{sep}\n# SOURCE TRANSCRIPTION : {display_path}\n{sep}\n"
        )
        anonymized: str = self._anonymize_text(header + raw_content)
        self.audit.scan_residual_names(anonymized, display_path)

        block_bytes: int = len(anonymized.encode("utf-8"))
        max_files_per_batch: int = getattr(self.config, "max_batch_files", 50)
        if self.current_chunk and (
            self.current_size + block_bytes > self.config.max_batch_bytes
            or len(self.current_chunk) >= max_files_per_batch
        ):
            self.flush_batch()

        self.current_chunk.append(anonymized)
        self.current_size += block_bytes
        self.current_batch_files.append(norm_path)
        self.processed_in_session += 1
        return True
    def finalize(self, start_time: float) -> None:
        """
        Clôture le dernier lot et affiche le bilan final avec les deux audits.

        :param start_time: Horodatage de début de session.
        """
        self.flush_batch()
        count: int = self.audit.write_report()
        elapsed: float = time.time() - start_time
        print(
            f"\n[100.0%] Terminé ({self.processed_in_session} fichier(s)"
            f" anonymisé(s) en {elapsed:.1f}s).",
            flush=True,
        )
        print(
            f" * Lots totaux  : {self.audit.written_batches} dans"
            f" '{self.config.results_dir}/lot_XX/'",
            flush=True,
        )
        print(
            f" * Audit détail : {count} nom(s) dans"
            f" '{self.config.audit_report_file}'",
            flush=True,
        )
        print(
            f" * Audit copier : {count} nom(s) dans"
            f" '{self.config.audit_copy_file}'",
            flush=True,
        )

    def run_on_existing_results(self) -> None:
        """
        Ré-anonymise sur place (écriture atomique) les lots déjà présents dans
        results_dir en utilisant uniquement contacts_manuels.txt.
        """
        start_time: float = time.time()
        print(
            "=== PASSE SUR PLACE : CONTACTS MANUELS SUR RÉSULTATS ===",
            flush=True,
        )
        if not self.initialize():
            return

        res_dir: str = self.config.results_dir
        if not os.path.isdir(res_dir):
            print(f"Erreur : '{res_dir}' introuvable.", flush=True)
            return

        lot_files: List[str] = sorted(
            os.path.join(rp, fn)
            for rp, _, files in os.walk(res_dir)
            for fn in files
            if fn.lower().endswith(".txt") and not fn.startswith(".")
        )
        if not lot_files:
            print(
                f"Aucun fichier de lot trouvé dans '{res_dir}'.",
                flush=True,
            )
            return

        self.audit.pair_counter.clear()
        self.audit.files_by_name.clear()
        self.audit.canonical_labels.clear()

        total: int = len(lot_files)
        for idx, lot_path in enumerate(lot_files, start=1):
            with open(lot_path, "r", encoding="utf-8", errors="ignore") as f_in:
                content: str = f_in.read()
            updated_content: str = self._anonymize_text(content)
            DocumentTextUtils.atomic_write_text(lot_path, updated_content)
            self.audit.scan_residual_names(
                updated_content, os.path.basename(lot_path)
            )
            if idx % 10 == 0 or idx == total or idx == 1:
                print(
                    f"    ... [{idx}/{total}] {os.path.basename(lot_path)}",
                    flush=True,
                )

        self.audit.save()
        count: int = self.audit.write_report()
        elapsed: float = time.time() - start_time
        print(
            f"\n[100.0%] Passe manuelle terminée sur {total} lot(s) "
            f"en {elapsed:.1f}s.",
            flush=True,
        )
        print(
            f" * Audit détail : {count} nom(s) dans"
            f" '{self.config.audit_report_file}'",
            flush=True,
        )
        print(
            f" * Audit copier : {count} nom(s) dans"
            f" '{self.config.audit_copy_file}'",
            flush=True,
        )

    def run(self) -> None:
        """
        Exécute l'anonymisation autonome des transcriptions en attente.
        """
        if self.manual_only:
            self.run_on_existing_results()
            return

        start_time: float = time.time()
        print("=== ÉTAPE 2 : ANONYMISATION & AUDIT ===", flush=True)
        if not self.initialize():
            return
        trans_dir: str = self.config.transcriptions_dir
        if not os.path.exists(trans_dir):
            print(f"Erreur : '{trans_dir}' introuvable.", flush=True)
            return

        all_txts: List[str] = sorted(
            os.path.join(rp, fn)
            for rp, _, files in os.walk(trans_dir)
            for fn in files
            if fn.lower().endswith(".txt")
            and fn != "rapport_extraction.txt"
            and not fn.startswith(".")
        )
        pending: List[str] = [
            f
            for f in all_txts
            if os.path.normpath(f) not in self.audit.already_processed
        ]
        total: int = len(pending)
        for idx, trans_path in enumerate(pending, start=1):
            if idx % 10 == 0 or idx == total or idx == 1:
                pct: float = (idx / total) * 100
                print(
                    f"    ... [{pct:5.1f}% | {idx}/{total}] ...",
                    flush=True,
                )
            self.anonymize_single_file(trans_path)

        self.finalize(start_time)

    def __init__(
        self, config: PipelineConfig, rules: ContactRulesLoader
    ) -> None:
        """
        Initialise le gestionnaire d'état et les compteurs d'audit.

        :param config: Instance de PipelineConfig.
        :param rules: Instance de ContactRulesLoader.
        """
        self.config: PipelineConfig = config
        self.rules: ContactRulesLoader = rules
        self.already_processed: Set[str] = set()
        self.auto_cache: Dict[str, str] = {}
        self.counters: Dict[str, int] = {"TEL": 1, "EMAIL": 1}
        self.pair_counter: Counter[str] = Counter()
        self.files_by_name: DefaultDict[str, Set[str]] = defaultdict(set)

    @staticmethod
    def _phonetic_french_key(word: str) -> str:
        """
        Calcule une clé phonétique française simplifiée pour rapprocher les
        erreurs de transcription vocale/OCR (ex: 'dubunt' ~ 'dupont').

        :param word: Mot normalisé sans accents.
        :return: Empreinte phonétique compacte.
        """
        s: str = DocumentTextUtils.normalize_word(word)
        s = re.sub(r"[^a-z]", "", s)
        if not s:
            return ""
        s = s.replace("ph", "f").replace("qu", "k").replace("q", "k")
        s = s.replace("c", "k").replace("g", "k")
        s = s.replace("p", "b").replace("t", "d")
        s = s.replace("v", "f").replace("z", "s")
        s = re.sub(r"[aeiouy]+", "a", s)
        s = re.sub(r"(.)\1+", r"\1", s)
        if len(s) > 3 and s[-1] in {"d", "s", "x"}:
            s = s[:-1]
        return s

    def _suggest_alias_from_fuzzy_last_name(
        self, display_name: str
    ) -> Optional[Tuple[str, str]]:
        """
        Recherche si le nom candidat (ou son nom de famille) est proche
        orthographiquement ou phonétiquement d'un nom connu dans contacts.

        :param display_name: Libellé candidat détecté par l'audit.
        :return: Tuple (alias_suggéré, nom_famille_proche) ou None.
        """
        if not self.rules.last_name_to_aliases:
            return None

        tokens: List[str] = display_name.strip().split()
        candidate_last: str = tokens[-1] if tokens else display_name.strip()
        norm_cand: str = DocumentTextUtils.normalize_word(candidate_last)
        if len(norm_cand) < 4:
            return None

        cand_phon: str = self._phonetic_french_key(norm_cand)
        best_score: float = 0.0
        best_known: str = ""

        for known_last in self.rules.last_name_to_aliases.keys():
            if len(known_last) < 4 or known_last == norm_cand:
                continue
            ratio: float = SequenceMatcher(
                None, norm_cand, known_last
            ).ratio()
            known_phon: str = self._phonetic_french_key(known_last)
            if cand_phon and cand_phon == known_phon:
                ratio = max(ratio, 0.85)
            if ratio > best_score:
                best_score = ratio
                best_known = known_last

        if best_score >= 0.72 and best_known:
            aliases: List[str] = sorted(
                self.rules.last_name_to_aliases[best_known]
            )
            if len(aliases) == 1:
                disp_known: str = self.rules.last_name_display.get(
                    best_known, best_known.capitalize()
                )
                return next(iter(aliases)), disp_known
        return None
    @staticmethod
    def _format_first_name_case(token: str) -> str:
        """
        Capitalise proprement un prénom (gère les prénoms composés et majuscules).

        :param token: Jeton prénom brut.
        :return: Prénom mis en forme (ex: 'FLORIAN' -> 'Florian').
        """
        parts: List[str] = re.split(r"(-)", token.strip())
        return "".join(p.capitalize() if p != "-" else p for p in parts)

    def _canonicalize_key(self, raw_key: str) -> Tuple[str, str]:
        """
        Ordonne une paire en 'Prénom Nom' grâce au référentiel des prénoms
        et retourne (clé_normalisée_unique, libellé_affichable).

        :param raw_key: Clé candidate ('Nom Prénom', 'Prénom Nom' ou nom seul).
        :return: Tuple (identifiant_canonique, libellé_d_affichage).
        """
        tokens: List[str] = raw_key.strip().split()
        if len(tokens) == 2:
            p1, p2 = tokens
            f1: bool = self.rules.is_first_name(p1)
            f2: bool = self.rules.is_first_name(p2)
            if f2 and not f1:
                first_tok, last_tok = p2, p1
            else:
                first_tok, last_tok = p1, p2
            display_first: str = self._format_first_name_case(first_tok)
            display_label: str = f"{display_first} {last_tok}"
        else:
            display_label = raw_key.strip()

        norm_id: str = DocumentTextUtils.normalize_word(display_label)
        existing: str = self.canonical_labels.get(norm_id, "")
        if not existing or (existing.isupper() and not display_label.isupper()):
            self.canonical_labels[norm_id] = display_label
        return norm_id, self.canonical_labels[norm_id]

    def _is_valid_word(self, word: str) -> bool:
        """
        Vérifie qu'un mot candidat est un nom propre plausible et non un
        jeton masqué, un mot-outil français ou un terme exclu.

        :param word: Mot brut à évaluer.
        :return: True si le mot est éligible pour l'audit.
        """
        cleaned: str = word.strip()
        if len(cleaned) < 2 or cleaned.upper().startswith(self.MASKED_PREFIXES):
            return False
        if any(ch.isdigit() for ch in cleaned):
            return False
        norm: str = DocumentTextUtils.normalize_word(cleaned)
        if len(norm) < 2:
            return False
        return (
            norm not in self.COMMON_FRENCH_STOPWORDS
            and norm not in self.config.excluded_words
            and norm not in self.rules.alias_words
            and norm not in self.config.masked_locations
        )

    def _is_plausible_pair(self, p1: str, p2: str) -> bool:
        """
        Valide une paire (p1, p2) en écartant les faux positifs liés aux
        prénoms ambigus (mots courants) et aux sigles tout en majuscules.

        :param p1: Premier mot candidat.
        :param p2: Second mot candidat.
        :return: True si la paire est un nom de personne probable.
        """
        if not (self._is_valid_word(p1) and self._is_valid_word(p2)):
            return False
        if p1.isupper() and p2.isupper() and len(p1) <= 4 and len(p2) <= 4:
            return False

        f1: bool = self.rules.is_first_name(p1)
        f2: bool = self.rules.is_first_name(p2)
        if not (f1 or f2) or (f1 and f2):
            return False

        first_tok: str = p1 if f1 else p2
        last_tok: str = p2 if f1 else p1
        norm_first: str = DocumentTextUtils.normalize_word(first_tok)
        norm_last: str = DocumentTextUtils.normalize_word(last_tok)

        if len(norm_last) < 3:
            return False
        if norm_first in self.AMBIGUOUS_FIRST_NAMES:
            if (
                norm_last not in self.rules.last_name_to_aliases
                and not last_tok.isupper()
            ):
                return False

        nk: str = DocumentTextUtils.normalize_word(f"{p1} {p2}")
        ni: str = DocumentTextUtils.normalize_word(f"{p2} {p1}")
        return (
            nk not in self.rules.full_targets
            and ni not in self.rules.full_targets
        )
    def _is_entry_still_valid(self, display_name: str) -> bool:
        """
        Vérifie si une entrée déjà présente dans l'état JSON reste valide
        après application des nouveaux filtres anti-faux positifs.

        :param display_name: Libellé enregistré dans l'état.
        :return: True si l'entrée doit être conservée.
        """
        tokens: List[str] = display_name.strip().split()
        if len(tokens) == 2:
            tok_first, tok_second = tokens
            return self._is_plausible_pair(tok_first, tok_second)
        if len(tokens) == 1:
            tok: str = next(iter(tokens))
            norm: str = DocumentTextUtils.normalize_word(tok)
            return (
                self._is_valid_word(tok)
                and len(norm) >= 3
                and norm not in self.rules.full_targets
            )
        return False

    def load(self) -> None:
        """Restaure, filtre et fusionne l'état JSON depuis le disque s'il existe."""
        os.makedirs(self.config.results_dir, exist_ok=True)
        if not os.path.exists(self.config.resume_state_file):
            return
        try:
            with open(
                self.config.resume_state_file, "r", encoding="utf-8"
            ) as f:
                st: Dict[str, Any] = json.load(f)
            self.already_processed = set(st.get("fichiers_traites", []))
            self.auto_cache = st.get("cache_auto", {})
            self.counters = st.get("compteurs", {"TEL": 1, "EMAIL": 1})
            self.file_idx = st.get("idx_fichier", 1)
            self.written_batches = st.get("nb_lots_ecrits", 0)

            raw_counts: Dict[str, int] = st.get("compteur_paires", {})
            raw_files: Dict[str, List[str]] = st.get("fichiers_par_nom", {})
            for raw_name, count in raw_counts.items():
                if not self._is_entry_still_valid(raw_name):
                    continue
                norm_id, _ = self._canonicalize_key(raw_name)
                self.pair_counter[norm_id] += int(count)
                self.files_by_name[norm_id].update(raw_files.get(raw_name, []))
        except Exception:
            pass

    def save(self) -> None:
        """Sauvegarde l'état courant sur le disque de façon atomique."""
        os.makedirs(self.config.results_dir, exist_ok=True)
        exported_counts: Dict[str, int] = {
            self.canonical_labels.get(nid, nid): cnt
            for nid, cnt in self.pair_counter.items()
        }
        exported_files: Dict[str, List[str]] = {
            self.canonical_labels.get(nid, nid): sorted(paths)
            for nid, paths in self.files_by_name.items()
        }
        payload: Dict[str, Any] = {
            "fichiers_traites": sorted(self.already_processed),
            "idx_fichier": self.file_idx,
            "nb_lots_ecrits": self.written_batches,
            "cache_auto": self.auto_cache,
            "compteurs": self.counters,
            "compteur_paires": exported_counts,
            "fichiers_par_nom": exported_files,
        }
        DocumentTextUtils.atomic_write_text(
            self.config.resume_state_file,
            json.dumps(payload, ensure_ascii=False, indent=2),
        )

    def pseudonymize_id(self, raw_val: str, category: str) -> str:
        """Remplace un téléphone ou e-mail par son alias ou un jeton SHA-256."""
        norm_id: str = DocumentTextUtils.normalize_identifier(raw_val)
        if norm_id in self.rules.contacts_id:
            return self.rules.contacts_id[norm_id]
        salted: bytes = (norm_id + self.config.secret_salt).encode("utf-8")
        digest: str = hashlib.sha256(salted).hexdigest()[:10]
        if digest not in self.auto_cache:
            self.auto_cache[digest] = (
                f"{category}_{self.counters[category]:03d}"
            )
            self.counters[category] += 1
        return self.auto_cache[digest]

    def scan_residual_names(self, text: str, display_path: str) -> None:
        """Détecte et regroupe les noms propres résiduels sous forme canonique."""
        for m in self.PAIR_PATTERN.finditer(text):
            p1, p2 = m.group(1), m.group(2)
            if not self._is_plausible_pair(p1, p2):
                continue
            norm_id, _ = self._canonicalize_key(f"{p1} {p2}")
            self.pair_counter[norm_id] += 1
            self.files_by_name[norm_id].add(display_path)

        for m in self.TITLE_PATTERN.finditer(text):
            name: str = m.group(1)
            norm_name: str = DocumentTextUtils.normalize_word(name)
            if (
                self._is_valid_word(name)
                and len(norm_name) >= 3
                and norm_name not in self.rules.full_targets
            ):
                norm_id, _ = self._canonicalize_key(name)
                self.pair_counter[norm_id] += 1
                self.files_by_name[norm_id].add(display_path)
    @staticmethod
    def _audit_sort_key(entry: Tuple[str, str, int]) -> Tuple[int, str]:
        """
        Calcule la clé de tri (occurrences décroissantes, puis libellé).

        :param entry: Triplet (identifiant_canonique, libellé, occurrences).
        :return: Tuple de comparaison pour sorted().
        """
        _, display_name, count = entry
        return (-count, display_name.lower())

    def _format_audit_entry(
        self, norm_id: str, display_name: str, count: int, pad_width: int
    ) -> Tuple[str, str]:
        """
        Formate la ligne complète d'audit ainsi que la ligne épurée prête
        à être copiée-collée dans contacts_manuels.txt.

        :param norm_id: Clé normalisée interne.
        :param display_name: Libellé canonique 'Prénom Nom'.
        :param count: Nombre total d'occurrences relevées.
        :param pad_width: Largeur d'alignement pour la colonne de gauche.
        :return: Tuple (ligne_audit_complet, ligne_copier_coller).
        """
        short_names: List[str] = sorted(
            {os.path.basename(p) for p in self.files_by_name[norm_id]}
        )
        preview: str = ", ".join(short_names[:3])
        extra: int = len(short_names) - 3
        if extra > 0:
            preview += f" (+{extra} autre(s))"

        suggestion = self._suggest_alias_from_fuzzy_last_name(display_name)
        if suggestion:
            suggested_alias, close_last = suggestion
            rule_col: str = f"{display_name:<{pad_width}} = {suggested_alias}"
            detailed: str = (
                f"{rule_col}   # [{count:>3}x] "
                f"(Proche de '{close_last}') {preview}\n"
            )
            copy_line: str = f"{display_name} = {suggested_alias}\n"
            return detailed, copy_line

        rule_col = f"{display_name:<{pad_width}} = A_MASQUER"
        detailed = f"{rule_col}   # [{count:>3}x] {preview}\n"
        copy_line = f"{display_name} = \n"
        return detailed, copy_line

    def write_report(self) -> int:
        """
        Actualise simultanément le rapport détaillé (audit_noms_residuels.txt)
        et le fichier épuré pour copier-coller (audit_noms_residuels_copier.txt).

        :return: Nombre total de noms propres résiduels détectés.
        """
        manual_name: str = os.path.basename(self.config.manual_contacts_file)
        copy_name: str = os.path.basename(self.config.audit_copy_file)
        results: List[Tuple[str, str, int]] = sorted(
            [
                (nid, self.canonical_labels.get(nid, nid), cnt)
                for nid, cnt in self.pair_counter.items()
                if cnt >= self.config.min_audit_occurrences
            ],
            key=self._audit_sort_key,
        )
        sep_main: str = "=" * 78
        sep_sub: str = "-" * 78
        report_lines: List[str] = [
            f"# {sep_main}\n",
            f"# RAPPORT D'AUDIT DES NOMS PROPRES RÉSIDUELS\n",
            f"# Fichiers analysés : {len(self.already_processed)} | "
            f"Noms détectés : {len(results)} | "
            f"Seuil min : {self.config.min_audit_occurrences}x\n",
            f"# {sep_main}\n",
            "# MODE D'EMPLOI :\n",
            f"#   1. Ouvrez '{copy_name}' pour récupérer directement les "
            "lignes 'Prénom Nom = ' sans texte superflu.\n",
            f"#   2. Complétez l'alias (ex: Prénom I.) et collez-les "
            f"dans '{manual_name}' (jamais écrasé).\n",
            "#   3. Ajoutez les faux positifs récurrents dans "
            "'excluded_words' (config.ini).\n",
            f"# {sep_sub}\n\n",
        ]
        copy_lines: List[str] = []

        if not results:
            report_lines.append(
                "Aucun nouveau nom propre résiduel probable détecté.\n"
            )
        else:
            pad_width: int = max(len(disp) for _, disp, _ in results) + 1
            tiers: List[Tuple[str, List[Tuple[str, str, int]]]] = [
                (
                    " PRIORITÉ HAUTE (5 occurrences et plus)",
                    [(nid, disp, c) for nid, disp, c in results if c >= 5],
                ),
                (
                    " PRIORITÉ MOYENNE (2 à 4 occurrences)",
                    [
                        (nid, disp, c)
                        for nid, disp, c in results
                        if 2 <= c <= 4
                    ],
                ),
                (
                    " OCCURRENCES UNIQUES (1 seule apparition)",
                    [(nid, disp, c) for nid, disp, c in results if c == 1],
                ),
            ]
            for title, items in tiers:
                if not items:
                    continue
                report_lines.append(
                    f"# {sep_sub}\n"
                    f"# {title} — {len(items)} entrée(s)\n"
                    f"# {sep_sub}\n"
                )
                for nid, disp, count in items:
                    det_ln, cp_ln = self._format_audit_entry(
                        nid, disp, count, pad_width
                    )
                    report_lines.append(det_ln)
                    copy_lines.append(cp_ln)
                report_lines.append("\n")

        DocumentTextUtils.atomic_write_text(
            self.config.audit_report_file, "".join(report_lines)
        )
        DocumentTextUtils.atomic_write_text(
            self.config.audit_copy_file, "".join(copy_lines)
        )

        return len(results)

    @staticmethod
    def decode_mime_header(raw_val: Optional[str]) -> str:
        """
        Décode un en-tête MIME (ex. =?UTF-8?Q?...?=) en chaîne Unicode.

        :param raw_val: Valeur brute de l'en-tête.
        :return: Texte décodé.
        """
        if not raw_val:
            return ""
        parts: List[str] = []
        for chunk, enc in decode_header(raw_val):
            if isinstance(chunk, bytes):
                parts.append(
                    chunk.decode(enc or "utf-8", errors="ignore")
                )
            else:
                parts.append(str(chunk))
        return "".join(parts).strip()

    @staticmethod
    def _extract_email_body(msg: Message) -> str:
        """
        Extrait le corps texte principal (text/plain ou repli text/html)
        d'un message MIME.

        :param msg: Objet email.message.Message.
        :return: Texte nettoyé du corps du message.
        """
        plain_parts: List[str] = []
        html_parts: List[str] = []

        if msg.is_multipart():
            for part in msg.walk():
                ctype: str = part.get_content_type()
                disp: str = str(part.get("Content-Disposition") or "")
                if "attachment" in disp.lower():
                    continue
                if ctype in ("text/plain", "text/html"):
                    payload = part.get_payload(decode=True)
                    if not isinstance(payload, bytes):
                        continue
                    charset: str = part.get_content_charset() or "utf-8"
                    decoded: str = payload.decode(charset, errors="ignore")
                    if ctype == "text/plain":
                        plain_parts.append(decoded)
                    else:
                        html_parts.append(decoded)
        else:
            payload = msg.get_payload(decode=True)
            if isinstance(payload, bytes):
                charset = msg.get_content_charset() or "utf-8"
                decoded = payload.decode(charset, errors="ignore")
                if msg.get_content_type() == "text/html":
                    html_parts.append(decoded)
                else:
                    plain_parts.append(decoded)

        if plain_parts:
            return "\n".join(plain_parts).strip()
        if html_parts:
            return DocumentTextUtils.html_to_plain_text(
                "\n".join(html_parts)
            ).strip()
        return ""

    def _extract_email_pdf_attachments(
        self, msg: Message, source_label: str
    ) -> List[str]:
        """
        Extrait le texte des pièces jointes PDF d'un message MIME.

        :param msg: Objet email.message.Message.
        :param source_label: Nom du fichier source pour l'affichage.
        :return: Liste des textes extraits des pièces jointes PDF.
        """
        attachments_txt: List[str] = []
        if not msg.is_multipart():
            return attachments_txt

        for part in msg.walk():
            raw_file_name: Optional[str] = part.get_filename()
            file_name: str = self.decode_mime_header(raw_file_name)
            ctype: str = part.get_content_type()
            if ctype == "application/pdf" or file_name.lower().endswith(".pdf"):
                pdf_bytes = part.get_payload(decode=True)
                if isinstance(pdf_bytes, bytes) and pdf_bytes:
                    disp_name: str = file_name or f"piece_jointe_{source_label}.pdf"
                    pdf_txt, err = self.doc_extractor.extract_pdf(
                        pdf_source=pdf_bytes,
                        display_name=disp_name,
                        is_stream=True,
                    )
                    if pdf_txt and not err:
                        attachments_txt.append(pdf_txt)
        return attachments_txt
    def extract_emails(
        self,
        file_path: str,
        allowed_emails: Set[str],
        discovered_emails: Set[str],
    ) -> Tuple[Dict[str, List[str]], Optional[str]]:
        """
        Extrait en streaming les e-mails d'un fichier .mbox ou .eml correspondant
        aux adresses de contacts.txt sans saturer la RAM.

        :param file_path: Chemin vers le fichier .mbox ou .eml.
        :param allowed_emails: Ensemble des e-mails filtrés depuis contacts.txt.
        :param discovered_emails: Collecteur global des e-mails rencontrés.
        :return: Tuple (dictionnaire_par_mois, message_erreur).
        """
        file_name: str = os.path.basename(file_path)
        is_eml: bool = file_name.lower().endswith(".eml")
        eml_filename_period: Optional[str] = (
            DocumentTextUtils.detect_period_in_text(file_name)
            if is_eml
            else None
        )
        by_month: DefaultDict[str, List[str]] = defaultdict(list)
        mbox_handle: Optional[mailbox.mbox] = None

        try:
            msg_iter: Iterable[Message]
            if is_eml:
                with open(file_path, "rb") as f_eml:
                    msg_iter = [email.message_from_binary_file(f_eml)]
            else:
                mbox_handle = mailbox.mbox(file_path)
                msg_iter = mbox_handle

            scanned_count: int = 0
            matched_count: int = 0

            for msg in msg_iter:
                scanned_count += 1
                if not is_eml and scanned_count % 500 == 0:
                    print(
                        f"    ... {file_name} : {scanned_count} e-mail(s) parcouru(s) "
                        f"({matched_count} retenu(s)) ...",
                        flush=True,
                    )

                raw_headers: List[str] = [
                    str(msg.get(h) or "")
                    for h in ("From", "To", "Cc", "Bcc", "Reply-To")
                ]
                parsed_addrs = getaddresses(raw_headers)
                msg_emails: Set[str] = {
                    DocumentTextUtils.normalize_identifier(addr)
                    for _, addr in parsed_addrs
                    if addr and "@" in addr
                }
                discovered_emails.update(msg_emails)

                if allowed_emails and not (msg_emails & allowed_emails):
                    continue

                matched_count += 1
                raw_date: str = str(msg.get("Date") or "").strip()
                period: str = eml_filename_period or "0000-00"
                date_str: str = raw_date or "Date inconnue"
                if raw_date:
                    try:
                        dt = parsedate_to_datetime(raw_date)
                        date_str = dt.strftime("%Y-%m-%d %H:%M:%S")
                        if not eml_filename_period:
                            period = dt.strftime("%Y-%m")
                    except Exception:
                        if not eml_filename_period:
                            detected = DocumentTextUtils.detect_period_in_text(
                                raw_date
                            )
                            if detected:
                                period = detected

                sender: str = self.decode_mime_header(msg.get("From"))
                recipients: str = self.decode_mime_header(msg.get("To"))
                subject: str = self.decode_mime_header(msg.get("Subject"))
                body: str = self._extract_email_body(msg)
                pdf_att: List[str] = self._extract_email_pdf_attachments(
                    msg, file_name
                )

                block_lines: List[str] = [
                    "========================================\n",
                    f"EMAIL ({file_name}) | DATE : {date_str}\n",
                    f"DE : {sender}\n",
                    f"À : {recipients}\n",
                    f"OBJET : {subject}\n",
                    "----------------------------------------\n",
                    f"{body}\n\n",
                ]
                if pdf_att:
                    block_lines.extend(pdf_att)
                by_month[period].append("".join(block_lines))

            if not is_eml:
                print(
                    f"    -> Fin {file_name} : {matched_count}/{scanned_count} "
                    "e-mail(s) retenu(s).",
                    flush=True,
                )

            return dict(by_month), None
        except Exception as exc:
            return {}, f"Erreur lecture e-mail ({exc})"
        finally:
            if mbox_handle is not None:
                mbox_handle.close()
    @staticmethod
    def _validate_safe_xml_header(xml_path: str) -> None:
        """
        Vérifie l'en-tête du fichier XML pour bloquer toute déclaration
        DOCTYPE ou ENTITY (protection contre Billion Laughs / XXE).

        :param xml_path: Chemin vers le fichier XML.
        """
        with open(xml_path, "rb") as f_head:
            head_bytes: bytes = f_head.read(4096)
        head_upper: bytes = head_bytes.upper()
        if b"<!DOCTYPE" in head_upper or b"<!ENTITY" in head_upper:
            raise ValueError(
                "Déclaration DOCTYPE/ENTITY interdite par sécurité dans le XML."
            )

    def parse_sms_xml(
        self,
        xml_path: str,
        target_phones: Dict[str, str],
        already_processed: Set[str],
        discovered_phones: Set[str],
        discovered_emails: Optional[Set[str]] = None,
    ) -> Tuple[Dict[str, List[Tuple[int, str]]], List[str]]:
        """
        Parcourt en streaming un export XML SMS/MMS (SMS Backup & Restore),
        sépare les destinataires de groupes (~ ou +) et filtre les RCS.

        :param xml_path: Chemin vers le fichier XML.
        :param target_phones: Dictionnaire numéro normalisé -> alias.
        :param already_processed: Clés 'fichier|numero' déjà extraites.
        :param discovered_phones: Collecteur global des numéros rencontrés.
        :param discovered_emails: Collecteur optionnel des e-mails rencontrés.
        :return: Tuple (messages_par_mois, liste_erreurs).
        """
        file_name: str = os.path.basename(xml_path)
        by_month: DefaultDict[str, List[Tuple[int, str]]] = defaultdict(list)
        errors: List[str] = []

        try:
            self._validate_safe_xml_header(xml_path)
            context = ET.iterparse(xml_path, events=("end",))
            for _, elem in context:
                if elem.tag not in ("sms", "mms"):
                    continue

                raw_addr: str = elem.attrib.get("address", "")
                indiv_addrs: List[str] = DocumentTextUtils.split_sms_addresses(
                    raw_addr
                )
                matched_aliases: List[str] = []

                for norm_addr in indiv_addrs:
                    if "@" in norm_addr:
                        if (
                            discovered_emails is not None
                            and not norm_addr.endswith("://google.com")
                        ):
                            discovered_emails.add(norm_addr)
                    else:
                        discovered_phones.add(norm_addr)

                    track_key: str = f"{file_name}|{norm_addr}"
                    if (
                        norm_addr in target_phones
                        and track_key not in already_processed
                    ):
                        alias_val: str = target_phones[norm_addr]
                        if alias_val not in matched_aliases:
                            matched_aliases.append(alias_val)

                if not matched_aliases:
                    elem.clear()
                    continue

                alias: str = ", ".join(matched_aliases)
                raw_ts: str = elem.attrib.get("date", "0")
                ts_ms: int = int(raw_ts) if raw_ts.isdigit() else 0
                dt = datetime.datetime.fromtimestamp(ts_ms / 1000.0)
                period: str = dt.strftime("%Y-%m")
                date_fmt: str = dt.strftime("%Y-%m-%d %H:%M:%S")

                msg_type: str = elem.attrib.get(
                    "type", elem.attrib.get("msg_box", "1")
                )
                direction: str = (
                    f"[ENVOYÉ par {self.MAIN_CHARACTER_TOKEN} à {alias}]"
                    if msg_type == "2"
                    else f"[REÇU par {self.MAIN_CHARACTER_TOKEN} de {alias}]"
                )

                body: str = ""
                if elem.tag == "sms":
                    body = elem.attrib.get("body", "").strip()
                else:
                    parts_txt: List[str] = []
                    for part in elem.iter("part"):
                        if part.attrib.get("ct") == "text/plain":
                            txt_p: str = part.attrib.get("text", "").strip()
                            if txt_p:
                                parts_txt.append(txt_p)
                    body = "\n".join(parts_txt).strip()

                if body:
                    block: str = (
                        f"[{date_fmt}] {direction}\n{body}\n"
                        "----------------------------------------\n"
                    )
                    by_month[period].append((ts_ms, block))
                elem.clear()

            return dict(by_month), errors
        except Exception as exc:
            errors.append(f"{file_name} -> Erreur parsing XML ({exc})")
            return {}, errors
    def extract_audio(
        self, file_path: str
    ) -> Tuple[Optional[str], Optional[str]]:
        """
        Transcrit un fichier audio via faster-whisper et identifie les
        participants si un numéro est présent dans le nom du fichier.

        :param file_path: Chemin vers le fichier audio.
        :return: Tuple (transcription_formatee, message_erreur).
        """
        model: Optional[WhisperModel] = self.whisper_model
        if not self.config.enable_audio or not model:
            return None, "Transcription audio désactivée"

        file_name: str = os.path.basename(file_path)
        phone_match = DocumentTextUtils.REGEX_PHONE.search(file_name)
        participants_line: str = ""
        if phone_match:
            raw_num: str = phone_match.group(0)
            participants_line = (
                f"PARTICIPANTS : {self.MAIN_CHARACTER_TOKEN} et {raw_num}\n"
            )

        try:
            segments, _ = model.transcribe(
                file_path, beam_size=5, vad_filter=True
            )
            lines: List[str] = []
            for seg in segments:
                txt: str = seg.text.strip()
                if txt:
                    start_m, start_s = divmod(int(seg.start), 60)
                    lines.append(f"[{start_m:02d}:{start_s:02d}] {txt}")

            if not lines:
                return None, "Aucune parole détectée dans l'audio"

            header: str = (
                "========================================\n"
                f"AUDIO : {file_name}\n"
                f"{participants_line}"
                "----------------------------------------\n"
            )
            return header + "\n".join(lines) + "\n\n", None
        except Exception as exc:
            return None, f"Erreur transcription audio ({exc})"

    def extract_audacity_aup3(
        self, file_path: str
    ) -> Tuple[Optional[str], Optional[str]]:
        """
        Extrait et transcrit la piste audio principale d'un projet Audacity
        (.aup3, base SQLite3) via un fichier WAV temporaire, en garantissant
        la fermeture de la base SQLite même en cas d'erreur.

        :param file_path: Chemin vers le fichier .aup3.
        :return: Tuple (transcription_formatee, message_erreur).
        """
        if not self.config.enable_audio or not self.whisper_model:
            return None, "Transcription audio désactivée"

        file_name: str = os.path.basename(file_path)
        temp_wav: str = f"{file_path}.tmp_extract.wav"
        conn: Optional[sqlite3.Connection] = None
        try:
            conn = sqlite3.connect(file_path)
            cur = conn.cursor()
            cur.execute("SELECT samples FROM sampleblocks ORDER BY blockid")
            rows: List[Any] = cur.fetchall()

            if not rows:
                return None, "Projet Audacity sans bloc audio"

            raw_bytes: bytes = b"".join(
                next(iter(r)) for r in rows if r and next(iter(r))
            )
            audio_arr = np.frombuffer(raw_bytes, dtype=np.float32)
            if audio_arr.size == 0:
                return None, "Piste audio Audacity vide"

            wavfile.write(temp_wav, 44100, audio_arr)
            res_txt, err = self.extract_audio(temp_wav)
            if res_txt:
                res_txt = res_txt.replace(
                    os.path.basename(temp_wav), file_name
                )
            return res_txt, err
        except Exception as exc:
            return None, f"Erreur lecture projet Audacity ({exc})"
        finally:
            if conn is not None:
                try:
                    conn.close()
                except sqlite3.Error:
                    pass
            if os.path.exists(temp_wav):
                try:
                    os.remove(temp_wav)
                except OSError:
                    pass

    @staticmethod
    def _normalize_token(text: str) -> str:
        """
        Normalise un terme en minuscules sans accents pour garantir la
        correspondance exacte avec les recherches normalisées.

        :param text: Chaîne brute issue de config.ini ou prenoms.txt.
        :return: Chaîne normalisée sans diacritiques.
        """
        nfkd: str = unicodedata.normalize("NFD", text.strip().lower())
        return "".join(
            ch for ch in nfkd if unicodedata.category(ch) != "Mn"
        )

    def _ensure_strong_secret_salt(self) -> None:
        """
        Génère et persiste un sel cryptographique aléatoire de 256 bits dans
        .secret_salt si config.ini utilise encore le sel par défaut.
        """
        if (
            self.secret_salt
            and self.secret_salt != self.DEFAULT_PLACEHOLDER_SALT
        ):
            return
        salt_file: str = os.path.join(self._base_dir, ".secret_salt")
        if os.path.exists(salt_file):
            try:
                with open(salt_file, "r", encoding="utf-8") as f_s:
                    stored: str = f_s.read().strip()
                if len(stored) >= 32:
                    self.secret_salt = stored
                    return
            except OSError:
                pass
        generated: str = secrets.token_hex(32)
        self.secret_salt = generated
        try:
            with open(salt_file, "w", encoding="utf-8") as f_out:
                f_out.write(f"{generated}\n")
        except OSError:
            pass

    def _resolve_path(self, raw_path: str) -> str:
        """
        Résout un chemin relativement au dossier contenant config.ini.

        :param raw_path: Chemin lu dans le fichier INI.
        :return: Chemin absolu normalisé.
        """
        cleaned: str = raw_path.strip()
        if os.path.isabs(cleaned):
            return os.path.normpath(cleaned)
        return os.path.normpath(os.path.join(self._base_dir, cleaned))

    def _parse_csv_set(self, raw_value: str) -> Set[str]:
        """
        Convertit une chaîne séparée par des virgules en ensemble nettoyé.

        :param raw_value: Chaîne brute issue du fichier INI.
        :return: Ensemble des éléments non vides.
        """
        return {item.strip() for item in raw_value.split(",") if item.strip()}

from collections import defaultdict
import os
import re
from typing import Any, DefaultDict, Dict, List, Optional, Pattern, Set, Tuple
from doc_anonymizer.config_manager import PipelineConfig
from doc_anonymizer.text_utils import DocumentTextUtils


class ContactRulesLoader:
    """
    Prépare contacts.txt, contacts_emails_decouverts.txt et contacts_manuels.txt,
    déduit les noms complets associés aux alias et compile des expressions
    régulières groupées par arbre préfixe (Trie) pour un remplacement rapide.
    """

    ACCENT_MAP: Dict[str, str] = {
        "a": "[aàâä]", "e": "[eéèêë]", "i": "[iîï]",
        "o": "[oôö]", "u": "[uùûü]", "c": "[cç]", "y": "[yÿ]",
    }

    TITLE_PREFIXES: Tuple[str, ...] = (
        "M.", "M", "Monsieur", "Mme", "Madame", "Mlle",
        "Dr", "Docteur", "Pr", "Professeur", "Me", "Maître",
    )

    def __init__(
        self, config: PipelineConfig, manual_only: bool = False
    ) -> None:
        """
        Initialise le chargeur de règles et les structures de recherche rapide.

        :param config: Instance de PipelineConfig.
        :param manual_only: Si True, charge uniquement contacts_manuels.txt.
        """
        self.config: PipelineConfig = config
        self.manual_only: bool = manual_only
        self.contacts_id: Dict[str, str] = {}
        self.compiled_rules: List[Tuple[Pattern[str], str]] = []
        self.glued_rules: List[Tuple[Pattern[str], str]] = []
        self.location_rules: List[Pattern[str]] = []
        self.combined_keyword_pattern: Optional[Pattern[str]] = None
        self.normalized_keyword_map: Dict[str, str] = {}
        self.combined_location_pattern: Optional[Pattern[str]] = None
        self.alias_words: Set[str] = set()
        self.full_targets: Set[str] = set()
        self.last_name_to_aliases: DefaultDict[str, Set[str]] = defaultdict(
            set
        )
        self.last_name_display: Dict[str, str] = {}

    def is_first_name(self, word: str) -> bool:
        """
        Vérifie si un mot figure dans le référentiel des prénoms.

        :param word: Mot brut à tester.
        :return: True si le mot est un prénom connu.
        """
        norm: str = DocumentTextUtils.normalize_word(word)
        if norm in self.config.known_first_names:
            return True
        if "-" in norm:
            parts: List[str] = [p for p in norm.split("-") if p]
            return bool(parts) and all(
                p in self.config.known_first_names for p in parts
            )
        return False

    @classmethod
    def _term_to_accent_regex(cls, term: str) -> str:
        """
        Convertit un terme normalisé en motif regex tolérant les accents.

        :param term: Terme normalisé en minuscules sans accents.
        :return: Chaîne de motif regex.
        """
        return "".join(
            cls.ACCENT_MAP.get(c, re.escape(c)) for c in term
        )

    @classmethod
    def build_regex(
        cls, term: str, whole_word: bool = True
    ) -> Pattern[str]:
        """
        Compile une regex insensible à la casse et aux accents.

        :param term: Terme cible à rechercher.
        :param whole_word: Impose des frontières de mots autour du terme.
        :return: Expression régulière compilée.
        """
        stripped: str = DocumentTextUtils.strip_accents(term).lower()
        pat: str = cls._term_to_accent_regex(stripped)
        if whole_word:
            pat = rf"(?<!\w){pat}(?!\w)"
        return re.compile(pat, re.IGNORECASE)
    @classmethod
    def _build_trie_regex(cls, terms: List[str]) -> Optional[Pattern[str]]:
        """
        Construit une expression régulière unique optimisée par arbre préfixe
        (Trie) à partir d'une liste de termes normalisés sans accents.

        :param terms: Liste de termes normalisés (minuscules, sans accents).
        :return: Pattern compilé ou None si la liste est vide.
        """
        if not terms:
            return None

        trie: Dict[str, Any] = {}
        for term in sorted(set(terms), key=len, reverse=True):
            if not term:
                continue
            node: Dict[str, Any] = trie
            for ch in term:
                node = node.setdefault(ch, {})
            node[""] = True

        def _render_node(node: Dict[str, Any]) -> str:
            if "" in node and len(node) == 1:
                return ""
            alternatives: List[str] = []
            for ch in sorted(k for k in node.keys() if k != ""):
                char_pat: str = cls.ACCENT_MAP.get(ch, re.escape(ch))
                sub_pat: str = _render_node(node.get(ch, {}))
                alternatives.append(f"{char_pat}{sub_pat}")
            has_end: bool = "" in node
            if len(alternatives) == 1:
                joined: str = next(iter(alternatives))
            else:
                joined = f"(?:{'|'.join(alternatives)})"
            return f"(?:{joined})?" if has_end else joined

        body: str = _render_node(trie)
        if not body:
            return None
        return re.compile(rf"(?<!\w){body}(?!\w)", re.IGNORECASE)

    def lookup_keyword_alias(self, matched_text: str) -> str:
        """
        Retrouve l'alias associé à une correspondance trouvée par la regex
        groupée en normalisant les accents et la casse.

        :param matched_text: Segment exact capturé dans le document.
        :return: Alias de remplacement correspondant.
        """
        norm_match: str = DocumentTextUtils.strip_accents(matched_text).lower()
        return self.normalized_keyword_map.get(norm_match, matched_text)

    def _add_name_variants(
        self, full_name: str, alias: str, keywords: Dict[str, str]
    ) -> None:
        """
        Ajoute Prénom Nom, Nom Prénom, Titre Nom, les variantes collées
        (NomPrénom / PrénomNom) et indexe le nom de famille pour l'audit flou.

        :param full_name: Nom complet de la personne.
        :param alias: Alias anonymisé cible.
        :param keywords: Dictionnaire cumulatif terme -> alias.
        """
        keywords[full_name] = alias
        self.full_targets.add(DocumentTextUtils.normalize_word(full_name))
        tokens: List[str] = full_name.split()
        if len(tokens) != 2:
            return
        tok1, tok2 = tokens
        first_is_fn: bool = self.is_first_name(tok1)
        second_is_fn: bool = self.is_first_name(tok2)
        if not (first_is_fn or second_is_fn):
            return

        inv: str = f"{tok2} {tok1}"
        keywords.setdefault(inv, alias)
        self.full_targets.add(DocumentTextUtils.normalize_word(inv))

        first_tok: str = tok1 if first_is_fn else tok2
        last_tok: str = tok2 if first_is_fn else tok1
        norm_first: str = DocumentTextUtils.normalize_word(first_tok)
        norm_last: str = DocumentTextUtils.normalize_word(last_tok)

        if (
            len(norm_last) >= 3
            and not self.is_first_name(last_tok)
            and norm_last not in self.config.excluded_words
            and norm_last not in self.config.masked_locations
        ):
            self.last_name_to_aliases[norm_last].add(alias)
            self.last_name_display.setdefault(norm_last, last_tok)
            for prefix in self.TITLE_PREFIXES:
                keywords.setdefault(f"{prefix} {last_tok}", alias)

            if len(norm_first) >= 4 and len(norm_last) >= 4:
                for glued in (
                    f"{first_tok}{last_tok}",
                    f"{last_tok}{first_tok}",
                ):
                    stripped_g: str = DocumentTextUtils.strip_accents(
                        glued
                    ).lower()
                    pat_g: str = self._term_to_accent_regex(stripped_g)
                    self.glued_rules.append(
                        (
                            re.compile(
                                rf"(?<!\w){pat_g}(?=[a-zàâäéèêëîïôöùûüç]|\b)",
                                re.IGNORECASE,
                            ),
                            alias,
                        )
                    )
    @staticmethod
    def _rule_entry_sort_key(item: Tuple[str, str]) -> Tuple[str, str, str]:
        """
        Calcule la clé de tri alphabétique d'une paire (cible, alias).

        :param item: Tuple (cible_a_masquer, alias).
        :return: Tuple de comparaison normalisé pour sorted().
        """
        left_col, right_col = item
        return (
            DocumentTextUtils.normalize_word(left_col),
            left_col.lower(),
            right_col.lower(),
        )

    def _ensure_manual_contacts_file(self) -> None:
        """
        Crée le fichier contacts_manuels.txt avec un en-tête explicatif
        s'il n'existe pas encore sur le disque.
        """
        manual_path: str = self.config.manual_contacts_file
        if os.path.exists(manual_path):
            return
        header: str = (
            "# RÈGLES D'ANONYMISATION MANUELLES\n"
            "# Ce fichier n'est jamais écrasé par lancer_contacts.py.\n"
            "# Ajoutez ici les noms issus de audit_noms_residuels.txt :\n"
            "# Format : Prénom Nom = Alias\n\n"
        )
        DocumentTextUtils.atomic_write_text(manual_path, header)

    def _parse_and_sort_rules_file(
        self, file_path: str, keywords: Dict[str, str]
    ) -> Tuple[int, int]:
        """
        Lit, dédoublonne, trie et réécrit atomiquement un fichier de règles
        au format canonique 'Cible = Alias', puis enregistre les cibles et alias.

        :param file_path: Chemin vers le fichier de règles à charger.
        :param keywords: Dictionnaire cumulatif terme -> alias.
        :return: Tuple (nombre_règles_uniques, nombre_doublons_supprimés).
        """
        if not os.path.exists(file_path):
            return 0, 0

        comments: List[str] = []
        unique_lines: Dict[str, Tuple[str, str]] = {}
        line_count: int = 0

        with open(file_path, "r", encoding="utf-8") as f:
            for raw in f:
                line: str = raw.strip()
                if not line:
                    continue
                if line.startswith("#"):
                    if line not in comments:
                        comments.append(line)
                    continue
                line_count += 1
                for sep in ("=", ";", ","):
                    if sep not in line:
                        continue
                    left, right = line.split(sep, 1)
                    target, alias, _ = (
                        DocumentTextUtils.resolve_contact_pair(left, right)
                    )
                    if target and alias:
                        norm_t: str = DocumentTextUtils.normalize_word(target)
                        unique_lines[norm_t] = (target, alias)
                        self.full_targets.add(norm_t)
                        if DocumentTextUtils.is_contact_identifier(target):
                            self.contacts_id[
                                DocumentTextUtils.normalize_identifier(target)
                            ] = alias
                        else:
                            self._add_name_variants(target, alias, keywords)
                        for w in re.split(r"[\s_'-]+", alias):
                            if w:
                                self.alias_words.add(
                                    DocumentTextUtils.normalize_word(w)
                                )
                    break

        sorted_entries: List[Tuple[str, str]] = sorted(
            unique_lines.values(),
            key=self._rule_entry_sort_key,
        )
        out_lines: List[str] = [f"{com}\n" for com in comments]
        if comments:
            out_lines.append("\n")
        for left_col, right_col in sorted_entries:
            out_lines.append(f"{left_col} = {right_col}\n")

        DocumentTextUtils.atomic_write_text(file_path, "".join(out_lines))
        return len(sorted_entries), line_count - len(sorted_entries)
    def _load_paired_names(self, keywords: Dict[str, str]) -> int:
        """
        Croise contacts_indexes.txt et contacts_anonymises.txt.

        :param keywords: Dictionnaire cumulatif terme -> alias.
        :return: Nombre de nouveaux noms complets ajoutés.
        """
        idx_path: str = self.config.indexed_contacts_file
        ano_path: str = self.config.anonymized_contacts_file
        if not (os.path.exists(idx_path) and os.path.exists(ano_path)):
            return 0

        coord_to_full: Dict[str, str] = {}
        with open(idx_path, "r", encoding="utf-8") as f_idx:
            for line in f_idx:
                s: str = line.strip()
                if s and not s.startswith("#") and "=" in s:
                    left, right = s.split("=", 1)
                    coord_to_full[
                        DocumentTextUtils.normalize_identifier(left)
                    ] = right.strip()

        added: int = 0
        with open(ano_path, "r", encoding="utf-8") as f_ano:
            for line in f_ano:
                s = line.strip()
                if not s or s.startswith("#") or "=" not in s:
                    continue
                left, right = s.split("=", 1)
                coord: str = DocumentTextUtils.normalize_identifier(left)
                full_raw: str = coord_to_full.get(coord, "")
                if not full_raw:
                    continue
                clean_full: str = re.sub(r"\s*\([^)]*\)", "", full_raw).strip()
                clean_alias: str = re.sub(r"\s*\([^)]*\)", "", right).strip()
                if (
                    clean_full
                    and clean_alias
                    and DocumentTextUtils.normalize_word(clean_full)
                    != DocumentTextUtils.normalize_word(clean_alias)
                ):
                    if clean_full not in keywords:
                        added += 1
                    self._add_name_variants(clean_full, clean_alias, keywords)
                    for w in re.split(r"[\s_'-]+", clean_alias):
                        if w:
                            self.alias_words.add(
                                DocumentTextUtils.normalize_word(w)
                            )
        return added

    def load_all(self) -> bool:
        """
        Charge les règles de contacts (toutes sources ou uniquement
        contacts_manuels.txt si manual_only est actif) et compile le Trie.

        :return: True si le chargement a réussi, False sinon.
        """
        self.contacts_id.clear()
        self.alias_words.clear()
        self.full_targets.clear()
        self.glued_rules.clear()
        self.last_name_to_aliases.clear()
        self.last_name_display.clear()

        self._ensure_manual_contacts_file()
        for tok in re.split(r"[\s_'-]+", self.config.main_character_name):
            if tok:
                self.alias_words.add(DocumentTextUtils.normalize_word(tok))

        keywords: Dict[str, str] = {}

        if self.manual_only:
            print(
                f"1. Lecture EXCLUSIVE de '{self.config.manual_contacts_file}'...",
                flush=True,
            )
            man_count, man_dup = self._parse_and_sort_rules_file(
                self.config.manual_contacts_file, keywords
            )
            auto_count = auto_dup = disc_count = paired_added = 0
        else:
            contacts_path: str = self.config.contacts_file
            if not os.path.exists(contacts_path):
                print(f"Erreur : '{contacts_path}' introuvable.", flush=True)
                return False

            print(
                "1. Lecture des règles de contacts (auto, découverts, manuels)...",
                flush=True,
            )
            auto_count, auto_dup = self._parse_and_sort_rules_file(
                contacts_path, keywords
            )
            disc_count, _ = self._parse_and_sort_rules_file(
                self.config.discovered_contacts_file, keywords
            )
            man_count, man_dup = self._parse_and_sort_rules_file(
                self.config.manual_contacts_file, keywords
            )
            paired_added = self._load_paired_names(keywords)

        self.normalized_keyword_map = {}
        for term in sorted(keywords.keys(), key=len, reverse=True):
            norm_k: str = DocumentTextUtils.strip_accents(term).lower()
            if norm_k:
                self.normalized_keyword_map.setdefault(norm_k, keywords[term])

        self.combined_keyword_pattern = self._build_trie_regex(
            list(self.normalized_keyword_map.keys())
        )
        self.compiled_rules = [
            (self.build_regex(term), keywords[term])
            for term in sorted(keywords.keys(), key=len, reverse=True)
        ]

        norm_locs: List[str] = sorted(
            {
                DocumentTextUtils.strip_accents(loc).lower()
                for loc in self.config.masked_locations
                if len(loc.strip()) >= 3
            },
            key=len,
            reverse=True,
        )
        self.combined_location_pattern = self._build_trie_regex(norm_locs)
        self.location_rules = [
            self.build_regex(loc, whole_word=True) for loc in norm_locs
        ]

        print(
            f"   -> {auto_count} auto ({auto_dup} doublon(s)), "
            f"{disc_count} découvert(s), "
            f"{man_count} manuel(s) ({man_dup} doublon(s)), "
            f"{paired_added} nom(s) complet(s), "
            f"{len(self.normalized_keyword_map)} motif(s) en Trie.",
            flush=True,
        )
        return True

import os
import re
from typing import Dict, List, Optional, Set, Tuple
from doc_anonymizer.config_manager import PipelineConfig
from doc_anonymizer.text_utils import DocumentTextUtils


class DiscoveredEmailContactsBuilder:
    """
    Génère et enrichit le fichier contacts_emails_decouverts.txt à partir
    de tous_emails_trouves.txt pour compléter contacts.txt et
    contacts_manuels.txt sans doublon, puis purge tous_emails_trouves.txt.
    """

    def __init__(self, config: PipelineConfig) -> None:
        """
        Initialise le générateur de contacts découverts.

        :param config: Instance de PipelineConfig.
        """
        self.config: PipelineConfig = config

    def _collect_existing_targets(self) -> Set[str]:
        """
        Lit contacts.txt et contacts_manuels.txt afin de recenser toutes
        les cibles (e-mails ou noms) déjà couvertes.

        :return: Ensemble des cibles normalisées déjà définies.
        """
        covered: Set[str] = set()
        for path in (
            self.config.contacts_file,
            self.config.manual_contacts_file,
        ):
            if not os.path.exists(path):
                continue
            with open(path, "r", encoding="utf-8") as f:
                for raw_line in f:
                    line: str = raw_line.strip()
                    if not line or line.startswith("#"):
                        continue
                    for sep in ("=", ";", ","):
                        if sep in line:
                            left_p, right_p = line.split(sep, 1)
                            target, alias, _ = (
                                DocumentTextUtils.resolve_contact_pair(
                                    left_p, right_p
                                )
                            )
                            if target and alias:
                                covered.add(
                                    DocumentTextUtils.normalize_identifier(
                                        target
                                    )
                                )
                                covered.add(
                                    DocumentTextUtils.normalize_word(target)
                                )
                            break
        return covered

    def _load_existing_discovered(self) -> Dict[str, str]:
        """
        Charge les règles déjà générées dans contacts_emails_decouverts.txt
        afin de préserver les alias existants entre deux exécutions.

        :return: Dictionnaire cible -> alias existant.
        """
        existing: Dict[str, str] = {}
        disc_path: str = self.config.discovered_contacts_file
        if not os.path.exists(disc_path):
            return existing
        with open(disc_path, "r", encoding="utf-8") as f:
            for raw_line in f:
                line: str = raw_line.strip()
                if not line or line.startswith("#") or "=" not in line:
                    continue
                left_p, right_p = line.split("=", 1)
                target, alias, _ = DocumentTextUtils.resolve_contact_pair(
                    left_p, right_p
                )
                if target and alias:
                    existing[target] = alias
        return existing

    def _infer_full_name_from_email(self, email: str) -> Optional[str]:
        """
        Déduit 'Prénom Nom' si l'adresse e-mail est au format
        'prenom.nom@domaine' ou 'nom.prenom@domaine' avec un prénom connu.

        :param email: Adresse e-mail normalisée.
        :return: Chaîne 'Prénom Nom' capitalisée ou None.
        """
        if "@" not in email:
            return None
        local_part, _ = email.split("@", 1)
        tokens: List[str] = [
            t for t in re.split(r"[\.\-_]+", local_part.strip()) if t
        ]
        if len(tokens) != 2:
            return None
        first_tok, second_tok = tokens
        if any(ch.isdigit() for ch in first_tok + second_tok):
            return None
        if len(first_tok) < 2 or len(second_tok) < 2:
            return None

        norm_1: str = DocumentTextUtils.normalize_word(first_tok)
        norm_2: str = DocumentTextUtils.normalize_word(second_tok)
        is_fn_1: bool = norm_1 in self.config.known_first_names
        is_fn_2: bool = norm_2 in self.config.known_first_names

        if is_fn_1 and not is_fn_2:
            return f"{first_tok.capitalize()} {second_tok.capitalize()}"
        if is_fn_2 and not is_fn_1:
            return f"{second_tok.capitalize()} {first_tok.capitalize()}"
        return None
    def _next_alias_index(self, existing_rules: Dict[str, str]) -> int:
        """
        Détermine le prochain numéro disponible pour les alias Contact_Mail_XXX.

        :param existing_rules: Dictionnaire des règles déjà présentes.
        :return: Prochain index entier >= 1.
        """
        max_idx: int = 0
        for alias in existing_rules.values():
            match = re.match(r"^Contact_Mail_(\d+)$", alias.strip())
            if match:
                max_idx = max(max_idx, int(match.group(1)))
        return max_idx + 1

    @staticmethod
    def _sort_discovered_item(item: Tuple[str, str]) -> Tuple[str, str]:
        """
        Calcule la clé de tri alphabétique pour une entrée découverte.

        :param item: Tuple (cible, alias).
        :return: Tuple normalisé pour le tri.
        """
        target_key, _ = item
        return (
            DocumentTextUtils.normalize_word(target_key),
            target_key.lower(),
        )

    def build_or_update(self) -> Tuple[int, int]:
        """
        Lit tous_emails_trouves.txt, complète contacts_emails_decouverts.txt
        avec les adresses et noms déduits absents des deux autres fichiers,
        puis purge tous_emails_trouves.txt de façon atomique.

        :return: Tuple (nouvelles_regles_ajoutees, total_regles_fichier).
        """
        emails_path: str = self.config.all_emails_file
        if not os.path.exists(emails_path):
            return 0, 0

        covered_targets: Set[str] = self._collect_existing_targets()
        discovered_rules: Dict[str, str] = self._load_existing_discovered()
        existing_norm_keys: Set[str] = {
            DocumentTextUtils.normalize_identifier(k)
            for k in discovered_rules.keys()
        } | {
            DocumentTextUtils.normalize_word(k)
            for k in discovered_rules.keys()
        }

        next_idx: int = self._next_alias_index(discovered_rules)
        added_count: int = 0

        with open(emails_path, "r", encoding="utf-8") as f_mail:
            raw_emails: List[str] = [
                DocumentTextUtils.normalize_identifier(ln)
                for ln in f_mail
                if "@" in ln
            ]

        for email in sorted(set(raw_emails)):
            if (
                not email
                or email in covered_targets
                or email in existing_norm_keys
            ):
                continue

            alias: str = f"Contact_Mail_{next_idx:03d}"
            next_idx += 1

            discovered_rules[email] = alias
            existing_norm_keys.add(email)
            added_count += 1

            inferred_name: Optional[str] = self._infer_full_name_from_email(
                email
            )
            if inferred_name:
                norm_name: str = DocumentTextUtils.normalize_word(
                    inferred_name
                )
                if (
                    norm_name not in covered_targets
                    and norm_name not in existing_norm_keys
                ):
                    discovered_rules[inferred_name] = alias
                    existing_norm_keys.add(norm_name)
                    added_count += 1

        if added_count > 0 or not os.path.exists(
            self.config.discovered_contacts_file
        ):
            out_path: str = self.config.discovered_contacts_file
            sorted_items: List[Tuple[str, str]] = sorted(
                discovered_rules.items(),
                key=self._sort_discovered_item,
            )
            out_lines: List[str] = [
                "# CONTACTS DÉCOUVERTS AUTOMATIQUEMENT DEPUIS "
                "tous_emails_trouves.txt\n",
                "# Complète contacts.txt et contacts_manuels.txt.\n\n",
            ]
            for target_k, alias_v in sorted_items:
                out_lines.append(f"{target_k} = {alias_v}\n")
            DocumentTextUtils.atomic_write_text(out_path, "".join(out_lines))

        DocumentTextUtils.atomic_write_text(emails_path, "")

        return added_count, len(discovered_rules)

import os
import re
import time
from typing import Dict, List, Optional, Set, Tuple
from doc_anonymizer.comm_extractors import CommunicationExtractor
from doc_anonymizer.config_manager import PipelineConfig
from doc_anonymizer.office_extractors import OfficeAndDocumentExtractor
from doc_anonymizer.text_utils import DocumentTextUtils


class DocumentExtractionRunner:
    """
    Orchestre l'étape 1 : inventaire des fichiers dans le répertoire
    sources configuré, extraction multi-formats brute et sauvegarde
    non anonymisée dans le dossier de transcriptions.
    """

    AUDIO_EXTS: Tuple[str, ...] = (
        ".wav", ".mp3", ".m4a", ".ogg", ".flac", ".aac", ".amr",
    )
    IMAGE_EXTS: Tuple[str, ...] = (
        ".jpg", ".jpeg", ".png", ".bmp", ".webp", ".tif", ".tiff",
    )
    HTML_EXTS: Tuple[str, ...] = (".html", ".htm")

    def __init__(self, config: PipelineConfig) -> None:
        """
        Initialise l'orchestrateur d'extraction.

        :param config: Instance de PipelineConfig.
        """
        self.config: PipelineConfig = config
        self.doc_extractor: OfficeAndDocumentExtractor = (
            OfficeAndDocumentExtractor(enable_ocr=config.enable_ocr)
        )
        self.comm_extractor: CommunicationExtractor = CommunicationExtractor(
            config=config, doc_extractor=self.doc_extractor
        )
        self.extraction_errors: List[str] = []

    @staticmethod
    def load_tracking_set(tracking_path: str) -> Set[str]:
        """
        Charge les identifiants déjà extraits depuis un fichier de suivi.

        :param tracking_path: Chemin vers le fichier de suivi.
        :return: Ensemble des clés enregistrées.
        """
        if not os.path.exists(tracking_path):
            return set()
        with open(tracking_path, "r", encoding="utf-8") as f:
            return {line.strip() for line in f if line.strip()}

    @staticmethod
    def append_tracking_entry(tracking_path: str, entry: str) -> None:
        """
        Ajoute une entrée dans le fichier de suivi.

        :param tracking_path: Chemin vers le fichier de suivi.
        :param entry: Identifiant unique traité.
        """
        parent_dir: str = os.path.dirname(tracking_path)
        if parent_dir:
            os.makedirs(parent_dir, exist_ok=True)
        with open(tracking_path, "a", encoding="utf-8") as f:
            f.write(f"{entry}\n")

    @classmethod
    def categorize_extension(cls, ext: str) -> str:
        """
        Détermine le sous-dossier thématique associé à l'extension.

        :param ext: Extension en minuscules avec point initial.
        :return: Nom de la catégorie.
        """
        if ext == ".pdf":
            return "PDF"
        if ext in cls.AUDIO_EXTS or ext == ".aup3":
            return "Audio"
        if ext in cls.IMAGE_EXTS:
            return "Images"
        if ext in cls.HTML_EXTS:
            return "Web"
        if ext in {".mbox", ".eml"}:
            return "Emails"
        if ext == ".xml":
            return "SMS"
        if ext == ".ics":
            return "Agendas"
        return "Documents"
    @staticmethod
    def _sanitize_filesystem_stem(stem: str) -> str:
        """
        Nettoie uniquement les caractères interdits par le système de fichiers
        sans appliquer aucune anonymisation au nom du fichier brut.

        :param stem: Nom de fichier sans extension.
        :return: Nom de fichier valide sur disque, non anonymisé.
        """
        cleaned: str = re.sub(r'[<>:"/\\|?*]', "_", stem).strip()
        return cleaned or "document_sans_nom"

    def _resolve_input_dir(self) -> str:
        """
        Résout le chemin complet du répertoire contenant les fichiers
        sources à traiter (ex: './fichiers sources').

        :return: Chemin normalisé du répertoire d'entrée.
        """
        if os.path.isabs(self.config.sources_dir):
            return os.path.normpath(self.config.sources_dir)
        return os.path.normpath(
            os.path.join(self.config.root_dir, self.config.sources_dir)
        )

    def _file_priority_key(self, file_path: str) -> Tuple[int, int, str]:
        """
        Calcule une clé de priorité pour traiter rapidement les fichiers
        légers et reporter les archives .mbox, audios et gros fichiers à la fin.

        :param file_path: Chemin complet du fichier candidat.
        :return: Tuple (groupe_priorite, taille_octets, chemin_normalise).
        """
        if not self.config.defer_heavy_files:
            return (0, 0, file_path)

        _, ext_raw = os.path.splitext(file_path)
        ext: str = ext_raw.lower()
        try:
            size_bytes: int = os.path.getsize(file_path)
        except OSError:
            size_bytes = 0

        threshold_bytes: int = int(
            self.config.heavy_file_threshold_mb * 1024 * 1024
        )
        is_heavy_type: bool = (
            ext in {".mbox", ".aup3"} or ext in self.AUDIO_EXTS
        )
        is_large: bool = size_bytes >= threshold_bytes

        if is_heavy_type or is_large:
            return (1, size_bytes, file_path)
        return (0, 0, file_path)

    def collect_source_files(self) -> List[str]:
        """
        Parcourt le répertoire d'entrée configuré (sources_dir) et ordonne
        les fichiers en reportant les fichiers lourds/mbox/audio à la fin.

        :return: Liste ordonnée des chemins de fichiers à analyser.
        """
        input_dir: str = self._resolve_input_dir()
        if not os.path.isdir(input_dir):
            os.makedirs(input_dir, exist_ok=True)
            print(
                f"[AVERTISSEMENT] Dossier '{input_dir}' créé : "
                "placez-y vos documents avant de relancer.",
                flush=True,
            )
            return []

        collected: List[str] = []
        excluded_files: Set[str] = {
            os.path.basename(self.config.contacts_file),
            os.path.basename(self.config.manual_contacts_file),
            os.path.basename(self.config.discovered_contacts_file),
            os.path.basename(self.config.all_sms_numbers_file),
            os.path.basename(self.config.all_emails_file),
            os.path.basename(self.config.audit_report_file),
            os.path.basename(self.config.audit_copy_file),
        }
        for root_path, dirs, files in os.walk(input_dir):
            dirs[:] = sorted(
                d
                for d in dirs
                if d not in self.config.ignored_dirs and not d.startswith(".")
            )
            for file_name in sorted(files):
                if (
                    file_name.startswith(".")
                    or file_name in excluded_files
                    or file_name.endswith(".py")
                    or file_name.endswith(".ini")
                ):
                    continue
                collected.append(os.path.join(root_path, file_name))

        collected.sort(key=self._file_priority_key)
        return collected
    def _write_transcription_block(
        self,
        period: str,
        category: str,
        raw_base_name: str,
        content: str,
    ) -> str:
        """
        Écrit le contenu extrait brut dans transcriptions/AAAA-MM/Catégorie/.

        :param period: Période AAAA-MM.
        :param category: Catégorie thématique du document.
        :param raw_base_name: Nom de fichier d'origine non anonymisé.
        :param content: Texte extrait prêt à enregistrer.
        :return: Chemin du fichier transcription généré.
        """
        target_dir: str = os.path.join(
            self.config.transcriptions_dir, period, category
        )
        os.makedirs(target_dir, exist_ok=True)
        out_path: str = os.path.join(target_dir, f"{raw_base_name}.txt")
        with open(out_path, "a", encoding="utf-8") as f_out:
            f_out.write(content)
        return out_path

    def process_single_file(
        self,
        file_path: str,
        target_phones: Dict[str, str],
        allowed_emails: Set[str],
        already_extracted: Set[str],
        sms_processed: Set[str],
        discovered_phones: Set[str],
        discovered_emails: Set[str],
    ) -> List[str]:
        """
        Traite un fichier source et retourne la liste des transcriptions
        brutes créées ou mises à jour.

        :param file_path: Chemin vers le fichier candidat.
        :param target_phones: Téléphones issus des fichiers de contacts.
        :param allowed_emails: E-mails issus des fichiers de contacts.
        :param already_extracted: Fichiers déjà traités.
        :param sms_processed: Clés SMS déjà extraites.
        :param discovered_phones: Collecteur des numéros rencontrés.
        :param discovered_emails: Collecteur des e-mails rencontrés.
        :return: Liste des chemins de transcriptions générées.
        """
        input_dir: str = self._resolve_input_dir()
        norm_rel: str = os.path.normpath(
            os.path.relpath(file_path, input_dir)
        )
        file_name: str = os.path.basename(file_path)
        stem, ext_raw = os.path.splitext(file_name)
        ext: str = ext_raw.lower()
        raw_stem: str = self._sanitize_filesystem_stem(stem)
        category: str = self.categorize_extension(ext)
        generated_paths: List[str] = []

        if ext == ".xml":
            local_phones: Set[str] = set()
            sms_by_month, xml_errors = self.comm_extractor.parse_sms_xml(
                xml_path=file_path,
                target_phones=target_phones,
                already_processed=sms_processed,
                discovered_phones=local_phones,
                discovered_emails=discovered_emails,
            )
            discovered_phones.update(local_phones)
            self.extraction_errors.extend(xml_errors)
            if sms_by_month:
                for period, entries in sorted(sms_by_month.items()):
                    entries.sort(key=lambda item: item)
                    merged_sms: str = "".join(block for _, block in entries)
                    out_p: str = self._write_transcription_block(
                        period, category, raw_stem, merged_sms
                    )
                    generated_paths.append(out_p)
                for num in local_phones:
                    if num in target_phones:
                        key: str = f"{file_name}|{num}"
                        if key not in sms_processed:
                            self.append_tracking_entry(
                                self.config.sms_tracking_file, key
                            )
                            sms_processed.add(key)
            return generated_paths

        if norm_rel in already_extracted:
            return []

        if ext in {".mbox", ".eml"}:
            emails_by_month, err = self.comm_extractor.extract_emails(
                file_path=file_path,
                allowed_emails=allowed_emails,
                discovered_emails=discovered_emails,
            )
            if err:
                self.extraction_errors.append(f"{norm_rel} -> {err}")
                return []
            if emails_by_month:
                for period, blocks in sorted(emails_by_month.items()):
                    out_p = self._write_transcription_block(
                        period, category, raw_stem, "".join(blocks)
                    )
                    generated_paths.append(out_p)
                self.append_tracking_entry(
                    self.config.extraction_tracking_file, norm_rel
                )
                already_extracted.add(norm_rel)
            return generated_paths
        extracted_text: Optional[str] = None
        err_msg: Optional[str] = None

        if ext == ".pdf":
            extracted_text, err_msg = self.doc_extractor.extract_pdf(
                file_path, file_name
            )
        elif ext in {".txt", ".md"}:
            extracted_text, err_msg = (
                self.doc_extractor.extract_text_or_markdown(file_path)
            )
        elif ext in self.HTML_EXTS:
            extracted_text, err_msg = self.doc_extractor.extract_html(
                file_path
            )
        elif ext == ".rtf":
            extracted_text, err_msg = self.doc_extractor.extract_rtf(
                file_path
            )
        elif ext == ".docx":
            extracted_text, err_msg = self.doc_extractor.extract_docx(
                file_path
            )
        elif ext == ".odt":
            extracted_text, err_msg = self.doc_extractor.extract_odt(
                file_path
            )
        elif ext == ".xlsx":
            extracted_text, err_msg = self.doc_extractor.extract_xlsx(
                file_path
            )
        elif ext == ".pptx":
            extracted_text, err_msg = self.doc_extractor.extract_pptx(
                file_path
            )
        elif ext in self.IMAGE_EXTS:
            extracted_text, err_msg = self.doc_extractor.extract_image(
                file_path
            )
        elif ext == ".ics":
            extracted_text, err_msg = self.doc_extractor.extract_ics(file_path)
        elif ext in self.AUDIO_EXTS:
            extracted_text, err_msg = self.comm_extractor.extract_audio(
                file_path
            )
        elif ext == ".aup3":
            extracted_text, err_msg = (
                self.comm_extractor.extract_audacity_aup3(file_path)
            )
        else:
            return []

        if err_msg or not extracted_text:
            if err_msg:
                self.extraction_errors.append(f"{norm_rel} -> {err_msg}")
            return []

        period: str = DocumentTextUtils.resolve_file_period(
            extracted_text, file_name, file_path
        )
        out_p = self._write_transcription_block(
            period, category, raw_stem, extracted_text
        )
        self.append_tracking_entry(
            self.config.extraction_tracking_file, norm_rel
        )
        already_extracted.add(norm_rel)
        return [out_p]

    def load_all_filter_contacts(
        self,
    ) -> Tuple[Optional[Dict[str, str]], Optional[Set[str]]]:
        """
        Fusionne les téléphones et e-mails des trois fichiers de contacts.

        :return: Tuple (dictionnaire_telephones, ensemble_emails) ou (None, None).
        """
        target_phones, allowed_emails = (
            DocumentTextUtils.load_filter_contacts(self.config.contacts_file)
        )
        if target_phones is None or allowed_emails is None:
            return None, None
        for extra_path in (
            self.config.discovered_contacts_file,
            self.config.manual_contacts_file,
        ):
            if os.path.exists(extra_path):
                ex_phones, ex_emails = DocumentTextUtils.load_filter_contacts(
                    extra_path
                )
                if ex_phones:
                    target_phones.update(ex_phones)
                if ex_emails:
                    allowed_emails.update(ex_emails)
        return target_phones, allowed_emails

    def finalize_reports(
        self,
        discovered_phones: Set[str],
        discovered_emails: Set[str],
    ) -> None:
        """
        Cumule et éclate les numéros de groupe MMS, purge les adresses RCS
        (@://google.com) et écrit les rapports d'identifiants et d'erreurs.

        :param discovered_phones: Numéros rencontrés pendant l'extraction.
        :param discovered_emails: E-mails rencontrés pendant l'extraction.
        """
        existing_sms_entries: Set[str] = self.load_tracking_set(
            self.config.all_sms_numbers_file
        )
        expanded_entries: Set[str] = set()
        for raw_entry in existing_sms_entries | discovered_phones:
            expanded_entries.update(
                DocumentTextUtils.split_sms_addresses(raw_entry)
            )

        clean_phones: Set[str] = {
            item for item in expanded_entries if "@" not in item
        }
        migrated_emails: Set[str] = {
            item
            for item in expanded_entries
            if "@" in item and not item.lower().endswith("://google.com")
        }

        if clean_phones or os.path.exists(self.config.all_sms_numbers_file):
            with open(
                self.config.all_sms_numbers_file, "w", encoding="utf-8"
            ) as f_num:
                for num in sorted(clean_phones):
                    f_num.write(f"{num}\n")

        merged_emails: Set[str] = (
            self.load_tracking_set(self.config.all_emails_file)
            | discovered_emails
            | migrated_emails
        )
        if merged_emails:
            with open(
                self.config.all_emails_file, "w", encoding="utf-8"
            ) as f_mail:
                for mail in sorted(merged_emails):
                    f_mail.write(f"{mail}\n")

        if self.extraction_errors:
            report_path: str = os.path.join(
                self.config.transcriptions_dir, "rapport_extraction.txt"
            )
            with open(report_path, "w", encoding="utf-8") as f_err:
                for err in self.extraction_errors:
                    f_err.write(f"{err}\n")

    def run(self) -> None:
        """
        Exécute la phase autonome d'extraction multi-formats.
        """
        start_time: float = time.time()
        print("=== ÉTAPE 1 : EXTRACTION MULTI-FORMATS ===", flush=True)

        target_phones, allowed_emails = self.load_all_filter_contacts()
        if target_phones is None or allowed_emails is None:
            return

        os.makedirs(self.config.transcriptions_dir, exist_ok=True)
        already_extracted: Set[str] = self.load_tracking_set(
            self.config.extraction_tracking_file
        )
        sms_processed: Set[str] = self.load_tracking_set(
            self.config.sms_tracking_file
        )
        discovered_phones: Set[str] = set()
        discovered_emails: Set[str] = set()

        files: List[str] = self.collect_source_files()
        print(f"Inventaire : {len(files)} fichier(s) candidat(s).", flush=True)

        processed_count: int = 0
        for idx, file_path in enumerate(files, start=1):
            print(f"[{idx}/{len(files)}] Analyse : {file_path}", flush=True)
            created: List[str] = self.process_single_file(
                file_path=file_path,
                target_phones=target_phones,
                allowed_emails=allowed_emails,
                already_extracted=already_extracted,
                sms_processed=sms_processed,
                discovered_phones=discovered_phones,
                discovered_emails=discovered_emails,
            )
            if created:
                processed_count += 1

        self.finalize_reports(discovered_phones, discovered_emails)
        elapsed: float = time.time() - start_time
        print(
            f"\nExtraction terminée : {processed_count} fichier(s)"
            f" en {elapsed:.1f}s.",
            flush=True,
        )

class OfficeAndDocumentExtractor:
    """
    Gère l'extraction de texte pour les formats PDF (natif + OCR paresseux),
    bureautiques (.docx, .odt, .rtf, .xlsx, .pptx), web (.html), texte brut,
    images et calendriers (.ics).
    """

    def __init__(
        self,
        config: Optional[PipelineConfig] = None,
        enable_ocr: bool = True,
    ) -> None:
        """
        Initialise l'extracteur sans charger immédiatement RapidOCR en RAM.

        :param config: Instance optionnelle de PipelineConfig.
        :param enable_ocr: Active ou désactive l'OCR si config n'est pas fourni.
        """
        self.config: Optional[PipelineConfig] = config
        self.enable_ocr: bool = (
            config.enable_ocr if config is not None else enable_ocr
        )
        self._ocr_engine: Optional[RapidOCR] = None

    @property
    def ocr_engine(self) -> Optional[RapidOCR]:
        """
        Charge le moteur RapidOCR uniquement à la première image ou page PDF
        scannée rencontrée.

        :return: Instance de RapidOCR ou None si l'OCR est désactivé.
        """
        if not self.enable_ocr:
            return None
        if self._ocr_engine is None:
            print("Chargement du moteur OCR (RapidOCR)...", flush=True)
            self._ocr_engine = RapidOCR()
            print("[OK] Moteur OCR chargé.", flush=True)
        return self._ocr_engine

    def run_ocr_on_bytes(self, image_bytes: bytes) -> str:
        """
        Exécute la reconnaissance optique de caractères sur un tampon binaire.

        :param image_bytes: Données binaires de l'image (PNG/JPEG).
        :return: Texte reconnu ligne par ligne.
        """
        engine: Optional[RapidOCR] = self.ocr_engine
        if not engine:
            return ""
        result, _ = engine(image_bytes)
        if not result:
            return ""
        return "\n".join(item[1] for item in result if len(item) > 1)
    def extract_pdf(
        self,
        pdf_source: Any,
        display_name: str,
        is_stream: bool = False,
    ) -> Tuple[Optional[str], Optional[str]]:
        """
        Extrait le texte d'un document PDF page par page, avec bascule
        automatique vers RapidOCR si une page est scannée, en garantissant
        la fermeture du document même en cas d'erreur.

        :param pdf_source: Chemin fichier (str) ou flux binaire (bytes).
        :param display_name: Nom affiché dans l'en-tête.
        :param is_stream: Indique si pdf_source est un flux bytes.
        :return: Tuple (texte_formate, message_erreur).
        """
        doc: Optional[Any] = None
        try:
            doc = (
                pymupdf.open(stream=pdf_source, filetype="pdf")
                if is_stream
                else pymupdf.open(pdf_source)
            )
            total_pages: int = len(doc)
            pages_text: List[str] = []
            for page_idx in range(total_pages):
                page = doc[page_idx]
                native_txt: str = page.get_text("text").strip()
                if len(native_txt) >= 30:
                    pages_text.append(
                        f"--- Page {page_idx + 1} ---\n{native_txt}"
                    )
                elif self.enable_ocr:
                    print(
                        f"    [OCR] {display_name} : page "
                        f"{page_idx + 1}/{total_pages}...",
                        flush=True,
                    )
                    pix = page.get_pixmap(dpi=150)
                    ocr_txt: str = self.run_ocr_on_bytes(pix.tobytes("png"))
                    if ocr_txt.strip():
                        pages_text.append(
                            f"--- Page {page_idx + 1} (OCR) ---\n"
                            f"{ocr_txt.strip()}"
                        )

            if not pages_text:
                return None, "PDF vide ou illisible"

            header: str = (
                "========================================\n"
                f"DOCUMENT PDF : {display_name}\n"
                "----------------------------------------\n"
            )
            return header + "\n\n".join(pages_text) + "\n\n", None
        except Exception as exc:
            return None, f"Erreur extraction PDF ({exc})"
        finally:
            if doc is not None:
                try:
                    doc.close()
                except Exception:
                    pass

    def extract_image(
        self, file_path: str
    ) -> Tuple[Optional[str], Optional[str]]:
        """
        Extrait le texte d'une image via RapidOCR.

        :param file_path: Chemin vers le fichier image.
        :return: Tuple (texte_formate, message_erreur).

            if not ocr_txt:

            header: str = (
                "========================================\n"
                f"IMAGE (OCR) : {file_name}\n"
                "----------------------------------------\n"
            )
            return header + ocr_txt + "\n\n", None
        except Exception as exc:
            return None, f"Erreur OCR image ({exc})"
    def extract_docx(
        self, file_path: str
    ) -> Tuple[Optional[str], Optional[str]]:
        """
        Extrait les paragraphes et tableaux d'un document Word (.docx).

        :param file_path: Chemin vers le fichier .docx.
        :return: Tuple (texte_formate, message_erreur).
        """
        file_name: str = os.path.basename(file_path)
        try:
            doc = DocxDocument(file_path)
            lines: List[str] = [
                p.text.strip() for p in doc.paragraphs if p.text.strip()
            ]
            for table in doc.tables:
                for row in table.rows:
                    cells: List[str] = [
                        c.text.strip().replace("\n", " ") for c in row.cells
                    ]
                    if any(cells):
                        lines.append(" | ".join(cells))
            if not lines:
                return None, "Document Word vide"
            header: str = (
                "========================================\n"
                f"DOCUMENT WORD : {file_name}\n"
                "----------------------------------------\n"
            )
            return header + "\n".join(lines) + "\n\n", None
        except Exception as exc:
            return None, f"Erreur extraction DOCX ({exc})"

    def extract_odt(
        self, file_path: str

        """
        Extrait le texte d'un document OpenDocument (.odt).

    def extract_xlsx(
        self, file_path: str
    ) -> Tuple[Optional[str], Optional[str]]:
        """
        Extrait le contenu des feuilles d'un classeur Excel (.xlsx).

        :return: Tuple (texte_formate, message_erreur).

        file_name: str = os.path.basename(file_path)

            wb = openpyxl.load_workbook(file_path, data_only=True)
            sheets_out: List[str] = []
            for sheet_name in wb.sheetnames:

                rows_txt: List[str] = []
                for row in ws.iter_rows(values_only=True):
                    vals: List[str] = [

                    ]
                    if any(vals):
                        rows_txt.append(" | ".join(vals))
                if rows_txt:
                    sheets_out.append(
                        f"--- Feuille : {sheet_name} ---\n"
                        + "\n".join(rows_txt)
                    )

            if not sheets_out:
                return None, "Classeur Excel vide"
            header: str = (
                "========================================\n"
                f"CLASSEUR EXCEL : {file_name}\n"
                "----------------------------------------\n"

    def extract_ics(
        self, file_path: str
    ) -> Tuple[Optional[str], Optional[str]]:
        """
        Extrait les événements d'un agenda iCalendar (.ics).

        :param file_path: Chemin vers le fichier .ics.
        :return: Tuple (texte_formate, message_erreur).
        """
        file_name: str = os.path.basename(file_path)
        try:
            with open(file_path, "r", encoding="utf-8", errors="ignore") as f:
                raw_ics: str = f.read()
            unfolded: str = re.sub(r"\r?\n[ \t]", "", raw_ics)
            events: List[str] = []
            current_ev: List[str] = []
            in_event: bool = False
            for line in unfolded.splitlines():
                if line.startswith("BEGIN:VEVENT"):
                    in_event = True
                    current_ev = []
                elif line.startswith("END:VEVENT"):
                    in_event = False
                    if current_ev:
                        events.append("\n".join(current_ev))
                elif in_event and any(
                    line.startswith(k)
                    for k in ("DTSTART", "DTEND", "SUMMARY", "DESCRIPTION", "LOCATION")
                ):
                    current_ev.append(line)
            if not events:
                return None, "Aucun événement ICS trouvé"
            header: str = (
                "========================================\n"
                f"AGENDA ICS : {file_name}\n"
                "----------------------------------------\n"
            )
            return header + "\n---\n".join(events) + "\n\n", None
        except Exception as exc:
            return None, f"Erreur extraction ICS ({exc})"

class DocAnonymizerPipelineRunner:
    """
    Ordonnanceur au fil de l'eau pour doc_anonymizer : anonymise d'abord les
    transcriptions en attente, extrait chaque document et lance son
    anonymisation, génère contacts_emails_decouverts.txt depuis
    tous_emails_trouves.txt puis relance une passe complète si enrichi.
    """

    def __init__(
        self, config: PipelineConfig, prevent_sleep: bool = True
    ) -> None:
        """
        Initialise l'ordonnanceur avec la configuration et le maintien d'éveil.

        :param config: Instance de PipelineConfig.
        :param prevent_sleep: Empêche la mise en veille OS durant l'exécution.
        """
        self.config: PipelineConfig = config
        self.prevent_sleep: bool = prevent_sleep

    def _anonymize_pending_transcriptions(
        self, anonymizer: DocumentAnonymizationRunner
    ) -> int:
        """
        Parcourt le dossier transcriptions et anonymise immédiatement tout
        fichier déjà extrait mais absent de l'état d'anonymisation.

        :param anonymizer: Instance de DocumentAnonymizationRunner initialisée.
        :return: Nombre de transcriptions en attente anonymisées.
        """
        trans_dir: str = self.config.transcriptions_dir
        if not os.path.isdir(trans_dir):
            return 0

        pending_files: List[str] = []
        for root_path, _, existing_files in os.walk(trans_dir):
            for file_name in sorted(existing_files):
                if (
                    file_name == "rapport_extraction.txt"
                    or file_name.startswith(".")
                    or not file_name.lower().endswith(".txt")
                ):
                    continue
                full_trans: str = os.path.join(root_path, file_name)
                norm_trans: str = os.path.normpath(full_trans)
                if norm_trans not in anonymizer.audit.already_processed:
                    pending_files.append(full_trans)

        if not pending_files:
            return 0

        print(
            f"Rattrapage : {len(pending_files)} transcription(s) "
            "en attente d'anonymisation...",
            flush=True,
        )
        processed_count: int = 0
        for idx, trans_path in enumerate(pending_files, start=1):
            if idx == 1 or idx % 25 == 0 or idx == len(pending_files):
                print(
                    f"    [Anonymisation {idx}/{len(pending_files)}] "
                    f"{os.path.basename(trans_path)}",
                    flush=True,
                )
            if anonymizer.anonymize_single_file(trans_path):
                processed_count += 1

        return processed_count

    def _reset_results_for_second_pass(self) -> None:
        """
        Nettoie intégralement les anciens lots et l'état JSON avant de relancer
        une passe complète avec les règles fraîchement découvertes.
        """
        res_dir: str = self.config.results_dir
        if os.path.isdir(res_dir):
            for entry in os.listdir(res_dir):
                full_p: str = os.path.join(res_dir, entry)
                if os.path.isdir(full_p) and entry.startswith("lot_"):
                    shutil.rmtree(full_p, ignore_errors=True)
                elif os.path.isfile(full_p) and entry.endswith(".txt"):
                    try:
                        os.remove(full_p)
                    except OSError:
                        pass
        if os.path.exists(self.config.resume_state_file):
            os.remove(self.config.resume_state_file)
    def _run_second_pass_with_discovered_emails(self) -> bool:
        """
        Génère contacts_emails_decouverts.txt depuis tous_emails_trouves.txt
        et exécute une seconde passe d'anonymisation si des règles ont été ajoutées.

        :return: True si une seconde passe a été exécutée, False sinon.
        """
        builder: DiscoveredEmailContactsBuilder = (
            DiscoveredEmailContactsBuilder(self.config)
        )
        added_count, total_count = builder.build_or_update()
        if added_count <= 0:
            return False

        pass_start: float = time.time()
        print(
            f"\n=== SECONDE PASSE D'ANONYMISATION ===\n"
            f"[CONTACTS DÉCOUVERTS] {added_count} nouvelle(s) règle(s) "
            f"ajoutée(s) dans '{self.config.discovered_contacts_file}' "
            f"(total : {total_count}).\n"
            "Relance complète de l'anonymisation sur les transcriptions...",
            flush=True,
        )
        self._reset_results_for_second_pass()
        second_anonymizer: DocumentAnonymizationRunner = (
            DocumentAnonymizationRunner(config=self.config)
        )
        if not second_anonymizer.initialize():
            return False

        self._anonymize_pending_transcriptions(second_anonymizer)
        second_anonymizer.finalize(pass_start)
        return True

    def run_all(self) -> None:
        """
        Exécute le rattrapage des transcriptions, l'extraction/anonymisation
        au fil de l'eau, puis la seconde passe avec les e-mails découverts.
        """
        start_time: float = time.perf_counter()
        wall_start: float = time.time()

        with SystemSleepInhibitor(enabled=self.prevent_sleep):
            print("=== PIPELINE DOC_ANONYMIZER (AU FIL DE L'EAU) ===", flush=True)
            print(f"[CONFIG] Fichier INI       : {self.config.config_path}", flush=True)
            print(f"[CONFIG] Contacts auto     : {self.config.contacts_file}", flush=True)
            print(f"[CONFIG] Contacts découverts : {self.config.discovered_contacts_file}", flush=True)
            print(f"[CONFIG] Contacts manuels  : {self.config.manual_contacts_file}", flush=True)
            print(f"[CONFIG] Sources           : {self.config.sources_dir}", flush=True)
            print(f"[CONFIG] Transcriptions    : {self.config.transcriptions_dir}", flush=True)
            print(f"[CONFIG] Résultats         : {self.config.results_dir}", flush=True)

            extractor: DocumentExtractionRunner = DocumentExtractionRunner(
                config=self.config
            )
            anonymizer: DocumentAnonymizationRunner = (
                DocumentAnonymizationRunner(config=self.config)
            )
            if not anonymizer.initialize():
                print(
                    "[ERREUR] Échec de l'initialisation de l'anonymiseur "
                    "(vérifiez contacts.txt). Arrêt du pipeline.",
                    flush=True,
                )
                return

            target_phones, allowed_emails = (
                extractor.load_all_filter_contacts()
            )
            if target_phones is None or allowed_emails is None:
                print(
                    "[ERREUR] Impossible de charger les filtres de contacts. "
                    "Arrêt du pipeline.",
                    flush=True,
                )
                return

            os.makedirs(self.config.transcriptions_dir, exist_ok=True)
            os.makedirs(self.config.results_dir, exist_ok=True)

            already_extracted: Set[str] = extractor.load_tracking_set(
                self.config.extraction_tracking_file
            )
            sms_processed: Set[str] = extractor.load_tracking_set(
                self.config.sms_tracking_file
            )
            discovered_phones: Set[str] = set()
            discovered_emails: Set[str] = set()
            try:
                self._anonymize_pending_transcriptions(anonymizer)

                files: List[str] = extractor.collect_source_files()
                total_files: int = len(files)
                print(
                    f"Inventaire : {total_files} fichier(s) source(s) "
                    "détecté(s).",
                    flush=True,
                )

                for idx, file_path in enumerate(files, start=1):
                    print(
                        f"\n[{idx}/{total_files}] Extraction : {file_path}",
                        flush=True,
                    )
                    generated_txts: List[str] = extractor.process_single_file(
                        file_path=file_path,
                        target_phones=target_phones,
                        allowed_emails=allowed_emails,
                        already_extracted=already_extracted,
                        sms_processed=sms_processed,
                        discovered_phones=discovered_phones,
                        discovered_emails=discovered_emails,
                    )
                    if not generated_txts:
                        print(
                            "    -> Aucune nouvelle transcription générée "
                            "(déjà extrait ou vide).",
                            flush=True,
                        )
                    for trans_txt in generated_txts:
                        print(
                            f"    -> Anonymisation immédiate : {trans_txt}",
                            flush=True,
                        )
                        ok: bool = anonymizer.anonymize_single_file(
                            trans_txt, force_reprocess=True
                        )
                        print(
                            f"    -> Résultat anonymisation : "
                            f"{'OK' if ok else 'IGNORÉ'} "
                            f"(tampon : {len(anonymizer.current_chunk)} fichier(s), "
                            f"{anonymizer.current_size} octets)",
                            flush=True,
                        )

                self._anonymize_pending_transcriptions(anonymizer)

                extractor.finalize_reports(discovered_phones, discovered_emails)
                anonymizer.finalize(wall_start)

                self._run_second_pass_with_discovered_emails()

            except KeyboardInterrupt:
                print(
                    "\n[INTERROMPU] Ctrl+C détecté : sauvegarde des lots "
                    "déjà anonymisés en cours...",
                    flush=True,
                )
                extractor.finalize_reports(discovered_phones, discovered_emails)
                anonymizer.flush_batch()
                raise

        elapsed: float = time.perf_counter() - start_time
        print(
            f"\nPipeline au fil de l'eau terminé en {elapsed:.1f}s.",
            flush=True,
        )

from __future__ import annotations
import ctypes
import os
import shutil
import subprocess
import sys
from typing import Any, Optional, Type


class SystemSleepInhibitor:
    """
    Gestionnaire de contexte multiplateforme empêchant la mise en veille
    ou l'extinction automatique sous Windows (SetThreadExecutionState)
    et sous Linux (systemd-inhibit) pendant un traitement long.
    """

    ES_CONTINUOUS: int = 0x80000000
    ES_SYSTEM_REQUIRED: int = 0x00000001

    def __init__(self, enabled: bool = True) -> None:
        """
        Initialise le contrôleur d'alimentation multiplateforme.

        :param enabled: Active ou désactive le blocage de la mise en veille.
        """
        self.enabled: bool = enabled
        self._win_active: bool = False
        self._linux_proc: Optional[subprocess.Popen[bytes]] = None

    def _enable_windows(self) -> None:
        """
        Active le maintien en éveil sous Windows via kernel32 avec typage c_uint.
        """
        try:
            flags: ctypes.c_uint = ctypes.c_uint(
                self.ES_CONTINUOUS | self.ES_SYSTEM_REQUIRED
            )
            res: int = ctypes.windll.kernel32.SetThreadExecutionState(flags)  # type: ignore[attr-defined]
            self._win_active = res != 0
            if self._win_active:
                print(
                    "[ALIMENTATION] Mise en veille Windows désactivée "
                    "pendant le traitement.",
                    flush=True,
                )
        except Exception as exc:
            print(
                f"[AVERTISSEMENT] Blocage mise en veille Windows : {exc}",
                flush=True,
            )

    def _enable_linux(self) -> None:
        """
        Active le maintien en éveil sous Linux via systemd-inhibit.
        """
        inhibit_bin: Optional[str] = shutil.which("systemd-inhibit")
        if not inhibit_bin:
            return
        try:
            self._linux_proc = subprocess.Popen(
                [
                    inhibit_bin,
                    "--what=idle:sleep",
                    "--who=doc_anonymizer",
                    "--why=Transcription et anonymisation en cours",
                    "--mode=block",
                    "sleep",
                    "infinity",
                ],
                stdout=subprocess.DEVNULL,
                stderr=subprocess.DEVNULL,
            )
            print(
                "[ALIMENTATION] Mise en veille Linux bloquée "
                "(systemd-inhibit).",
                flush=True,
            )
        except Exception as exc:
            self._linux_proc = None
            print(
                f"[AVERTISSEMENT] Blocage mise en veille Linux : {exc}",
                flush=True,
            )

    def __enter__(self) -> SystemSleepInhibitor:
        """
        Active le maintien en éveil selon le système d'exploitation.

        :return: Instance courante de SystemSleepInhibitor.
        """
        if not self.enabled:
            return self
        if os.name == "nt":
            self._enable_windows()
        elif sys.platform.startswith("linux"):
            self._enable_linux()
        return self

    def __exit__(
        self,
        exc_type: Optional[Type[BaseException]],
        exc_val: Optional[BaseException],
        exc_tb: Optional[Any],
    ) -> None:
        """
        Restaure les paramètres normaux d'alimentation à la sortie du bloc.
        """
        if self._win_active:
            try:
                ctypes.windll.kernel32.SetThreadExecutionState(  # type: ignore[attr-defined]
                    ctypes.c_uint(self.ES_CONTINUOUS)
                )
                print(
                    "[ALIMENTATION] Paramètres de mise en veille "
                    "Windows restaurés.",
                    flush=True,
                )
            except Exception:
                pass
            self._win_active = False

        if self._linux_proc is not None:
            try:
                self._linux_proc.terminate()
                self._linux_proc.wait(timeout=3)
                print(
                    "[ALIMENTATION] Verrou systemd-inhibit Linux libéré.",
                    flush=True,
                )
            except Exception:
                try:
                    self._linux_proc.kill()
                except Exception:
                    pass
            self._linux_proc = None

    @staticmethod
    def atomic_write_text(target_path: str, content: str) -> None:
        """
        Écrit un fichier texte de façon atomique via un fichier temporaire
        dans le même répertoire afin d'éviter toute corruption en cas d'arrêt.

        :param target_path: Chemin final du fichier à écrire.
        :param content: Contenu UTF-8 complet à enregistrer.
        """
        parent_dir: str = os.path.dirname(os.path.abspath(target_path))
        if parent_dir:
            os.makedirs(parent_dir, exist_ok=True)
        fd, tmp_path = tempfile.mkstemp(
            dir=parent_dir, prefix=".tmp_write_", suffix=".tmp"
        )
        try:
            with os.fdopen(fd, "w", encoding="utf-8", newline="\n") as f_tmp:
                f_tmp.write(content)
            os.replace(tmp_path, target_path)
        except Exception:
            if os.path.exists(tmp_path):
                try:
                    os.remove(tmp_path)
                except OSError:
                    pass
            raise

    @staticmethod
    def strip_accents(text: str) -> str:
        """
        Retire les diacritiques et accents d'une chaîne Unicode.

        :param text: Texte source.
        :return: Texte sans accents.
        """
        return "".join(
            ch
            for ch in unicodedata.normalize("NFD", text)
            if unicodedata.category(ch) != "Mn"
        )

    @classmethod
    def normalize_word(cls, word: str) -> str:
        """
        Normalise un mot en minuscules sans accents ni espaces superflus.

        :param word: Mot brut.
        :return: Mot normalisé.
        """
        return cls.strip_accents(word).lower().strip()

    @staticmethod
    def normalize_identifier(value: str) -> str:
        """
        Uniformise un numéro de téléphone français (format 0XXXXXXXXX)
        ou international ainsi qu'une adresse e-mail.

        :param value: Identifiant brut extrait.
        :return: Identifiant normalisé.
        """
        val: str = re.sub(r"^mailto:", "", value.strip(), flags=re.IGNORECASE)
        if "@" in val:
            local_part, domain_part = val.lower().split("@", 1)
            if "+" in local_part:
                local_part = next(iter(local_part.split("+", 1)), local_part)
            return f"{local_part}@{domain_part}"
        num: str = re.sub(r"[\s.\-\u00a0()]", "", val)
        if num.startswith("+330"):
            num = "0" + num[4:]
        elif num.startswith("+33"):
            num = "0" + num[3:]
        elif num.startswith("00330"):
            num = "0" + num[5:]
        elif num.startswith("0033"):
            num = "0" + num[4:]
        return num
    @classmethod
    def split_sms_addresses(cls, raw_address: str) -> List[str]:
        """
        Découpe un attribut address SMS/MMS (y compris les groupes séparés
        par '~' ou concaténés via '+33...') en identifiants individuels normalisés.

        :param raw_address: Chaîne brute de l'attribut address du XML.
        :return: Liste des numéros ou adresses individuels normalisés.
        """
        if not raw_address:
            return []
        expanded: str = re.sub(r"(?<=\d)\+(?=\d)", "~+", raw_address.strip())
        parts: List[str] = re.split(r"[~;,]+", expanded)
        results: List[str] = []
        for part in parts:
            norm: str = cls.normalize_identifier(part)
            if norm and norm not in results:
                results.append(norm)
        return results

    @classmethod
    def is_contact_identifier(cls, value: str) -> bool:
        """
        Vérifie si une chaîne correspond à un e-mail ou un téléphone.

        :param value: Chaîne brute à tester.
        :return: True s'il s'agit d'un e-mail ou d'un numéro valide.
        """
        cleaned: str = value.strip()
        if not cleaned:
            return False
        if "@" in cleaned:
            return True
        norm: str = cls.normalize_identifier(cleaned)
        return bool(re.match(r"^[\d+]{6,}$", norm))

    @classmethod
    def resolve_contact_pair(
        cls, left_part: str, right_part: str
    ) -> Tuple[str, str, bool]:
        """
        Détermine automatiquement quelle partie est la cible à masquer
        et quelle partie est l'alias, quel que soit le sens autour du '='.

        :param left_part: Partie gauche avant le séparateur.
        :param right_part: Partie droite après le séparateur.
        :return: Tuple (cible_a_masquer, alias, est_alias_a_gauche).
        """
        left_clean: str = left_part.strip()
        right_clean: str = right_part.strip()
        if cls.is_contact_identifier(
            right_clean
        ) and not cls.is_contact_identifier(left_clean):
            return right_clean, left_clean, True
        return left_clean, right_clean, False

    @classmethod
    def mask_identifiers_in_filename(
        cls,
        file_name: str,
        phone_map: Dict[str, str],
        email_map: Optional[Dict[str, str]] = None,
    ) -> str:
        """
        Remplace tout numéro de téléphone, adresse e-mail, URL, NIR ou IBAN
        présent dans un nom de fichier par son alias ou une balise masquée.

        :param file_name: Nom du fichier d'origine.
        :param phone_map: Dictionnaire numéro normalisé -> alias.
        :param email_map: Dictionnaire optionnel e-mail normalisé -> alias.
        :return: Nom de fichier assaini.
        """
        def _replace_phone(match: re.Match[str]) -> str:
            norm_num: str = cls.normalize_identifier(match.group(0))
            return phone_map.get(norm_num, "NUMERO_MASQUE")

        def _replace_email(match: re.Match[str]) -> str:
            norm_mail: str = cls.normalize_identifier(match.group(0))
            if email_map and norm_mail in email_map:
                return email_map[norm_mail]
            return "EMAIL_MASQUE"

        masked: str = cls.REGEX_SSN.sub("NUM_SECU_MASQUE", file_name)
        masked = cls.REGEX_IBAN.sub("IBAN_MASQUE", masked)
        masked = cls.REGEX_EMAIL.sub(_replace_email, masked)
        masked = cls.REGEX_PHONE.sub(_replace_phone, masked)
        return cls.REGEX_URL.sub("URL_MASQUEE", masked)

    @staticmethod
    def html_to_plain_text(raw_html: str) -> str:
        """
        Convertit un contenu HTML en texte brut structuré.

        :param raw_html: Chaîne HTML source.
        :return: Texte lisible nettoyé des balises.
        """
        txt: str = re.sub(r"(?is)<(script|style).*?>.*?</\1>", "", raw_html)
        txt = re.sub(r"(?i)<br\s*/?>", "\n", txt)
        txt = re.sub(r"(?i)</(p|div|tr|li|h[1-6])>", "\n", txt)
        txt = re.sub(r"<[^>]+>", " ", txt)
        txt = html.unescape(txt)
        lines: List[str] = [
            re.sub(r"[ \t]+", " ", line).strip() for line in txt.splitlines()
        ]
        return "\n".join(line for line in lines if line)
    @classmethod
    def detect_period_in_text(cls, text: str) -> Optional[str]:
        """
        Recherche les dates valides dans un texte ou nom de fichier (y compris
        les horodatages compacts AAAAMMJJHHMMSS) et retourne la plus ancienne AAAA-MM.

        :param text: Contenu textuel ou nom de fichier à analyser.
        :return: Chaîne 'AAAA-MM' si trouvée, sinon None.
        """
        if not text:
            return None
        sample: str = cls.strip_accents(text[:15000]).lower()
        candidates: List[str] = []

        for m in re.finditer(
            r"(?<!\d)(199\d|20[0-3]\d)[-/. _]"
            r"(0[1-9]|1[0-2])[-/. _](0[1-9]|[12]\d|3[01])(?!\d)",
            sample,
        ):
            candidates.append(f"{m.group(1)}-{m.group(2)}")

        for m in re.finditer(
            r"(?<!\d)(0[1-9]|[12]\d|3[01])[-/. _]"
            r"(0[1-9]|1[0-2])[-/. _](199\d|20[0-3]\d)(?!\d)",
            sample,
        ):
            candidates.append(f"{m.group(3)}-{m.group(2)}")

        for m in re.finditer(
            r"(?<!\d)(199\d|20[0-3]\d)(0[1-9]|1[0-2])(0[1-9]|[12]\d|3[01])"
            r"(?:(?:[01]\d|2[0-3])[0-5]\d[0-5]\d)?(?!\d)",
            sample,
        ):
            candidates.append(f"{m.group(1)}-{m.group(2)}")

        months_pattern: str = "|".join(cls.FRENCH_MONTHS.keys())
        for m in re.finditer(
            rf"(?<!\d)\d{{1,2}}(?:er)?[\s_-]+({months_pattern})\.?"
            rf"[\s_-]+(199\d|20[0-3]\d)(?!\d)",
            sample,
        ):
            month_num: Optional[str] = cls.FRENCH_MONTHS.get(m.group(1))
            if month_num:
                candidates.append(f"{m.group(2)}-{month_num}")

        for m in re.finditer(
            rf"(?<![a-z])({months_pattern})\.?[\s_-]+(199\d|20[0-3]\d)(?!\d)",
            sample,
        ):
            month_num = cls.FRENCH_MONTHS.get(m.group(1))
            if month_num:
                candidates.append(f"{m.group(2)}-{month_num}")

        return min(candidates) if candidates else None

    @classmethod
    def resolve_file_period(
        cls, extracted_text: str, file_name: str, file_path: str
    ) -> str:
        """
        Détermine la période AAAA-MM par priorité : nom de fichier d'abord,
        puis contenu extrait, puis date de modification disque.

        :param extracted_text: Texte extrait du document.
        :param file_name: Nom du fichier.
        :param file_path: Chemin complet sur le disque.
        :return: Chaîne 'AAAA-MM' ou '0000-00'.
        """
        period_name: Optional[str] = cls.detect_period_in_text(file_name)
        if period_name:
            return period_name
        period: Optional[str] = cls.detect_period_in_text(extracted_text)
        if period:
            return period
        try:
            timestamp: float = os.path.getmtime(file_path)
            return datetime.datetime.fromtimestamp(timestamp).strftime("%Y-%m")
        except Exception:
            return "0000-00"

    @classmethod
    def load_filter_contacts(
        cls, contacts_path: str
    ) -> Tuple[Optional[Dict[str, str]], Optional[Set[str]]]:
        """
        Charge les numéros de téléphone et adresses e-mail indexés depuis
        contacts.txt, quel que soit le sens autour du signe '='.

        :param contacts_path: Chemin vers le fichier contacts.txt.
        :return: Tuple (dictionnaire_telephones, ensemble_emails) ou None.
        """
        phones: Dict[str, str] = {}
        emails: Set[str] = set()
        if not os.path.exists(contacts_path):
            print(f"Erreur : '{contacts_path}' introuvable.", flush=True)
            return None, None
        with open(contacts_path, "r", encoding="utf-8") as f:
            for raw_line in f:
                line: str = raw_line.strip()
                if not line or line.startswith("#"):
                    continue
                for sep in ("=", ";", ","):
                    if sep in line:
                        left_part, right_part = line.split(sep, 1)
                        target_val, alias_val, _ = cls.resolve_contact_pair(
                            left_part, right_part
                        )
                        if target_val and alias_val:
                            norm_id: str = cls.normalize_identifier(target_val)
                            if "@" in norm_id:
                                emails.add(norm_id)
                            elif re.match(r"^[\d+]+$", norm_id):
                                phones[norm_id] = alias_val
                        break
        return phones, emails
